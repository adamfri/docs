<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Translation content ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="schemaId" type="string">

      Translation schema ID.

      Constraints: `format GUID`, `immutable`.

    </ResponseField>

    <ResponseField name="entityId" type="string">

      A unique identifier defined by the app developer to differentiate translation content for various entities within the site. For example, if an app developer is creating translation content for blog posts, the `entityId` can be `'blog-posts-1'`.

      Constraints: `minLength 1`, `maxLength 100`, `immutable`.

    </ResponseField>

    <ResponseField name="locale" type="string">

      Translation content locale.

      Constraints: `format LANGUAGE_TAG`, `immutable`.

    </ResponseField>

    <ResponseField name="fields" type="object">

      List of fields for the translation content. This property uses a string to map to a `ContentField` (`Map<string, ContentField>`). The string serves as a key, which you'll need to access each field in the content and when referencing a translation schema.

      Constraints: `format map`, `minItems 0`, `maxItems 2000`.

      <Expandable title="fields properties">

        <ResponseField name="key" type="string" />

        <ResponseField name="value" type="ContentField">

          <Expandable title="value properties">

            <ResponseField name="id" type="string">

              Field unique identifier. Validated according to Regex `^[A-Za-z0-9-_)(]+$`. Field IDs may contain parentheses to reference repeated items, such as images. These parentheses can't be nested and each opening parenthesis must be followed by a closing one. For example, `title()()` is valid, while `title)(` and `title(())` aren't valid. The value inside the parenthesis is validated according to Regex `^[^)(\\]\\[\\.]+$`.

              **Read-only.** Constraints: `minLength 1`, `maxLength 256`.

            </ResponseField>

            <ResponseField name="published" type="boolean">

              Whether to publish the translation content to the live site, making it visible to site visitors.

              Default: `false`

            </ResponseField>

            <ResponseField name="updatedBy" type="string">

              **Required.** The source that updated the translation content.

              **Supported values:**

              - `UNKNOWN_UPDATER_IDENTITY` — Unknown updater.
              - `USER` — The Wix user.
              - `EXTERNAL_APP` — An external translation app.
              - `MACHINE` — Machine translation service.

            </ResponseField>

            <ResponseField name="schemaFieldKey" type="string">

              Unique string identifier that links a specific content field to its corresponding field in the schema.

              **Read-only.** Constraints: `maxLength 100`.

            </ResponseField>

            <ResponseField name="index" type="integer">

              Field index. Used for cases where the order of the fields are relevant.

              **Read-only.** Constraints: `min 0`.

            </ResponseField>

          </Expandable>

          <Expandable title="One of the following">

            <ResponseField name="value" type="value">

              value

              <Expandable title="value properties">

                <ResponseField name="textValue" type="string">

                  Plain text. Use this field if the corresponding schema `fields.type` is `SHORT_TEXT`, `LONG_TEXT`, `HTML`, or `IMAGE_LINK`.

                  Constraints: `maxLength 100000`.

                </ResponseField>

                <ResponseField name="richContent" type="RichContent">

                  Translation Content rich content.

                  &lt;widget src="https://apps.wix.com/_serverless/ricos-playground-services/goto/api-component" exampleid="71a5e527-a334-49e6-bc2d-9f5b3e831fc5">See Ricos document reference&lt;/widget>

                  _Nested structure omitted for brevity — see the `RichContent` object._

                </ResponseField>

                <ResponseField name="image" type="Image">

                  Wix Media Manager image.

                  _Nested structure omitted for brevity — see the `Image` object._

                </ResponseField>

                <ResponseField name="video" type="VideoV2">

                  Wix Media Manager video.

                  _Nested structure omitted for brevity — see the `VideoV2` object._

                </ResponseField>

                <ResponseField name="document" type="Document">

                  Wix Media Manager document.

                  _Nested structure omitted for brevity — see the `Document` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="parentEntityId" type="string">

      A reference to the parent content. For example, if the content is a menu item, this property would contain the `entityId` of the menu it belongs to.

    </ResponseField>

    <ResponseField name="publishStatus" type="string">

      The aggregated published status across all content fields.

      **Read-only.**

      **Supported values:**

      - `UNKNOWN_PUBLISH_STATUS` — Unknown publish status.
      - `UNPUBLISHED` — None of the fields have been sent for publication.
      - `PARTIALLY_PUBLISHED` — Some of the fields have been sent for publication.
      - `PUBLISHED` — All of the fields have been sent for publication.

    </ResponseField>

    <ResponseField name="previewField" type="string">

      Contains the value of the preview field if the schema defines one.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the translation content was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the translation content was updated.

      **Read-only.**

    </ResponseField>
