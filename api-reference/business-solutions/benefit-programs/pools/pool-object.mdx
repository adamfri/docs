A pool contains benefits that can be redeemed by the pool's beneficiary.

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Pool ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the pool is updated.
      To prevent conflicting changes, the current revision must be passed when updating the pool.

      Ignored when creating a pool.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the pool was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the pool was updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="poolDefinitionId" type="string">

      ID of the associated pool definition.

      **Read-only.**

    </ResponseField>

    <ResponseField name="programDefinitionId" type="string">

      ID of the associated program definition.

      **Read-only.**

    </ResponseField>

    <ResponseField name="programId" type="string">

      ID of the program that contains this pool.

      **Read-only.**

    </ResponseField>

    <ResponseField name="status" type="string">

      Pool status.

      **Read-only.**

      **Supported values:**

      - `ACTIVE` — Pool is active.
      - `PAUSED` — Pool is paused. It can be resumed.
      - `ENDED` — Pool is ended. It can't be resumed.
      - `PROVISIONING` — Pool provisioning is in progress. This status exists for a short time during processing.
      - `RENEWING` — Pool renewing is in progress. This status exists for a short time during processing.
      - `PENDING` — Pool is pending

    </ResponseField>

    <ResponseField name="beneficiary" type="IdentificationData">

      Pool beneficiary.

      **Read-only.** Constraints: `immutable`.

      <Expandable title="One of the following">

        <ResponseField name="id" type="id">

          id

          <Expandable title="id properties">

            <ResponseField name="anonymousVisitorId" type="string">

              ID of a site visitor that hasn't logged in to the site.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="memberId" type="string">

              ID of a site member.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="wixUserId" type="string">

              ID of a Wix user.

              Constraints: `format GUID`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="details" type="Details">

      Pool benefits and settings.

      <Expandable title="details properties">

        <ResponseField name="benefits" type="Benefit[]">

          List of benefits.

          Constraints: `maxItems 10`.

          <Expandable title="benefits properties">

            <ResponseField name="benefitKey" type="string">

              Unique identifier for this benefit.

              This key is consistent across the pool definition and all associated pools that contain this benefit.

              Constraints: `maxLength 64`.

            </ResponseField>

            <ResponseField name="itemSetId" type="string">

              ID that is used to associated items with this benefit.

              **Read-only.**

            </ResponseField>

            <ResponseField name="price" type="string">

              Price of the benefit in credits. The price is the same for all of this benefit's items.

            </ResponseField>

            <ResponseField name="additionalData" type="object">

              Additional information for this benefit.

            </ResponseField>

            <ResponseField name="providerAppId" type="string">

              ID of the app providing the benefit items.

            </ResponseField>

            <ResponseField name="displayName" type="string">

              Benefit name.

            </ResponseField>

            <ResponseField name="description" type="string">

              Benefit description.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="creditConfiguration" type="CreditConfiguration">

          Credit settings.

          If this object is empty, you can't set a price for the benefit.

          <Expandable title="creditConfiguration properties">

            <ResponseField name="amount" type="string">

              Initial available amount for associated balances.

              Constraints: `>= 0`.

            </ResponseField>

            <ResponseField name="rolloverConfiguration" type="RolloverConfiguration">

              Rollover settings.

              <Expandable title="rolloverConfiguration properties">

                <ResponseField name="enabled" type="boolean">

                  Whether unused credits roll over to a new cycle when a program renews.

                </ResponseField>

                <ResponseField name="balanceCap" type="string">

                  Maximum number of credits that can roll over to the next cycle when a program renews.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="unitDisplayName" type="string">

              Credit unit display name.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="additionalData" type="object">

          Additional information relating to this object.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="displayName" type="string">

      Pool name.

      You may want to use the same name that's used in the associated pool definition.

      Constraints: `maxLength 64`.

    </ResponseField>

    <ResponseField name="namespace" type="string">

      Namespace for your app or site's benefit programs. Namespaces allow you to distinguish between entities that you created and entities that other apps created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Custom field data for the pool object.

      [Extended fields](https://dev.wix.com/docs/build-apps/develop-your-app/extensions/backend-extensions/schema-plugins/about-schema-plugin-extensions) must be configured in the app dashboard before they can be accessed with API calls.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="programDefinition" type="ProgramDefinitionInfo">

      Associated program definition information.

      **Read-only.**

      <Expandable title="programDefinition properties">

        <ResponseField name="id" type="string">

          Program definition ID.

          **Read-only.** Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="externalId" type="string">

          Program definition external ID.

          **Read-only.**

        </ResponseField>

        <ResponseField name="displayName" type="string">

          Display name of the program definition.

          **Read-only.**

        </ResponseField>

        <ResponseField name="description" type="string">

          Description of the program definition.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="program" type="ProgramInfo">

      Information about the program containing the pool.

      **Read-only.**

      <Expandable title="program properties">

        <ResponseField name="id" type="string">

          Program ID.

          **Read-only.** Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="externalId" type="string">

          Program external ID.

          **Read-only.**

        </ResponseField>

        <ResponseField name="displayName" type="string">

          Display name of the program.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="poolDefinitionRevision" type="integer">

      Version of the pool definition that this pool was created from.

      **Read-only.**

    </ResponseField>

    <ResponseField name="renewalCount" type="integer">

      Number of times this pool has been renewed.

      **Read-only.**

    </ResponseField>
