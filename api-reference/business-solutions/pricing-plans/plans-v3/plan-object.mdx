<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Plan ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the plan is updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the plan was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the plan was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Plan name displayed to customers.

      Constraints: `minLength 1`, `maxLength 50`.

    </ResponseField>

    <ResponseField name="description" type="string">

      Plan description that explains what customers get with this plan.

    </ResponseField>

    <ResponseField name="image" type="Image">

      Plan image displayed during checkout and on the site's **Plans & Pricing** page.

      <Expandable title="image properties">

        <ResponseField name="id" type="string">

          WixMedia image ID.

          Constraints: `maxLength 255`.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image height.

          **Read-only.**

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image width.

          **Read-only.**

        </ResponseField>

        <ResponseField name="altText" type="string">

          Image alt text.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="slug" type="string">

      Unique identifier for this plan within the Wix site. If not provided, generated automatically from the plan name.

    </ResponseField>

    <ResponseField name="termsAndConditions" type="string">

      Terms and conditions text that customers must agree to when purchasing this plan.

    </ResponseField>

    <ResponseField name="maxPurchasesPerBuyer" type="integer" deprecated>

      Deprecated. Use `purchase_limits` with `PER_MEMBER_LIFETIME` type instead.
      Maximum number of times the same customer can purchase this plan. Use `0` for unlimited purchases or `1` to limit to one purchase per customer.

      **Deprecated.** Deprecated. Use `purchase_limits` with `PER_MEMBER_LIFETIME` type instead.
      Maximum number of times the same customer can purchase this plan. Use `0` for unlimited purchases or `1` to limit to one purchase per customer. Use `purchase_limits` instead. Removal: 2026-11-30. Constraints: `min 0`, `max 10`.

    </ResponseField>

    <ResponseField name="pricingVariants" type="PricingVariant[]">

      Pricing variants that define different billing and pricing options for this plan. Currently limited to 1 variant per plan.

      Constraints: `minItems 1`, `maxItems 20`.

      <Expandable title="pricingVariants properties">

        <ResponseField name="id" type="string">

          Pricing variant ID.

          Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="name" type="string">

          Name for this pricing variant, such as "Monthly" or "Yearly".

          Constraints: `minLength 1`, `maxLength 255`.

        </ResponseField>

        <ResponseField name="freeTrialDays" type="integer">

          Number of days customers can use the plan for free before the first charge. Set to `0` for no free trial.

          Constraints: `min 0`, `max 999`.

        </ResponseField>

        <ResponseField name="fees" type="Fee[]">

          Additional fees applied to this pricing variant.

          Constraints: `maxItems 10`.

          <Expandable title="fees properties">

            <ResponseField name="id" type="string">

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="priceType" type="string">

              Pricing type used for this fee.

              **Supported values:**

              - `FIXED_AMOUNT` — Fee is charged as a fixed monetary amount.

            </ResponseField>

            <ResponseField name="name" type="string">

              Fee name displayed to customers, such as "Setup Fee" or "Processing Fee".

              Constraints: `minLength 1`, `maxLength 40`.

            </ResponseField>

            <ResponseField name="appliedAt" type="string">

              When this fee is charged during the customer's plan lifecycle.

              **Supported values:**

              - `FIRST_PAYMENT` — Fee is added to the first payment.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="billingTerms" type="BillingTerms">

          Billing configuration that defines when and how often customers are charged.

          <Expandable title="billingTerms properties">

            <ResponseField name="billingCycle" type="Duration">

              Duration of each billing cycle, such as 1 month or 2 weeks. Can't be less than 7 days. Only applicable for recurring plans.

              <Expandable title="billingCycle properties">

                <ResponseField name="period" type="string">

                  Duration of period.

                  **Supported values:**

                  - `DAY`
                  - `WEEK`
                  - `MONTH`
                  - `YEAR`

                </ResponseField>

                <ResponseField name="count" type="string">

                  Number of periods.

                  Constraints: `min 1`.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="startType" type="string">

              When the billing starts.

              **Supported values:**

              - `ON_PURCHASE` — Billing starts immediately after purchase.
              - `CUSTOM` — Customer can choose when billing starts.

            </ResponseField>

            <ResponseField name="endType" type="string">

              How long the plan remains active. Plans can run until canceled or for a specific number of billing cycles. Plan duration can't exceed 10 years.

              **Supported values:**

              - `UNTIL_CANCELLED` — Plan continues until customer cancels.
              - `CYCLES_COMPLETED` — Plan ends after a specified number of billing cycles.

            </ResponseField>

          </Expandable>

          <Expandable title="One of the following">

            <ResponseField name="endOption" type="endOption">

              endOption

              <Expandable title="endOption properties">

                <ResponseField name="cyclesCompletedDetails" type="CyclesCompletedDetails">

                  How many billing cycles the plan lasts.

                  _Nested structure omitted for brevity — see the `CyclesCompletedDetails` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="promotion" type="string">

          Promotional message displayed with this pricing variant.

        </ResponseField>

        <ResponseField name="pricingStrategies" type="PricingStrategy[]">

          Pricing strategies that determine how much customers pay. Currently limited to 1 strategy per variant.

          Constraints: `minItems 1`, `maxItems 1`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="perks" type="Perk[]">

      List of text snippets describing what the plan offers. For display purposes only.

      Constraints: `maxItems 500`.

      <Expandable title="perks properties">

        <ResponseField name="id" type="string">

          Perk ID.

          Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="description" type="string">

          Perk description.

          Constraints: `minLength 1`, `maxLength 1400`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="visibility" type="string">

      Plan visibility.

      **Supported values:**

      - `PUBLIC` — Plan is visible and available for purchase by all customers.
      - `PRIVATE` — Plan isn't visible to new customers unless they get a specific link to the plan. Existing buyers can continue using it.

    </ResponseField>

    <ResponseField name="buyable" type="boolean">

      Whether customers can currently purchase this plan. If set to `false`, customer won't be able to buy a plan themselves. In this case a Wix user has to assign the plan through the [dashboard](https://support.wix.com/en/article/pricing-plans-selling-plans-offline).

    </ResponseField>

    <ResponseField name="formId" type="string">

      ID of the form associated with the plan at checkout.

    </ResponseField>

    <ResponseField name="buyerCanCancel" type="boolean">

      Whether buyers can cancel their subscription to this plan.

    </ResponseField>

    <ResponseField name="currency" type="string">

      Three-letter currency code in [ISO-4217 alphabetic](https://en.wikipedia.org/wiki/ISO_4217#Active_codes) format.
      When the site currency changes, monetary amounts stored on the plan (prices, fees, and override amounts)
      are automatically rounded to the number of decimal places supported by the new currency.

      **Read-only.** Constraints: `format CURRENCY`.

    </ResponseField>

    <ResponseField name="purchaseLimits" type="PurchaseLimit[]">

      Caps how many subscriptions buyers can have or the site can sell for this plan.
      When multiple types are present, all must pass (AND logic).
      Duplicate types are rejected by the server.

      **Developer Preview.** Constraints: `maxItems 4`.

      <Expandable title="purchaseLimits properties">

        <ResponseField name="type" type="string">

          The type of purchase limit.

          **Supported values:**

          - `PER_MEMBER_LIFETIME` — Limits the total number of subscriptions (any status) a single member can have over their lifetime.
          - `PER_MEMBER_ACTIVE` — Limits the number of ongoing (ACTIVE, PENDING, PAUSED) subscriptions a single member can have at any time.
          - `TOTAL_ACTIVE` — Limits the total number of ongoing (ACTIVE, PENDING, PAUSED) subscriptions across all members at any time.
          - `TOTAL_SOLD` — Limits the total number of subscriptions ever created (any status) across all members.

        </ResponseField>

        <ResponseField name="maxCount" type="integer">

          The maximum number of subscriptions allowed.

          Constraints: `min 1`, `max 10000`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Data extensions. Learn more about [extended fields](https://dev.wix.com/docs/build-apps/develop-your-app/extensions/backend-extensions/schema-plugins/about-schema-plugin-extensions).

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>
