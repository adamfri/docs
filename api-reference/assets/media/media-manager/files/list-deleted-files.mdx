    <div className="wix-ref">

    Retrieves a list of files in the Media Manager's trash bin. <br />

    <Note>

    The Media Manager's trash bin (`TRASH-ROOT` folder) only contains temporarily deleted files, not permanently deleted files.

    </Note>

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-MEDIA.READ-MEDIAMANAGER">Read Media Manager</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    GET https://www.wixapis.com/site-media/v1/trash-bin/files
    ```

    ## Query Params

    <ParamField query="parentFolderId" type="string">

      ID of the file's parent folder. &lt;br />
      Default: `media-root`.

    </ParamField>

    <ParamField query="mediaTypes" type="string[]">

      File media type.

      Constraints: `maxItems 30`.

      **Supported values:**

      - `IMAGE`
      - `VIDEO`
      - `AUDIO`
      - `DOCUMENT`
      - `VECTOR`
      - `ARCHIVE`
      - `MODEL3D`
      - `OTHER`

    </ParamField>

    <ParamField query="private" type="boolean">

      \n`true`: Returns only private files. \n`false`: Returns only public files. \n`undefined`: Returns public and private files. \n Learn more [about private files](https://dev.wix.com/docs/api-reference/assets/media/media-manager/files/private-files).

    </ParamField>

    <ParamField query="sort" type="Sorting">

      Field name and order to sort by. One of: &lt;br />
      * `displayName`
      * `updatedDate`
      * `sizeInBytes`
      Default: `updatedDate` in `desc` order.

      <Expandable title="sort properties">

        <ParamField query="fieldName" type="string">

          Name of the field to sort by.

          Constraints: `maxLength 512`.

        </ParamField>

        <ParamField query="order" type="string">

          Sort order.

          **Supported values:**

          - `ASC`
          - `DESC`

        </ParamField>

        <ParamField query="origin" type="AddressLocation">

          Origin point for geo-distance sorting on a GEO field
          results are ordered by distance from this point (ASC = nearest first, DESC = farthest first).

          <Expandable title="origin properties">

            <ParamField query="latitude" type="number">

              Address latitude.

            </ParamField>

            <ParamField query="longitude" type="number">

              Address longitude.

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField query="paging" type="CursorPaging">

      Cursor and paging information.

      <Expandable title="paging properties">

        <ParamField query="limit" type="integer">

          Maximum number of items to return in the results.

        </ParamField>

        <ParamField query="cursor" type="string">

          Pointer to the next or previous page in the list of results.

          Pass the relevant cursor token from the `pagingMetadata` object in the previous call's response.
          Not relevant for the first request.

        </ParamField>

      </Expandable>

    </ParamField>

    ## Response Object

    <ResponseField name="files" type="FileDescriptor[]">

      List of files in the Media Manager's trash bin.

      Constraints: `maxItems 200`.

      See the [File Descriptor Object](/api-reference/assets/media/media-manager/files/file-descriptor-object) for this object's fields.

    </ResponseField>

    <ResponseField name="nextCursor" type="PagingMetadataV2">

      The next cursor if it exists.

      <Expandable title="nextCursor properties">

        <ResponseField name="total" type="integer">

          Total number of items that match the query. Returned if offset paging is used and the `tooManyToCount` flag is not set.

        </ResponseField>

        <ResponseField name="cursors" type="Cursors">

          Cursors to navigate through the result pages using `next` and `prev`. Returned if cursor paging is used.

          <Expandable title="cursors properties">

            <ResponseField name="next" type="string">

              Cursor string pointing to the next page in the list of results.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    ## Errors

    | Status | Code | Application code | Description |
    | --- | --- | --- | --- |
    | `400` | `INVALID_ARGUMENT` | `UNSUPPORTED_MEDIA_TYPE` | Media type isn't supported for filtering. |
    | `400` | `INVALID_ARGUMENT` | `UNSUPPORTED_SORT_VALUE` | Sort value isn't supported. |

    <Note>

      This method may also return standard errors. [Learn more about standard Wix errors](https://dev.wix.com/docs/rest/articles/getting-started/errors).

    </Note>

    ```javascript Bulk restores all non-permanently deleted files of chosen types (with $w)
    /***********************************************
     * Backend code - bulk-restore-files.web.js/ts *
     **********************************************/
    import { Permissions, webMethod } from '@wix/web-methods';
    import { files } from '@wix/media';
    import { auth } from '@wix/essentials';

    export const restoreFiles = webMethod(Permissions.Anyone, async (mediaTypes) => {
      let idList = []
      try {
        const options = {
            mediaTypes: mediaTypes
        };
        const elevatedListDeletedFiles = auth.elevate(files.listDeletedFiles)
        const deletedFiles = await elevatedListDeletedFiles(options);

        idList = deletedFiles.files.map((file) => {
         return file._id
        });
      } catch (error) {
        console.error(error);
      }

      try {
        const elevatedBulkRestoreFilesFromTrashBin = auth.elevate(files.bulkRestoreFilesFromTrashBin);
        const restoredFiles = await elevatedBulkRestoreFilesFromTrashBin(idList);

        console.log('Restored Files:', restoredFiles);
        return restoredFiles;
      } catch (error) {
        console.error(error);
      }
    });

    /*************
     * Page code *
     ************/

    import { restoreFiles } from 'backend/bulk-restore-files.web';

    $w.onReady(() => {
      $w('#restoreMedia').onClick(async () => {
        const mediaTypes = []
        if ($w('#image').checked) {
            mediaTypes.push("IMAGE")
        }
        if ($w('#video').checked) {
            mediaTypes.push("VIDEO")
        }
        if ($w('#audio').checked) {
            mediaTypes.push("AUDIO")
        }
        if ($w('#document').checked) {
            mediaTypes.push("DOCUMENT")
        }
        if ($w('#vector').checked) {
            mediaTypes.push("VECTOR")
        }
        if ($w('#archive').checked) {
            mediaTypes.push("ARCHIVE")
        }
        if ($w('#model3d').checked) {
            mediaTypes.push("MODEL3D")
        }
        await restoreFiles(mediaTypes);

        $w('#filesRestoredMsg').show();
        setTimeout(() => {
          $w('#filesRestoredMsg').hide();
        }, 5000);
      });
    ```
