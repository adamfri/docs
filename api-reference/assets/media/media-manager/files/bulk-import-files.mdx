      **Deprecated.** This method is deprecated.

    Imports a bulk of files to the Media Manager using external urls. <br />

    Returns information about the imported files.

    Use the `parentFolderId` and `filePath` parameters to specify the folder you want each file to be imported to.
    If no folder is specified, the file is imported to the `media-root` folder.

    <Note>

    - The `media` property isn't returned in the `files` response object.
    - When you import a file, it's not immediately available, meaning you can't manage or use the file straight away. Learn more about [knowing when a file is ready](https://dev.wix.com/docs/api-reference/assets/media/media-manager/files/importing-files).

    </Note>

    To import files, you need to do one of the following for each file:
    - Specify its [MIME type](https://developer.mozilla.org/en-US/docs/Web/HTTP/Basics_of_HTTP/MIME_types) in the `mimeType` field of the request. For example, `mimeType: 'image/jpeg'`.
    - Include its extension in either the `displayName` or `url` field of the request. For example, `displayName: 'Example Image.jpeg` or `url: https://www.example.com/image.jpeg`.
    - Ensure the server hosting the file supports HEAD requests. In these cases the Wix servers can retrieve the MIME type from the hosting server.
    <Note>

    If you want to validate the media type, specify the file's expected media type in the optional `mediaType` field of the request.  For example, `mediaType: 'IMAGE'`.

    </Note>

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    ## Endpoint

    ```http
    POST https://www.wixapis.com/site-media/v1/bulk/files/import
    ```

    <ParamField body="importFileRequests" type="ImportFileRequest[]" required>

      Constraints: `minItems 1`, `maxItems 100`.

      <Expandable title="importFileRequests properties">

          Constraints: `format WEB_URL`, `minLength 2`.

        </ParamField>

        <ParamField body="mediaType" type="string">

          Media type of the file to import.
          excluding: OTHER media type

          **Supported values:**

          - `IMAGE`
          - `VIDEO`
          - `AUDIO`
          - `DOCUMENT`
          - `VECTOR`
          - `ARCHIVE`
          - `MODEL3D`
          - `OTHER`

        </ParamField>

        <ParamField body="displayName" type="string">

          File name that appears in the Media Manager.
          The file name is truncated to 255 characters if longer.

        </ParamField>

        <ParamField body="parentFolderId" type="string">

          ID of the file's parent folder. &lt;br />
          This folder is the path root for the `filePath`. &lt;br />
          Default: `media-root`.

        </ParamField>

        <ParamField body="private" type="boolean">

          Whether the file will be public or private. Learn more [about private files](https://dev.wix.com/docs/api-reference/assets/media/media-manager/files/private-files).

        </ParamField>

        <ParamField body="labels" type="string[]">

        </ParamField>

        <ParamField body="mimeType" type="string">

          Constraints: `maxLength 100`.

        </ParamField>

              External information to specify in the File Ready or File Failed events.

              Constraints: `maxLength 1000`.

            </ParamField>

            <ParamField body="externalIds" type="string[]">

              External IDs to specify in the File Ready or File Failed events.

              Constraints: `maxLength 4000`, `maxItems 30`.

            </ParamField>

          </Expandable>

        </ParamField>

        <ParamField body="urlParams" type="object">

          Optional parameters that should be sent with the external URL.

        </ParamField>

          Optional headers that should be sent with the external URL.

        </ParamField>

        <ParamField body="filePath" type="string">

          Path to the folder where the file will be stored.
          For example, `/videos/2024/december`. &lt;br/>
          If `parentFolderId` is defined, the parent folder is used as the path root. Otherwise, the root is `media-root`.

    /**********************************************
     * Backend code - bulk-import-files.web.js/ts *
     *********************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { files } from '@wix/media';
    import { auth } from '@wix/essentials';

    export const importFiles = webMethod(Permissions.Anyone, async (importFileRequests) => {
      try {
        const elevatedBulkImportFiles = auth.elevate(files.bulkImportFiles)
        const importedFiles = await elevatedBulkImportFiles(importFileRequests);

        console.log("Imported files:", importedFiles);
        return importedFiles;
      } catch (error) {
        console.error(error);
        // Handle the error
      }
    });

    /*************
     * Page code *
     ************/

    import { importFiles } from 'backend/bulk-import-files.web';

    $w.onReady(() => {
      const newMedia = [];

      $w('#addMedia').onClick( () => {
        const newMediaFile = {
          displayName: $w('#displayName').value,
          mediaType: $w('#mediaType').value,
          parentFolderId: $w('#parentFolderId').value,
          url: $w('#url').value
        }
        newMedia.push(newMediaFile);

        // show message to remind user to submit media
        // after adding all media files for 5 seconds
        $w('#rememberToUploadMessage').show();
        setTimeout(() => {
          $w('#rememberToUploadMessage').hide();
        }, 5000)
      });

      $w('#uploadMedia').onClick(async () => {
        if (newMedia.length === 0) {
          // if no media has been added then
          // show message to add media
          $w('#addMediaMsg').show()
        } else {
          await importFiles(newMedia);

          $w('#successfulUploadMsg').show();
          setTimeout(() => {
            $w('#successfulUploadMsg').hide();
          }, 10000);
        }
      });
    });
