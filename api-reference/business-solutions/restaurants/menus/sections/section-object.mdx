---
title: "Section Object"
description: "The Section object in the Wix Section API."
icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Section ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the section is updated. To prevent conflicting changes, the current revision must be passed when updating the section. Ignored when creating a section.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the section was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the section was updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Section name.

      Constraints: `minLength 1`, `maxLength 500`.

    </ResponseField>

    <ResponseField name="description" type="string">

      Section description.

    </ResponseField>

    <ResponseField name="image" type="Image">

      Main section image.

      <Expandable title="image properties">

        <ResponseField name="id" type="string">

          WixMedia image ID.

        </ResponseField>

        <ResponseField name="url" type="string">

          Image URL.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image height.

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image width.

        </ResponseField>

        <ResponseField name="altText" type="string">

          Image alt text.

        </ResponseField>

        <ResponseField name="filename" type="string">

          Image filename.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="additionalImages" type="Image[]">

      Additional section images.

      Constraints: `maxItems 100`.

      <Expandable title="additionalImages properties">

        <ResponseField name="id" type="string">

          WixMedia image ID.

        </ResponseField>

        <ResponseField name="url" type="string">

          Image URL.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image height.

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image width.

        </ResponseField>

        <ResponseField name="altText" type="string">

          Image alt text.

        </ResponseField>

        <ResponseField name="filename" type="string">

          Image filename.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="itemIds" type="string[]">

      Item IDs.

      Constraints: `format GUID`, `maxItems 300`.

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Extended fields.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="visible" type="boolean">

      Whether the section is visible in the menu for site visitors.

    </ResponseField>

    <ResponseField name="businessLocationId" type="string">

      ID of the business location ([SDK](https://dev.wix.com/docs/sdk/backend-modules/restaurants/wix-restaurants-new/about-business-locations) | [REST](https://dev.wix.com/docs/rest/business-solutions/restaurants/wix-restaurants-new/about-business-locations)) where this section is available.

      **Read-only.**

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "section": {
          "id": "9ea7135d-c232-437f-a9c3-33dd87265542",
          "revision": "2",
          "createdDate": "2024-01-14T10:06:30.782Z",
          "updatedDate": "2024-01-14T10:10:37.372Z",
          "name": "Mains",
          "description": "A diverse range of flavorful dishes which are all sourced daily and locally.",
          "image": null,
          "additionalImages": [],
          "itemIds": [
            "b3dfd67a-7c77-43e8-a707-38eb53850d4a",
            "ec225729-29ae-4a73-bdf6-f873bcabcd53",
            "b8e179cc-f339-4ccb-afa9-d9e04d56c461",
            "0517e5d4-8fd3-4b28-a743-68cf631e534a",
            "cbbf4358-292c-4e9e-a820-7e73b4c4ac00"
          ]
        }
      }
      ```

    </CodeGroup>
