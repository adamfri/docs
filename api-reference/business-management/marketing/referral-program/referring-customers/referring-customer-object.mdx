icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      ID of the referring customer.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="contactId" type="string">

      Contact ID associated with the referring customer.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="referralCode" type="string">

      Unique code for the referral. For example, `GxpxwAoMqxH8`.

      **Read-only.** Constraints: `minLength 12`, `maxLength 12`.

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the referring customer is updated.
      To prevent conflicting changes, the current revision must be passed when updating the referring customer.

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the referring customer was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the referring customer was last updated.

      **Read-only.**

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "referringCustomer": {
          "id": "5cc0b472-e488-4de3-92ba-bd718a44933a",
          "contactId": "5cc0b472-e488-4de3-92ba-bd718a44933a",
          "referralCode": "9zb9JvjwrvQF",
          "createdDate": "2024-09-17T10:58:00.100Z",
          "updatedDate": "2024-09-17T10:58:00.100Z"
        }
      }
      ```

    </CodeGroup>
