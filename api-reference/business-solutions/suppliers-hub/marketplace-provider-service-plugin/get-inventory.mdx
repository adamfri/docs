    Retrieves current inventory levels for a product's variants.

    Wix calls this method when a site owner opens a product details page and wants to see current stock levels.

    Wix uses the response to display available inventory for each product variant.

    ## Endpoint

    ```http
    POST /v2/get-inventory
    ```

    ## Body Params

    <ParamField body="providerProductId" type="string">

      Unique product identifier on the provider's platform.

      Constraints: `minLength 1`, `maxLength 36`.

    </ParamField>

    <ParamField body="choices" type="Choice[]">

      Choices to filter by. For example, only return inventory for variants matching "Size: M" and "Color: Black".

      If empty, inventory for all available variants is returned.

      Constraints: `maxItems 10`.

      <Expandable title="choices properties">

        <ParamField body="name" type="string">

          Choice name. For example, "Size", "Color", or "Material".

          Constraints: `maxLength 50`.

        </ParamField>

        <ParamField body="value" type="ChoiceValue">

          Choice value.

          <Expandable title="One of the following">

            <ParamField body="value" type="value">

              value

              _Nested structure omitted for brevity — see the `value` object._

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    ## Response Object

    <ResponseField name="providerProductId" type="string">

      Unique product identifier on the provider's platform.

      Constraints: `minLength 1`, `maxLength 36`.

    </ResponseField>

    <ResponseField name="inventoryVariants" type="InventoryVariant[]">

      Inventory data for each variant matching the requested choices.

      Constraints: `maxItems 100`.

      <Expandable title="inventoryVariants properties">

        <ResponseField name="choices" type="Choice[]">

          Choices that define this variant. For example, "Size: M" and "Color: Black".

          Constraints: `maxItems 10`.

          <Expandable title="choices properties">

            <ResponseField name="name" type="string">

              Choice name. For example, "Size", "Color", or "Material".

              Constraints: `maxLength 50`.

            </ResponseField>

            <ResponseField name="value" type="ChoiceValue">

              Choice value.

              _Nested structure omitted for brevity — see the `ChoiceValue` object._

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="quantity" type="integer">

          Available inventory quantity for this variant. `null` indicates the quantity is unknown or not tracked by the provider.

        </ResponseField>

      </Expandable>

    </ResponseField>
