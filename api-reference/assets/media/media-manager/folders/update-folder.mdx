    Updates a folder. <br />

    You can use the `parentFolderId` parameter to move a folder from its current parent folder to a different parent folder.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-MEDIA.MANAGE-MEDIAMANAGER">Manage Media Manager</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    PATCH https://www.wixapis.com/site-media/v1/folders/{folder.id}
    ```

    ## Path Params

    <ParamField path="folder.id" type="string" required>

      Folder ID. Generated when a folder is created in the Media Manager.

    </ParamField>

    ## Body Params

    <ParamField body="folder" type="Folder" required>

      Folder to update.

      See the [Folder Object](/api-reference/assets/media/media-manager/folders/folder-object) for this object's fields.

    </ParamField>

    ## Response Object

    <ResponseField name="folder" type="Folder">

      Information about the updated folder.

      See the [Folder Object](/api-reference/assets/media/media-manager/folders/folder-object) for this object's fields.

    </ResponseField>

    ## Errors

    | --- | --- | --- | --- |
    | `400` | `INVALID_ARGUMENT` | `FIELD_MASK_IS_EMPTY_OR_MISSING` | `fieldMask` is empty or missing from the request. |

    <Note>

      This method may also return standard errors. [Learn more about standard Wix errors](https://dev.wix.com/docs/rest/articles/getting-started/errors).

    </Note>

    ```javascript Update a folder's location (with $w)
    /******************************************
     * Backend code - create-folder.web.js/ts *
     *****************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { folders } from '@wix/media';
    import { auth } from '@wix/essentials';

    export const listFolders = webMethod(Permissions.Anyone, async () => {
      try {
        const elevatedListFolders = auth.elevate(folders.listFolders)
        const foldersList = await elevatedListFolders();

        return foldersList;
      } catch (error) {
        console.error(error);
      }
    });

    export const updateFolder = webMethod(Permissions.Anyone, async (id, options) => {
      try {
        const elevatedUpdateFolder = auth.elevate(folders.updateFolder);
        const updatedFolder = await elevatedUpdateFolder(id, options);

        return updatedFolder;
      } catch (error) {
        console.error(error);
      }
    });

    /*************
     * Page code *
     ************/

    import { updateFolder, listFolders } from 'backend/create-folder.web';

    $w.onReady(async () => {
      await populateFoldersDropdowns();

      $w('#updateFolderLocation').onClick(async () => {
        const folderId = $w('#updateFolderDropdown').value;
        const parentFolderId = $w('#parentFolderDropdown').value;
        if (folderId === parentFolderId) {
            $w('#folderIsParentFolderMsg').show();
            return;
        };
        const options = {parentFolderId};
        const updatedFolder = await updateFolder(folderId, options);

        console.log(`Successfully updated folder ${updatedFolder.displayName}:`, updatedFolder);
        $w('#updateFolderSuccessMsg').show();
      });
    });

    async function populateFoldersDropdowns() {
      const folderList = await listFolders();
      const dropdownOptions = folderList.folders.map((folder) => {
        return {
          label: folder.displayName,
          value: folder._id
        };
      });

      $w('#updateFolderDropdown').options = dropdownOptions;
      $w('#parentFolderDropdown').options = dropdownOptions;
    };
    ```
