---
title: "Gallery Object"
description: "The Gallery object in the Wix Gallery API."
icon: "cube"
---

<ResponseField name="id" type="string">

  Gallery ID.

  **Read-only.**

</ResponseField>

<ResponseField name="name" type="string">

  Gallery name.

</ResponseField>

<ResponseField name="totalItems" type="integer">

  Total number of items in the gallery.

  **Read-only.**

</ResponseField>

<ResponseField name="items" type="Item[]">

  Media items in the gallery.

  <Expandable title="items properties">

    <ResponseField name="id" type="string">

      Item ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="sortOrder" type="number">

      Index that determines which position a media item is displayed in the gallery. &lt;br />

      Default: [Epoch](https://www.epoch101.com/) timestamp. &lt;br />

      >**Note:** If you assign the same sort order index to more than one media item in a gallery, the function fails.

    </ResponseField>

    <ResponseField name="title" type="string">

      Item title.

    </ResponseField>

    <ResponseField name="description" type="string">

      Item description.

    </ResponseField>

    <ResponseField name="link" type="Link">

      Link from the item. You can link to Wix sites or external URLs.

      <Expandable title="link properties">

        <ResponseField name="text" type="string">

          Display text of the link.

        </ResponseField>

        <ResponseField name="url" type="string">

          Target URL of the link.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="type" type="string">

      Type of item.

      **Read-only.**

      **Supported values:**

      - `UNDEFINED`
      - `IMAGE`
      - `VIDEO`
      - `TEXT`

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the item was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the item was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="tags" type="Tags">

      Item tags.

      <Expandable title="tags properties">

        <ResponseField name="values" type="string[]">

          List of tags assigned to the media item.

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Expandable>

</ResponseField>

<ResponseField name="sortOrder" type="number">

  Index that determines which position a gallery is displayed on the site. &lt;br />

  >**Note:** If you assign the same sort order index to more than one gallery, the function fails.

</ResponseField>

<ResponseField name="createdDate" type="datetime">

  Date and time the gallery was created.

  **Read-only.**

</ResponseField>
