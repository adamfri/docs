icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Referral event ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the referral event is updated.
      To prevent conflicting changes, the current revision must be passed when updating the referral event.

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the referral event was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the referral event was last updated.

      **Read-only.**

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "referralEvent": {
          "id": "017fad99-aa0f-47a9-8d4f-177fc6a07da3",
          "revision": "1",
          "createdDate": "2024-07-08T11:22:00.826Z",
          "updatedDate": "2024-07-08T11:22:00.826Z",
          "rewardEvent": {
            "referralRewardId": "c8c732cf-2ca2-4025-9b1c-3be02f099cdb",
            "rewardType": "COUPON",
            "rewardedReferringCustomerId": "4fb41de4-da9c-4818-b54f-82c955ecc866"
          }
        }
      }
      ```

    </CodeGroup>
