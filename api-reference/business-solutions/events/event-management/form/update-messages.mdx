    Updates a set of defined form messages that are displayed to a site visitor before, during, and after the registration flow.
    Includes the configuration of form titles, response labels, "thank you" messages, and call-to-action texts.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-EVENTS.MANAGE-EVENTS">Manage Events</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    PATCH https://www.wixapis.com/events/v1/events/{eventId}/form/messages
    ```

    ## Path Params

    <ParamField path="eventId" type="string" required>

      Event ID to which the form belongs.

    </ParamField>

    ## Body Params

    <ParamField body="fields" type="FieldMask">

      Set of field paths, specifying which parts of this resource to update.
      When fields are empty, request is interpreted as full update.
      Behavior follows [google.protobuf.FieldMask](https://developers.google.com/protocol-buffers/docs/reference/google.protobuf#google.protobuf.FieldMask) semantics.

      <Expandable title="fields properties">

        <ParamField body="paths" type="string[]" />

      </Expandable>

    </ParamField>

    <ParamField body="messages" type="FormMessages">

      Set of configured form messages to update.

      <Expandable title="messages properties">

        <ParamField body="rsvp" type="RsvpFormMessages">

          [RSVP form](https://dev.wix.com/docs/rest/business-solutions/events/rsvp-v2/introduction) messages.

          <Expandable title="rsvp properties">

            <ParamField body="rsvpYesOption" type="string">

              Label text indicating RSVP's `status` is `"YES"`.

            </ParamField>

            <ParamField body="rsvpNoOption" type="string">

              Label text indicating RSVP's `status` is `"NO"`.

            </ParamField>

            <ParamField body="positiveMessages" type="Positive">

              Messages displayed when an RSVP's `status` is set to `"YES"`.

              _Nested structure omitted for brevity — see the `Positive` object._

            </ParamField>

            <ParamField body="waitlistMessages" type="Positive">

              Messages displayed when an RSVP's `status` is set to `"WAITLIST"`, for when the event is full and a waitlist is available).

              _Nested structure omitted for brevity — see the `Positive` object._

            </ParamField>

            <ParamField body="negativeMessages" type="Negative">

              Messages displayed when an RSVP's `status` is set to `"NO"`.

              _Nested structure omitted for brevity — see the `Negative` object._

            </ParamField>

            <ParamField body="submitActionLabel" type="string">

              "Submit form" call-to-action label text.

            </ParamField>

          </Expandable>

        </ParamField>

        <ParamField body="checkout" type="CheckoutFormMessages">

          Checkout form messages.

          <Expandable title="checkout properties">

            <ParamField body="title" type="string">

              Main form title for response.

            </ParamField>

            <ParamField body="submitActionLabel" type="string">

              Submit form call-to-action label text.

            </ParamField>

            <ParamField body="confirmation" type="ResponseConfirmation">

              Confirmation messages shown after checkout.

              _Nested structure omitted for brevity — see the `ResponseConfirmation` object._

            </ParamField>

          </Expandable>

        </ParamField>

        <ParamField body="registrationClosed" type="RegistrationClosedMessages">

          Messages shown when event registration is closed.

          <Expandable title="registrationClosed properties">

            <ParamField body="message" type="string">

              Message shown when event registration is closed.

            </ParamField>

            <ParamField body="exploreEventsActionLabel" type="string">

              "Explore other events" call-to-action label text.

            </ParamField>

          </Expandable>

        </ParamField>

        <ParamField body="ticketsUnavailable" type="TicketsUnavailableMessages">

          Messages shown when event tickets are unavailable.

          <Expandable title="ticketsUnavailable properties">

            <ParamField body="message" type="string">

              Message shown when event tickets are unavailable.

            </ParamField>

            <ParamField body="exploreEventsActionLabel" type="string">

              "Explore other events" call-to-action label text.

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    ## Response Object

    <ResponseField name="form" type="Form">

      Modified event form.

      See the [Form Object](/api-reference/business-solutions/events/event-management/form/form-object) for this object's fields.

    </ResponseField>
