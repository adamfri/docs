    <Warning>

      **Deprecated.** This method is deprecated.

    </Warning>

    Marks a pricing plan as the primary pricing plan. When viewing pricing plans on a site, the primary plan is highlighted with a customizable ribbon. Only a single plan can be marked as a primary plan at any given time. If there is an existing plan marked as primary, calling Make Plan Primary causes the existing primary plan to lose its primary status.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-PAIDPLANS.MANAGE-PLANS">Manage Pricing Plans</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/pricing-plans/v2/plans/{id}/make-primary
    ```

    ## Path Params

    <ParamField path="id" type="string" required>

      Pricing plan ID.

    </ParamField>

    ## Response Object

    <ResponseField name="plan" type="Plan">

      Pricing plan.

      See the [Plan Object](/api-reference/business-solutions/pricing-plans/plans-deprecated/plan-object) for this object's fields.

    </ResponseField>

    ## Errors

    | Status | Code | Application code | Description |
    | --- | --- | --- | --- |
    | `428` | `FAILED_PRECONDITION` | `PRICING_PLANS_NOT_INSTALLED` | The user does not have the Wix Pricing Plans app installed on the site. |
    | `404` | `NOT_FOUND` | `plan_not_found` | Pricing plan was not found. |
    | `400` | `INVALID_ARGUMENT` | `set_primary_not_supported` | Cannot make plan primary because it is archived. |

    <Note>

      This method may also return standard errors. [Learn more about standard Wix errors](https://dev.wix.com/docs/rest/articles/getting-started/errors).

    </Note>

    ```javascript Make a plan primary (with $w)
    /*******************************************
     * Backend code - plan-functions.web.js/ts *
     ******************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { plans } from '@wix/pricing-plans';
    import { auth } from '@wix/essentials';

    const elevatedListPublicPlans = auth.elevate(plans.listPublicPlans);
    const elevatedMakePlanPrimary = auth.elevate(plans.makePlanPrimary);

    export const listPublicPlans = webMethod(Permissions.Anyone, async () => {
      try {
        const response = await elevatedListPublicPlans();
        const plansList = response.plans;

        return plansList;
      } catch (error) {
        console.error(error);
        // Handle the error
      }
    });

    export const makePlanPrimaryFunction = webMethod(Permissions.Anyone, async (_id) => {
      try {
        const primaryPlan = await elevatedMakePlanPrimary(_id);

        return primaryPlan;
      } catch (error) {
        console.error(error);
        // Handle the error
      }
    });

    /*************
     * Page code *
     *************/

    import { listPublicPlans, makePlanPrimaryFunction } from 'backend/plan-functions.web';

    $w.onReady(function () {
      populateDropdown();
      $w('#plansList').onChange(async () => {
        await makePlanPrimaryFunction(
          $w('#plansList').value
        )
      });
    });

    async function populateDropdown() {
      const plans = await listPublicPlans();
      $w('#plansList').options = plans.map(item => ({
        label: item.name,
        value: item._id
      })
      )
    }
    ```
