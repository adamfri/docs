    <Warning>

      **Deprecated.** This method is deprecated.

    </Warning>

    Retrieves a list of up to 100 public pricing plans. Public plans are visible plans that are available to site visitors and can be purchased.

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-PAIDPLANS.READ-ORDERS">Read Orders</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`, `MEMBER`, `VISITOR`.

    ## Endpoint

    ```http
    GET https://www.wixapis.com/pricing-plans/v2/plans/public
    ```

    ## Query Params

    <ParamField query="limit" type="integer">

      Number of items to list. Defaults to `75`. See [limits](https://dev.wix.com/docs/api-reference/articles/work-with-wix-apis/data-retrieval/about-the-wix-api-query-language#paging).

    </ParamField>

    <ParamField query="offset" type="integer">

      Number of items to skip. Defaults to `0`. See [paging](https://dev.wix.com/docs/api-reference/articles/work-with-wix-apis/data-retrieval/about-the-wix-api-query-language#paging).

    </ParamField>

    <ParamField query="planIds" type="string[]">

      IDs of plans to list. Non-existent IDs are ignored and don't cause errors. You can specify a maximum of 100 IDs.

      Constraints: `format GUID`, `maxItems 100`.

    </ParamField>

    ## Response Object

    <ResponseField name="plans" type="PublicPlan[]">

      List of public pricing plans.

      <Expandable title="plans properties">

        <ResponseField name="id" type="string">

          Plan ID.

          Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="name" type="string">

          Plan name.

        </ResponseField>

        <ResponseField name="description" type="string">

          Plan description.

        </ResponseField>

        <ResponseField name="perks" type="StringList">

          What is included with this plan (for example, 1 weekly entrance to a specific class).

          <Expandable title="perks properties">

            <ResponseField name="values" type="string[]">

              Constraints: `minLength 1`, `maxLength 1400`, `maxItems 500`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="pricing" type="Pricing">

          Plan price, payment schedule, and expiration.

          <Expandable title="pricing properties">

            <ResponseField name="price" type="Money">

              Amount for a single payment. For subscriptions, this is the amount to pay each payment cycle and it is required. For plans that are not recurring plans, it is the single payment amount for the whole subscription.

              _Nested structure omitted for brevity — see the `Money` object._

            </ResponseField>

            <ResponseField name="freeTrialDays" type="integer">

              Free trial period for the plan in days. Available only for recurring plans, meaning plans whose pricing model is `subscription`. Set to `0` to remove the free trial.

            </ResponseField>

          </Expandable>

          <Expandable title="One of the following">

            <ResponseField name="pricingModel" type="pricingModel">

              pricingModel

              _Nested structure omitted for brevity — see the `pricingModel` object._

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="primary" type="boolean">

          Whether the plan is marked as primary.

        </ResponseField>

        <ResponseField name="createdDate" type="datetime">

          Date plan was created.

        </ResponseField>

        <ResponseField name="updatedDate" type="datetime">

          Date plan was last updated.

        </ResponseField>

        <ResponseField name="slug" type="string">

          URL-friendly version of plan name. Unique across all plans in the same site.

        </ResponseField>

        <ResponseField name="maxPurchasesPerBuyer" type="integer">

          Number of times the same buyer can purchase the plan. An empty value or a value of 0 means no limitation.

        </ResponseField>

        <ResponseField name="allowFutureStartDate" type="boolean">

          Whether the buyer can start the plan at a later date. Defaults to false.

        </ResponseField>

        <ResponseField name="buyerCanCancel" type="boolean">

          Whether the buyer is allowed to cancel their plan. Defaults to false.

        </ResponseField>

        <ResponseField name="termsAndConditions" type="string">

          Any terms and conditions that apply to the plan. This information is displayed during checkout.

        </ResponseField>

        <ResponseField name="formId" type="string">

          ID of the form associated with the plan at checkout.

          Learn more about [forms](https://support.wix.com/en/article/wix-forms-an-overview).

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="pagingMetadata" type="PagingMetadataV2">

      Object containing paging-related data (such as the number of plans returned, the offset, and so on).

      <Expandable title="pagingMetadata properties">

        <ResponseField name="count" type="integer">

          Number of items returned in the response.

        </ResponseField>

        <ResponseField name="offset" type="integer">

          Offset that was requested.

        </ResponseField>

        <ResponseField name="total" type="integer">

          Total number of items that match the query. Returned if offset paging is used and the `tooManyToCount` flag is not set.

        </ResponseField>

        <ResponseField name="tooManyToCount" type="boolean">

          Flag that indicates the server failed to calculate the `total` field.

        </ResponseField>

        <ResponseField name="cursors" type="Cursors">

          Cursors to navigate through the result pages using `next` and `prev`. Returned if cursor paging is used.

          <Expandable title="cursors properties">

            <ResponseField name="next" type="string">

              Cursor string pointing to the next page in the list of results.

            </ResponseField>

            <ResponseField name="prev" type="string">

              Cursor pointing to the previous page in the list of results.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>
