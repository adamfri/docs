---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Item ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the item is updated. To prevent conflicting changes, the current revision must be passed when updating the item. Ignored when creating a item.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the item was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the item was updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Item name.

      Constraints: `minLength 1`, `maxLength 200`.

    </ResponseField>

    <ResponseField name="description" type="string">

      Item description.

    </ResponseField>

    <ResponseField name="image" type="Image">

      Main item image.

      <Expandable title="image properties">

        <ResponseField name="id" type="string">

          WixMedia image ID.

        </ResponseField>

        <ResponseField name="url" type="string">

          Image URL.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image height.

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image width.

        </ResponseField>

        <ResponseField name="altText" type="string">

          Image alt text.

        </ResponseField>

        <ResponseField name="filename" type="string">

          Image filename.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="additionalImages" type="Image[]">

      Additional item images.

      Constraints: `maxItems 15`.

      <Expandable title="additionalImages properties">

        <ResponseField name="id" type="string">

          WixMedia image ID.

        </ResponseField>

        <ResponseField name="url" type="string">

          Image URL.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image height.

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image width.

        </ResponseField>

        <ResponseField name="altText" type="string">

          Image alt text.

        </ResponseField>

        <ResponseField name="filename" type="string">

          Image filename.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="labels" type="Label[]">

      Item labels.

      Constraints: `maxItems 10`.

      <Expandable title="labels properties">

        <ResponseField name="id" type="string">

          Item label ID.

          Constraints: `format GUID`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="visible" type="boolean">

      Whether the item is visible in the menu for site visitors.

    </ResponseField>

    <ResponseField name="featured" type="boolean">

      Whether the item is marked as featured/best seller.
      Default: `false`.

    </ResponseField>

    <ResponseField name="orderSettings" type="OrderSettings">

      Online order settings.

      <Expandable title="orderSettings properties">

        <ResponseField name="inStock" type="boolean">

          Whether the item is in stock.
          Default: `true`.

        </ResponseField>

        <ResponseField name="acceptSpecialRequests" type="boolean">

          Whether a customer can add a special request when ordering this item.
          Default: `true`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="modifierGroups" type="ModifierGroup[]">

      Item modifier groups.

      Constraints: `maxItems 25`.

      <Expandable title="modifierGroups properties">

        <ResponseField name="id" type="string">

          Modifier group ID.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Extended fields.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="businessLocationIds" type="string[]">

      IDs of the business locations ([SDK](https://dev.wix.com/docs/sdk/backend-modules/restaurants/wix-restaurants-new/about-business-locations) | [REST](https://dev.wix.com/docs/rest/business-solutions/restaurants/wix-restaurants-new/about-business-locations)) where these items are available.

      **Read-only.** Constraints: `format GUID`, `maxItems 100`.

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "item": {
          "id": "0517e5d4-8fd3-4b28-a743-68cf631e534a",
            "revision": "1",
            "createdDate": "2024-01-14T12:42:50.267Z",
            "updatedDate": "2024-01-14T12:42:50.267Z",
            "name": "Classic burger",
            "description": "Our classic burger with lettuce, pickles, heirloom tomatoes, served with a side of fries",
            "priceVariants": {
              "variants":[
                {
                  "variantId": "ef7dc1ed-06df-481a-8ecc-ad87ef9b6b0c",
                  "priceInfo": {"price": "7.00"}
                }
              ]
              },
              "image": {
                "id": "ec845c_e3be4aec9afe4edaa505e903d8e13344~mv2.jpg",
                "url": "https://static.wixstatic.com/media/ec845c_e3be4aec9afe4edaa505e903d8e13344~mv2.jpg",
                "height": 3335,
                "width":5000
              },
              "additionalImages":[],
              "labels":[],
              "orderSettings": {
                "inStock":false,
                "acceptSpecialRequests":true,
                "specialRequestInstructions": "Add allergies or other notes..."
              },
              "modifierGroups":[]
        }
      }
      ```

    </CodeGroup>
