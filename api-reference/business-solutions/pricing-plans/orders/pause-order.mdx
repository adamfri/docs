    Pauses an order. Calling this method changes the order status to `"PAUSED"` and updates the `pausePeriods` array.

    Only orders with `status`: `ACTIVE` can be paused.
    For orders with recurring payments, it also pauses the payment schedule. Buyers are not charged when an order is paused.
    Pausing an order affects the end date of the order by adding the time the order is paused to the `endDate`.
    The `endDate` and the `earliestEndDate` for the order are adjusted to include the pause period when the order is resumed.

    To resume a paused order, call Resume Order.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-PAIDPLANS.MANAGE-ORDERS">Manage Orders</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/pricing-plans/v2/orders/{id}/pause
    ```

    ## Path Params

    <ParamField path="id" type="string" required>

      Order ID.

    </ParamField>

    ## Response Object

    Returns `PauseOrderResponse`.

    ## Errors

      This method doesn't return any custom errors, but may return standard errors. [Learn more about standard Wix errors](https://dev.wix.com/docs/rest/articles/getting-started/errors).

          return;
        } catch (error) {
          console.error(error);
          // Handle the error
        }
      });

    export const pauseOrder = webMethod(
      Permissions.Anyone, 
      async (orderId) => {

    import { listCurrentMemberOrders, requestCancellation, pauseOrder } from 'backend/order.web';

    $w.onReady(function () {

      $w('#pauseOrderBtn').disable();

      let selectedOrderId;
      populateOrdersDropdown();

        $w('#pauseOrderBtn').enable();

        await pauseOrder(selectedOrderId);
      });
    });

    async function populateOrdersDropdown() {
      const activeOrdersList = await listCurrentMemberOrders({orderStatuses: 'ACTIVE'});
      $w('#ordersDropdown').options = activeOrdersList.orders.map((order) => {
        return {
          label: order.planName,
          value: order._id

      });
      $w('#ordersDropdown').enable();
