  <Tab title="REST">

    <ResponseField name="id" type="string">

      RSVP ID.

      Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="eventId" type="string">

      Event ID.

      Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="contactId" type="string">

      Contact ID associated with this RSVP.

      Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="memberId" type="string">

      Member ID associated with this RSVP.

      Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="created" type="datetime">

      RSVP created timestamp.

    </ResponseField>

    <ResponseField name="modified" type="datetime">

      RSVP modified timestamp.

    </ResponseField>

    <ResponseField name="firstName" type="string">

      First name.

    </ResponseField>

    <ResponseField name="lastName" type="string">

      Last name.

    </ResponseField>

    <ResponseField name="email" type="string">

      Guest email.

      Constraints: `format EMAIL`.

    </ResponseField>

    <ResponseField name="rsvpForm" type="FormResponse">

      RSVP form response.

      <Expandable title="rsvpForm properties">

        <ResponseField name="inputValues" type="InputValue[]">

          Form field inputs.

          Constraints: `maxItems 200`.

          <Expandable title="inputValues properties">

            <ResponseField name="inputName" type="string">

              Form field input name.

              Constraints: `maxLength 100`.

            </ResponseField>

            <ResponseField name="value" type="string">

              Form field value.

              Constraints: `maxLength 5000`.

            </ResponseField>

            <ResponseField name="values" type="string[]">

              Form field values.

              Constraints: `maxLength 5000`, `maxItems 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="status" type="string">

      RSVP response status.

      **Supported values:**

      - `YES`
      - `NO`
      - `WAITING`

    </ResponseField>

    <ResponseField name="totalGuests" type="integer">

      Total number of attendees.

    </ResponseField>

    <ResponseField name="guests" type="Guest[]">

      List of guests.

      <Expandable title="guests properties">

        <ResponseField name="index" type="integer">

          Index in the RSVP guest list.

        </ResponseField>

        <ResponseField name="fullName" type="string">

          Guest full name.

        </ResponseField>

        <ResponseField name="checkIn" type="CheckIn">

          Guest check-in.

          <Expandable title="checkIn properties">

            <ResponseField name="created" type="datetime">

              Time of a ticket's check-in.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="id" type="integer">

          Unique guest ID per RSVP.

          Constraints: `min 1`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="anonymized" type="boolean">

      Whether RSVP is anonymized by GDPR delete.

    </ResponseField>

    <ResponseField name="marketingConsent" type="boolean">

      Whether marketing consent was given

    </ResponseField>
