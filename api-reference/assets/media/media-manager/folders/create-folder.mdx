    <div className="wix-ref">

    Creates a new folder in the Media Manager. <br />

    Use the `parentFolderId` parameter to specify in which existing folder you want the new folder to be created.
    If no folder is specified, the new folder is created in the `media-root` folder.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-MEDIA.MANAGE-MEDIAMANAGER">Manage Media Manager</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/site-media/v1/folders
    ```

    ## Body Params

    <ParamField body="displayName" type="string" required>

      Folder name that appears in the Media Manager.

      Constraints: `maxLength 1000`.

    </ParamField>

    <ParamField body="parentFolderId" type="string">

      ID of the folder's parent folder.

    </ParamField>

    ## Response Object

    <ResponseField name="folder" type="Folder">

      Information about the newly created folder.

      See the [Folder Object](/api-reference/assets/media/media-manager/folders/folder-object) for this object's fields.

    </ResponseField>

    ## Errors

    <Note>

      This method doesn't return any custom errors, but may return standard errors. [Learn more about standard Wix errors](https://dev.wix.com/docs/rest/articles/getting-started/errors).

    </Note>

    </div>

    ```javascript Create a folder from visitor input (with $w)
    /******************************************
     * Backend code - create-folder.web.js/ts *
     *****************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { folders } from '@wix/media';
    import { auth } from '@wix/essentials';

    export const listFolders = webMethod(Permissions.Anyone, async () => {
      try {
        const elevatedListFolders = auth.elevate(folders.listFolders)
        const foldersList = await elevatedListFolders();

        return foldersList;
      } catch (error) {
        console.error(error);
      }
    });

    export const createFolder = webMethod(Permissions.Anyone, async (displayName, options) => {
      try {
        const elevateCreateFolder = auth.elevate(folders.createFolder);
        const newFolder = await elevateCreateFolder(displayName, options);

        return newFolder;
      } catch (err){
        console.error(err);
      }
    });

    /*************
     * Page code *
     ************/

    import { createFolder, listFolders } from 'backend/create-folder.web';

    $w.onReady(async () => {
      await populateFoldersDropdown();

      $w('#createFolder').onClick(async () => {
        const folderName = $w('folderName').value;
        const createOptions = {parentFolderId: $w('foldersDropdown').value};
        const newFolder = await createFolder(folderName, createOptions);

        console.log(`Successfully created new folder "${folderName}" in ${$w('#foldersDropdown').label}.`)
        $w('#createdFolderSuccessMsg').show();
      });
    });

    async function populateFoldersDropdown() {
      const folderList = await listFolders();
      const dropdownOptions = folderList.folders.map((folder) => {
        return {
          label: folder.displayName,
          value: folder._id
        };
      });

      $w('#foldersDropdown').options = dropdownOptions;
    };
    ```
