---
title: "Availability Exception Object"
description: "The Availability Exception object in the Wix Availability Exception API."
icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Availability exception ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the availability exception is updated.
      To prevent conflicting changes, the current revision must be specified when updating the availability exception.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the availability exception was created in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#combined_date_and_time_representations) format.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the availability exception was last updated in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#combined_date_and_time_representations) format.

      **Read-only.**

    </ResponseField>

    <ResponseField name="startTime" type="datetime">

      The start time of the availability exception in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#combined_date_and_time_representations) format.

    </ResponseField>

    <ResponseField name="endTime" type="datetime">

      The end time of the availability exception in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#combined_date_and_time_representations) format.

    </ResponseField>

    <ResponseField name="available" type="boolean">

      Whether the exception makes the [`start_time`, `end_time`] range available.
      If `true`, the exception makes the restaurant available for online ordering during the time range. If `false`, the exception makes the restaurant unavailable for ordering during the time range.
      Currently, only `false` is supported.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Exception name.

    </ResponseField>

    <ResponseField name="affectedFulfillmentMethods" type="AffectedFulfillmentMethods">

      Fulfillment methods for which this exception applies.

      <Expandable title="affectedFulfillmentMethods properties">

        <ResponseField name="affectedMethods" type="string">

          The type of the fulfillment methods this exception applies to.

          **Supported values:**

          - `ALL_FULFILLMENT_METHODS` — All fulfillment methods of this operation applied for this exception.
          - `SPECIFIC_FULFILLMENT_METHODS` — Specific fulfillment methods applied for this exception.
          - `ALL_PICKUP_FULFILLMENT_METHODS` — All pickup fulfillment methods applied for this exception.
          - `ALL_DELIVERY_FULFILLMENT_METHODS` — All delivery fulfillment methods applied for this exception.

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="affectedMethodsOptions" type="affectedMethodsOptions">

          affectedMethodsOptions

          <Expandable title="affectedMethodsOptions properties">

            <ResponseField name="specificFulfillmentMethodsOptions" type="SpecificFulfillmentMethodsImpactScope">

              Settings for specific fulfillment methods.

              <Expandable title="specificFulfillmentMethodsOptions properties">

                <ResponseField name="fulfillmentMethodIds" type="string[]">

                  IDs of the fulfillment methods this exception applies to.

                  Constraints: `format GUID`, `maxItems 500`.

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="operationId" type="string">

      ID of the restaurant operation this exception is associated with. (See the Restaurants Operations API for more information.)

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Extended fields.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="businessLocationId" type="string">

      Business location ID ([SDK](https://dev.wix.com/docs/sdk/backend-modules/restaurants/wix-restaurants-new/about-business-locations) | [REST](https://dev.wix.com/docs/rest/business-solutions/restaurants/wix-restaurants-new/about-business-locations)) of the operation this availability exception applies to.

      **Read-only.**

    </ResponseField>

    <ResponseField name="tags" type="Tags">

      Tags ([SDK](https://dev.wix.com/docs/sdk/backend-modules/tags/tags/introduction) | [REST](https://dev.wix.com/docs/rest/business-management/tags/introduction)) used to classify and sort different types of availability exceptions.

      <Expandable title="tags properties">

        <ResponseField name="privateTags" type="TagList">

          Tags that require an additional permission in order to access them, normally not given to site members or visitors.

          <Expandable title="privateTags properties">

            <ResponseField name="tagIds" type="string[]">

              List of tag IDs

              Constraints: `maxLength 5`, `maxItems 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="tags" type="TagList">

          Tags that are exposed to anyone who has access to the labeled entity itself, including site members and visitors.

          <Expandable title="tags properties">

            <ResponseField name="tagIds" type="string[]">

              List of tag IDs

              Constraints: `maxLength 5`, `maxItems 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "availabilityException": {
          "id": "63eb0bf7-96ea-4034-822b-7dd9a24197e9",
          "revision": "1",
          "createdDate": "2024-07-09T10:17:08.865Z",
          "updatedDate": "2024-07-09T10:17:08.865Z",
          "startTime": "2024-07-08T21:00:00.587Z",
          "endTime": "2024-07-09T20:59:59.587Z",
          "available": false,
          "name": "Passover",
          "affectedFulfillmentMethods": {
            "affectedMethods": "ALL_FULFILLMENT_METHODS"
          },
          "operationId": "48605ac5-53fc-49b4-96b2-87f7636f5ce0"
        }
      }
      ```

    </CodeGroup>
