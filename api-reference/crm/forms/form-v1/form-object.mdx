---
title: "Form Object"
description: "The Form object in the Wix Form API."
icon: "cube"
---

<ResponseField name="formId" type="string">

  Form unique ID

</ResponseField>

<ResponseField name="formName" type="string">

  Form name

</ResponseField>

<ResponseField name="collectionId" type="string">

  ID of form submissions collection in WixData

</ResponseField>

<ResponseField name="fields" type="FormFieldDefinition[]">

  List of form field definitions

  <Expandable title="fields properties">

    <ResponseField name="fieldId" type="string">

      Field unique ID

      Constraints: `maxLength 100`.

    </ResponseField>

    <ResponseField name="fieldName" type="string">

      Field name

      Constraints: `maxLength 500`.

    </ResponseField>

    <ResponseField name="label" type="string">

      Field label

    </ResponseField>

    <ResponseField name="placeholder" type="string">

      Placeholder to show to UoU

    </ResponseField>

    <ResponseField name="fieldCollectionType" type="string">

      Column type in WixData collection

    </ResponseField>

    <ResponseField name="fieldCollectionKey" type="string">

      Column name in WixData collection

    </ResponseField>

    <ResponseField name="type" type="string">

      **Supported values:**

      - `UNDEFINED`
      - `TEXT`
      - `PARAGRAPH`
      - `CHECKBOX`
      - `DATE_PICKER`
      - `DROP_DOWN`
      - `RADIO_GROUP`
      - `CHECKBOX_GROUP`
      - `NUMBER`
      - `EMAIL`
      - `PHONE`
      - `URL`
      - `SIGNATURE`
      - `RATING`
      - `MULTI_LINE_ADDRESS`
      - `PHONE_COUNTRY_CODE`
      - `AUTO_COMPLETE_ADDRESS`
      - `ATTACHMENTS`
      - `TIME_PICKER`

    </ResponseField>

    <ResponseField name="options" type="FieldOption[]">

      Constraints: `maxItems 100`.

      <Expandable title="options properties">

        <ResponseField name="value" type="string" />

        <ResponseField name="label" type="string" />

      </Expandable>

    </ResponseField>

  </Expandable>

</ResponseField>

<ResponseField name="submissions" type="number">

  Number of times the form was submitted on the website

</ResponseField>

<ResponseField name="lastSubmission" type="datetime">

  Timestamp of the last submission

</ResponseField>

<ResponseField name="status" type="string">

  Form status

  **Supported values:**

  - `UNDEFINED`
  - `PUBLISHED` — Form is published and available on the site
  - `DELETED` — Form was published but removed from the site
  - `DRAFT` — Form was created in the editor, but was not yet published
  - `DISCARDED` — Form was created in the editor, but deleted before the site was published
  - `ARCHIVED` — Form marked as archived

</ResponseField>

<ResponseField name="createdAt" type="datetime">

  Form creation time

</ResponseField>

<ResponseField name="plugins" type="string[]">

  List of plugins associated with this form

  **Supported values:**

  - `FORM_BUILDER` — Regular form
  - `GET_SUBSCRIBERS` — Get subscribers form plugin
  - `REGISTRATION_FORM` — Registration form plugin
  - `PAYMENT_FORM` — Payment form plugin
  - `CAPTCHA` — Form secured with CAPTCHA
  - `MULTI_STEP` — Multi step form
  - `INBOX_OPT_OUT` — See PluginV2
  - `LIMIT_RULES` — Submissions limiter plugin
  - `APP` — App plugin

</ResponseField>

<ResponseField name="useAdvancedSpamFilter" type="boolean">

  Whether to use new spam filter

</ResponseField>
