Instructor entity

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      ID of the instructor that equals to memberId

      **Read-only.**

    </ResponseField>

    <ResponseField name="userId" type="string">

      User ID for real instructors (will be filled only for real instructors)

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      The name of the instructor

    </ResponseField>

    <ResponseField name="description" type="string">

      A description of the instructor

    </ResponseField>

    <ResponseField name="photo" type="Image">

      URL of profile photo

      <Expandable title="photo properties">

        <ResponseField name="id" type="string">

          Wix Media image ID,
          set when the member selects an image from Wix Media.

        </ResponseField>

        <ResponseField name="url" type="string">

          Image URL.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image width.

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image height.

        </ResponseField>

        <ResponseField name="offsetX" type="integer">

          X-axis offset.

          Default: `0`.

        </ResponseField>

        <ResponseField name="offsetY" type="integer">

          Y-axis offset.

          Default: `0`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="photoAltText" type="string">

      Alt text for profile photo (needed for a11y)

    </ResponseField>

    <ResponseField name="slug" type="string">

      Slug that determines the instructor's profile page URL.

      **Read-only.**

    </ResponseField>

    <ResponseField name="role" type="string">

      Role of the instructor

      **Read-only.**

      **Supported values:**

      - `DEFAULT` — VIRTUAL instructor
      - `OWNER` — REAL instructor who is the owner of the site
      - `INSTRUCTOR` — REAL instructor who has access to specified list of programs
      - `MANAGER` — REAL instructor who has the same rights as program owner

    </ResponseField>

    <ResponseField name="programIds" type="string[]">

      IDs of programs instructor is assigned to

      **Read-only.** Constraints: `format GUID`, `maxItems 100`.

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time when the instructor was created

      **Read-only.**

    </ResponseField>

  </Tab>
