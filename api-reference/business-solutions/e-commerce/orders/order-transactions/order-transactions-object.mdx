---
title: "Order Transactions Object"
description: "A record of all payments and refunds associated with an order."
icon: "cube"
---

A record of all payments and refunds associated with an order.

Use order transactions to track payment status, add payment records, process refunds, and monitor chargebacks.

<Tabs>

  <Tab title="REST">

    <ResponseField name="orderId" type="string">

      Order ID.

      Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="payments" type="Payment[]">

      Record of payments made to the merchant.

      Constraints: `maxItems 100`.

      <Expandable title="payments properties">

        <ResponseField name="id" type="string">

          Payment ID.

          **Read-only.**

        </ResponseField>

        <ResponseField name="createdDate" type="datetime">

          Date and time the payment was created in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#Combined_date_and_time_representations) format. Defaults to current time when not provided.

        </ResponseField>

        <ResponseField name="updatedDate" type="datetime">

          Date and time the payment was last updated in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#Combined_date_and_time_representations) format.

          **Read-only.**

        </ResponseField>

        <ResponseField name="amount" type="Price">

          Payment amount.

          <Expandable title="amount properties">

            <ResponseField name="amount" type="string">

              Amount.

              Constraints: `>= 0`, `<= 1000000000000000`.

            </ResponseField>

            <ResponseField name="formattedAmount" type="string">

              Amount formatted with currency symbol.

              **Read-only.**

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="refundDisabled" type="boolean">

          Whether refunds for this payment are disabled.
          + `true`: This payment is not refundable.
          + `false`: This payment may be refunded. However, this ultimately depends on the payment provider.

        </ResponseField>

        <ResponseField name="cashRounding" type="CashRoundingDetails">

          Details about cash rounding applied to this payment, when relevant.

          **Developer Preview.**

          <Expandable title="cashRounding properties">

            <ResponseField name="unroundedAmount" type="Price">

              Payment amount before cash rounding was applied.
              When provided, balance and status calculations use this value instead of `amount`
              to prevent false overpayment or underpayment due to rounding.

              **Developer Preview.**

              <Expandable title="unroundedAmount properties">

                <ResponseField name="amount" type="string">

                  Amount.

                  Constraints: `>= 0`, `<= 1000000000000000`.

                </ResponseField>

                <ResponseField name="formattedAmount" type="string">

                  Amount formatted with currency symbol.

                  **Read-only.**

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="roundingAdjustment" type="CashRounding">

              The difference between 'amount' and 'unroundedAmount' (amount minus unroundedAmount).
              A positive value indicates the price was rounded up; a negative value indicates a round-down.

              **Read-only.** **Developer Preview.**

              <Expandable title="roundingAdjustment properties">

                <ResponseField name="amount" type="string">

                  Amount, can be negative or positive.

                  Constraints: `>= -1000000000000000`, `<= 1000000000000000`.

                </ResponseField>

                <ResponseField name="formattedAmount" type="string">

                  Amount formatted with currency symbol.

                  **Read-only.**

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="refunds" type="Refund[]">

      Record of refunds made to the buyer.

      Constraints: `maxItems 300`.

      <Expandable title="refunds properties">

        <ResponseField name="id" type="string">

          Refund ID.

          **Read-only.** Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="transactions" type="RefundTransaction[]">

          List of transactions.

          Constraints: `minItems 0`, `maxItems 50`.

          <Expandable title="transactions properties">

            <ResponseField name="paymentId" type="string">

              ID of the payment associated with this refund.

              Constraints: `format GUID`, `immutable`.

            </ResponseField>

            <ResponseField name="amount" type="Price">

              Refund amount.

              Constraints: `immutable`.

              <Expandable title="amount properties">

                <ResponseField name="amount" type="string">

                  Amount.

                  Constraints: `>= 0`, `<= 1000000000000000`.

                </ResponseField>

                <ResponseField name="formattedAmount" type="string">

                  Amount formatted with currency symbol.

                  **Read-only.**

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="refundStatus" type="string">

              Refund status.

              **Supported values:**

              - `PENDING` — Refund was initiated on payment provider side. PENDING status was assigned by provider.
              - `SUCCEEDED` — Refund transaction succeeded.
              - `FAILED` — Refund transaction failed.
              - `SCHEDULED` — Refund request acknowledged, and will be executed soon.
              - `STARTED` — Refund was initiated on payment provider side.

            </ResponseField>

            <ResponseField name="refundStatusInfo" type="RefundStatusInfo">

              Optional details of current refund status.

              <Expandable title="refundStatusInfo properties">

                <ResponseField name="paymentGatewayReasonCode" type="string">

                  Reason code for the refund's current status.

                  Learn more about [reason codes](https://dev.wix.com/docs/rest/business-management/payments/service-plugins/payment-service-provider-service-plugin/reason-codes).

                </ResponseField>

                <ResponseField name="description" type="string">

                  Free text explanation of current refund status.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="gatewayRefundId" type="string">

              Payment gateway's refund ID.
              This field is only returned when the value of `external_refund` is `false`.

            </ResponseField>

            <ResponseField name="providerRefundId" type="string">

              ID of the refund in the payment provider's system. For example, at PayPal, Square, Stripe, etc. Not returned for external refunds.

            </ResponseField>

            <ResponseField name="externalRefund" type="boolean">

              Whether refund was made externally and manually on the payment provider's side.

              Constraints: `immutable`.

            </ResponseField>

          </Expandable>

        </ResponseField>
