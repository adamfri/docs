    Retrieves a list of files in the Media Manager.

    To retrieve a list of files within a specific folder in the Media Manager, specify the folder's ID in the `parentFolderId` parameter. If no folder is specified, the method retrieves the list of files in the root folder of the Media Manager.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-MEDIA.READ-MEDIAMANAGER">Read Media Manager</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    GET https://www.wixapis.com/site-media/v1/files
    ```

    ## Query Params

    <ParamField query="parentFolderId" type="string">

      ID of the file's parent folder.

      Default: `media-root`.

    </ParamField>

    <ParamField query="mediaTypes" type="string[]">

      File media type.

      Constraints: `maxItems 30`.

      **Supported values:**

      - `IMAGE`
      - `VIDEO`
      - `AUDIO`
      - `DOCUMENT`
      - `VECTOR`
      - `ARCHIVE`
      - `MODEL3D`
      - `OTHER`

    </ParamField>

    <ParamField query="private" type="boolean">

      Whether to return only private or only public files.
      + `true`: Returns only private files.
      + `false`: Returns only public files.
      + `undefined`: Returns public and private files.

      Learn more about [private files](https://dev.wix.com/docs/rest/assets/media/media-manager/files/private-files).

    </ParamField>

    <ParamField query="sort" type="Sorting">

      Field name and order to sort by. One of:

      `displayName`
      `updatedDate`
      `sizeInBytes`

      Default: `updatedDate` in `desc` order.

      <Expandable title="sort properties">

        <ParamField query="fieldName" type="string">

          Name of the field to sort by.

          Constraints: `maxLength 512`.

        </ParamField>

        <ParamField query="order" type="string">

          Sort order.

          **Supported values:**

          - `ASC`
          - `DESC`

        </ParamField>

        <ParamField query="origin" type="AddressLocation">

          Origin point for geo-distance sorting on a GEO field
          results are ordered by distance from this point (ASC = nearest first, DESC = farthest first).

          <Expandable title="origin properties">

            <ParamField query="latitude" type="number">

              Address latitude.

            </ParamField>

            <ParamField query="longitude" type="number">

              Address longitude.

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField query="paging" type="CursorPaging">

      Cursor and paging information.

      <Expandable title="paging properties">

        <ParamField query="limit" type="integer">

          Maximum number of items to return in the results.

        </ParamField>

        <ParamField query="cursor" type="string">

          Pointer to the next or previous page in the list of results.

          Pass the relevant cursor token from the `pagingMetadata` object in the previous call's response.
          Not relevant for the first request.

        </ParamField>

      </Expandable>

    </ParamField>

    ## Response Object

    <ResponseField name="files" type="FileDescriptor[]">

      List of files in the Media Manager.

      Constraints: `maxItems 200`.

      See the [File Descriptor Object](/api-reference/assets/media/media-manager/files/file-descriptor-object) for this object's fields.

    </ResponseField>

    <ResponseField name="nextCursor" type="PagingMetadataV2">

      The next cursor if it exists.

      <Expandable title="nextCursor properties">

        <ResponseField name="total" type="integer">

          Total number of items that match the query. Returned if offset paging is used and the `tooManyToCount` flag is not set.

        </ResponseField>

        <ResponseField name="cursors" type="Cursors">

          Cursors to navigate through the result pages using `next` and `prev`. Returned if cursor paging is used.

          <Expandable title="cursors properties">

            <ResponseField name="next" type="string">

              Cursor string pointing to the next page in the list of results.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    ```javascript Generate a download url for all images in media (with $w)
    /***********************************************
     * Backend code - download-files-url.web.js/ts *
     **********************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { files } from '@wix/media';
    import { auth } from '@wix/essentials';

    export const generateDownloadUrl = webMethod(Permissions.Anyone, async (fileIds) => {
      try {
        const elevatedGenerateFilesDownloadUrl = auth.elevate(files.generateFilesDownloadUrl)
        const generatedUrl = await elevatedGenerateFilesDownloadUrl(fileIds);

        return generatedUrl.downloadUrl;
      } catch (error) {
        console.error(error);
      }
    });

    export const listImages = webMethod(Permissions.Anyone, async () => {
      const listOptions = {
        mediaTypes: ["IMAGE"]
      }

      try {
        const elevatedListFiles = auth.elevate(files.listFiles)
        const images = await elevatedListFiles(listOptions);

        return images.files;
      } catch (error) {
        console.error(error);
      }
    });

    /*************
     * Page code *
     ************/

    import { listImages, generateDownloadUrl } from 'backend/download-files-url.web';

    $w.onReady(() => {
      $w('#downloadFiles').disable();
      getDownloadUrl();
    });

    async function getDownloadUrl() {
      const images = await listImages();

      const imageIds = images.map((image) => {
        return image._id;
      });
      const downloadUrl = await generateDownloadUrl(imageIds);

      $w('#downloadFiles').link = downloadUrl;
      await $w('#downloadFiles').enable();
    }
    ```

    ```javascript Update a file's display name from visitor input (with $w)
    /****************************************
     * Backend code - update-file.web.js/ts *
     ***************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { files } from '@wix/media';
    import { auth } from '@wix/essentials';

    export const updateFile = webMethod(Permissions.Anyone, async (fileId, file) => {
      try {
        const elevatedUpdateFileDescriptor = auth.elevate(files.updateFileDescriptor);
        const updatedDescriptor = await elevatedUpdateFileDescriptor({
          _id: fileId, 
          ...file 
        });

        return updatedDescriptor;
      } catch (error) {
        console.error(error);
      }
    });

    export const listImages = webMethod(Permissions.Anyone, async () => {
      const listOptions = {
        mediaTypes: ["IMAGE"]
      }

      try {
        const elevatedListFiles = auth.elevate(files.listFiles)
        const images = await elevatedListFiles(listOptions);

        return images.files;
      } catch (error) {
        console.error(error);
      }
    });

    /*************
     * Page code *
     ************/

    import { updateFile, listImages } from 'backend/update-file.web';

    $w.onReady(async () => {
      await populateImagesDropdown();

      $w('#update').onClick(async () => {
        const imageId = $w('#imagesDropdown').value;

        const fileUpdates = {
          displayName: $w('#newDisplayName').value
        }

        const updatedLocation = await updateFile(imageId, fileUpdates);
        console.log('Display name successfully updated to:', updatedLocation.displayName)
        $w('#updateSuccessMsg').show();
      });
    });

    async function populateImagesDropdown() {
      const images = await listImages();
      const dropdownOptions = images.map((image) => {
        return {
          label: image.displayName,
          value: image._id
        };
      });
    ```
