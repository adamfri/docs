    Queries categories using [WQL - Wix Query Language](https://dev.wix.com/api/rest/getting-started/api-query-language).

    Supports filtering, sorting, and pagination.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.SUPPLIERS_HUB.READ_MARKETPLACE">Read Marketplace</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/suppliers-hub/v1/categories/query
    ```

    ## Body Params

    <ParamField body="query" type="CursorQuery">

      Query parameters including filter, sort, and paging options.

      <Expandable title="query properties">

        <ParamField body="filter" type="object">

          Filter object in the following format:
          `"filter" : {`
          "fieldName1": "value1",
          "fieldName2":&#123;"$operator":"value2"&#125;
          &#125;``
          Example of operators: `$eq`, `$ne`, `$lt`, `$lte`, `$gt`, `$gte`, `$in`, `$hasSome`, `$hasAll`, `$startsWith`, `$contains`

        </ParamField>

        <ParamField body="sort" type="Sorting[]">

          Sort object in the following format:
          `[{"fieldName":"sortField1","order":"ASC"},{"fieldName":"sortField2","order":"DESC"}]`

          Constraints: `maxItems 5`.

          <Expandable title="sort properties">

            <ParamField body="fieldName" type="string">

              Name of the field to sort by.

              Constraints: `maxLength 512`.

            </ParamField>

            <ParamField body="order" type="string">

              Sort order.

              **Supported values:**

              - `ASC`
              - `DESC`

            </ParamField>

            <ParamField body="selectItemsBy" type="object[]">

              Filters to specify which element to use for sorting when `fieldName` refers to a repeated field marked as `MATCH_ITEMS`.

              These filters ensure the correct element is selected from the collection for sorting. Multiple filters are combined using the `AND` operator.

              For example, for a document like `{"id": "1", "nestedField": [{"price": 10, "region": "EU"}, {"price": 20, "region": "US"}]}`,
              if `nestedField` is marked as `MATCH_ITEMS`, sorting by price in the US region requires:
              `{fieldName: "nestedField.price", "selectItemsBy": [{"nestedField.region": "US"}]}`.

              Constraints: `maxItems 10`.

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

      <Expandable title="One of the following">

        <ParamField body="pagingMethod" type="pagingMethod">

          pagingMethod

          <Expandable title="pagingMethod properties">

            <ParamField body="cursorPaging" type="CursorPaging">

              Cursor token pointing to a page of results. Not used in the first request. Following requests use the cursor token and not `filter` or `sort`.

              _Nested structure omitted for brevity — see the `CursorPaging` object._

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="namespace" type="string" required>

      The namespace this category belongs to.

      **Supported values:**

      - `DROPSHIPPING_NAMESPACE` — Namespace for the dropshipping category tree.
      - `WHOLESALE_NAMESPACE` — Namespace for the wholesale category tree.
      - `PRINT_ON_DEMAND_NAMESPACE` — Namespace for the print on demand category tree.

    </ParamField>

    <ParamField body="returnNonVisibleCategories" type="boolean">

      Whether non visible categories should be included in the response. Default is `false`.

    </ParamField>

    ## Response Object

    <ResponseField name="categories" type="Category[]">

      Categories matching the query.

      <Expandable title="categories properties">

        <ResponseField name="id" type="string">

          Category ID.

          **Read-only.**

        </ResponseField>

        <ResponseField name="name" type="string">

          Category name.

        </ResponseField>

        <ResponseField name="description" type="string">

          Category description.

        </ResponseField>

        <ResponseField name="itemCount" type="integer">

          Number of items in this category.

          **Read-only.**

        </ResponseField>

        <ResponseField name="breadcrumbs" type="Breadcrumb[]">

          Navigation path from root to this category.

          The current category isn't included in the list.

          **Read-only.** Constraints: `maxItems 5`.

          <Expandable title="breadcrumbs properties">

            <ResponseField name="categoryId" type="string">

              Ancestor category ID.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="categoryName" type="string">

              Ancestor category name.

              Constraints: `minLength 1`, `maxLength 80`.

            </ResponseField>

            <ResponseField name="categorySlug" type="string">

              Ancestor category slug.

              Constraints: `minLength 1`, `maxLength 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="parentCategory" type="ParentCategory">

          Parent category information in the hierarchy.

          Constraints: `immutable`.

          <Expandable title="parentCategory properties">

            <ResponseField name="id" type="string">

              Parent category ID.

            </ResponseField>

            <ResponseField name="index" type="integer">

              Index position of the category within the parent category.

              **Read-only.**

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="slug" type="string">

          URL-friendly category identifier.

        </ResponseField>

        <ResponseField name="visible" type="boolean">

          Whether the category is visible in the Marketplace.

          Default: `true`

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="pagingMetadata" type="CursorPagingMetadata">

      Paging information for retrieving additional results.

      <Expandable title="pagingMetadata properties">

        <ResponseField name="count" type="integer">

          Number of items returned in the response.

        </ResponseField>

        <ResponseField name="cursors" type="Cursors">

          Cursor strings that point to the next page, previous page, or both.

          <Expandable title="cursors properties">

            <ResponseField name="next" type="string">

              Cursor string pointing to the next page in the list of results.

            </ResponseField>

            <ResponseField name="prev" type="string">

              Cursor pointing to the previous page in the list of results.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="hasNext" type="boolean">

          Whether there are more pages to retrieve following the current page.

          + `true`: Another page of results can be retrieved.
          + `false`: This is the last page.

        </ResponseField>

      </Expandable>

    </ResponseField>
