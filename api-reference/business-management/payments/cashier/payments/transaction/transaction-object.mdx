        <ResponseField name="subscription" type="Subscription">

          Subscription information.

          <Expandable title="subscription properties">

            <ResponseField name="frequency" type="string">

              Frequency of recurring payment.

              For example, if payments are made every 3 months, `frequency` is `MONTH` and `interval` is `3`.

              **Supported values:**

              - `UNDEFINED`
              - `DAY`
              - `WEEK`
              - `MONTH`
              - `YEAR`

            </ResponseField>

            <ResponseField name="interval" type="integer">

              Interval of recurring payment.

              For example, if payments are made every 3 months, `frequency` is `MONTH` and `interval` is `3`.

            </ResponseField>

            <ResponseField name="status" type="string">

              Subscription status.

              **Supported values:**

              - `SUBSCRIPTION_UNDEFINED` — Subscription status is undefined.
              - `SUBSCRIPTION_ACTIVE` — Recurring payment is active.
              - `SUBSCRIPTION_CANCELLED` — Recurring payment was cancelled.
              - `SUBSCRIPTION_FINISHED` — Total billing cycle for the optional trial or the regular payment period was completed. This option is not currently supported.
              - `SUBSCRIPTION_PENDING` — Recurring payment is not active yet. The first payment has not been paid.
              - `SUBSCRIPTION_SUSPENDED` — Recurring payment was suspended.

            </ResponseField>

            <ResponseField name="billingCycles" type="integer">

              Number of billing cycles for the subscription period.

              If `0`, subscription is active until cancelled.

            </ResponseField>

          </Expandable>

        </ResponseField>

<ResponseField name="refundability" type="Refundability">

  Transaction refundability status.

  <Expandable title="refundability properties">

    <ResponseField name="isRefundable" type="boolean">

      Whether the transaction is refundable.

    </ResponseField>

    <ResponseField name="reason" type="NonRefundabilityReason">

      Reason the transaction is not refundable. Relevant only when the transaction isn't refundable.

      <Expandable title="reason properties">

        <ResponseField name="value" type="string">

          **Supported values:**

          - `ALREADY_REFUNDED` — Refund has already been processed.
          - `PROVIDER_IS_DOWN` — The provider is disabled due to temporary issues.
          - `NOT_SUPPORTED` — The provider doesn't currently support refunds, the transaction is not refundable, or necessary information is missing.
          - `PENDING_REFUND` — A refund has been initiated and is awaiting confirmation from the provider.
          - `EXPIRED` — Refund period has expired.
          - `FORBIDDEN` — Wix user doesn't have required permissions to refund this transaction.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="flowType" type="string">

      Refund flow type.

      **Supported values:**

      - `NA`
      - `SYNC`
      - `ASYNC`

    </ResponseField>

  </Expandable>

</ResponseField>
