Manage Discount Rules.

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Discount rule ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the discount rule is updated.
      To prevent conflicting changes, the current `revision` must be passed when updating the discount rule.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the discount rule was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the discount rule was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="active" type="boolean">

      Whether the discount rule is active.

      Default: `true`

    </ResponseField>

    <ResponseField name="name" type="string">

      Discount rule name.

    </ResponseField>

    <ResponseField name="trigger" type="DiscountTrigger">

      Discount rule trigger.
      A set of conditions that must be met for the `discounts` to be applied.
      Not passing a trigger will cause the discount to always apply.

      <Expandable title="trigger properties">

        <ResponseField name="triggerType" type="string">

          Trigger type.

          **Supported values:**

          - `AND` — Operator used for chaining multiple triggers. Currently 1 `AND` chain operator is supported.
          - `SUBTOTAL_RANGE` — Subtotal must be within the specified `subtotalRange` values.
          - `ITEM_QUANTITY_RANGE` — Quantity of items in scope must be within specified `itemQuantityRange` values.
          - `CUSTOM` — Custom trigger type defined in `customTrigger` object.
          - `OR` — Chain multiple triggers with OR operator
          - `CUSTOMER_ELIGIBILITY` — Customer eligibility type must be within specified `customerEligibility` values.

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="trigger" type="trigger">

          trigger

          <Expandable title="trigger properties">

            <ResponseField name="and" type="And">

              Chain multiple triggers with the `and` operator.

              <Expandable title="and properties">

                <ResponseField name="triggers" type="DiscountTrigger[]">

                  List of triggers to have an "AND" operator between their results.

                  Constraints: `minItems 2`, `maxItems 5`.

                  _Recursive reference to `DiscountTrigger[]`._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="or" type="Or">

              Chain multiple triggers with the `or` operator.

              <Expandable title="or properties">

                <ResponseField name="triggers" type="DiscountTrigger[]">

                  List of triggers to have an "OR" operator between their results.

                  Constraints: `minItems 2`, `maxItems 5`.

                  _Recursive reference to `DiscountTrigger[]`._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="customTrigger" type="Custom">

              Custom trigger.

              <Expandable title="customTrigger properties">

                <ResponseField name="id" type="string">

                  Trigger ID.

                  Constraints: `minLength 1`, `maxLength 100`.

                </ResponseField>

                <ResponseField name="appId" type="string">

                  ID of the app that created the trigger.

                  Constraints: `format GUID`.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="subtotalRange" type="SubtotalRange">

              Subtotal trigger range.

              <Expandable title="subtotalRange properties">

                <ResponseField name="scopes" type="Scope[]">

                  Relevant scopes for `SPECIFIC_ITEMS` target type.

                  Constraints: `maxItems 50`.

                  _Nested structure omitted for brevity — see the `Scope[]` object._

                </ResponseField>

                <ResponseField name="from" type="string">

                  Minimum subtotal price (inclusive).

                </ResponseField>

                <ResponseField name="to" type="string">

                  Maximum subtotal price (inclusive).

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="itemQuantityRange" type="ItemQuantityRange">

              Item quantity trigger range.

              <Expandable title="itemQuantityRange properties">

                <ResponseField name="scopes" type="Scope[]">

                  Relevant scopes for `SPECIFIC_ITEMS` target type.

                  Constraints: `maxItems 50`.

                  _Nested structure omitted for brevity — see the `Scope[]` object._

                </ResponseField>

                <ResponseField name="from" type="integer">

                  Minimum item quantity (inclusive).

                </ResponseField>

                <ResponseField name="to" type="integer">

                  Maximum item quantity (inclusive).

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="customerEligibility" type="CustomerEligibility">

              Customer Eligibility info

              **Developer Preview.**

              <Expandable title="customerEligibility properties">

                <ResponseField name="eligibilityType" type="string">

                  Customer eligibility type.

                  **Supported values:**

                  - `INDIVIDUAL_MEMBERS` — Only specific customers based on member id.

                </ResponseField>

              </Expandable>

              <Expandable title="One of the following">

                <ResponseField name="options" type="options">

                  options

                  _Nested structure omitted for brevity — see the `options` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="activeTimeInfo" type="ActiveTimeInfo">

      Time frame in which the discount rule is active.

      <Expandable title="activeTimeInfo properties">

        <ResponseField name="start" type="datetime">

          Date and time the discount rule is active **from**, in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#Combined_date_and_time_representations) format.

        </ResponseField>

        <ResponseField name="end" type="datetime">

          Date and time the discount rule is active **till**, in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#Combined_date_and_time_representations) format.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="discounts" type="Discounts">

      List of discounts that are applied when one or more triggers are met.

      > **Notes:**
      >
      > + Currently, only 1 discount can be defined per discount rule.
      > + A discount rule can be applied on multiple items per cart.
      > + A discount rule can only be applied once per item.

      <Expandable title="discounts properties">

        <ResponseField name="values" type="Discount[]">

          Discounts.

          Constraints: `minItems 1`, `maxItems 1`.

          <Expandable title="values properties">

            <ResponseField name="targetType" type="string">

              Discount target.

              + `SPECIFIC_ITEMS`: Discount applies to a specific set of items.

              **Supported values:**

              - `SPECIFIC_ITEMS` — Target type is a set of specific items

            </ResponseField>

            <ResponseField name="specificItemsInfo" type="SpecificItemsInfo">

              Data related to `SPECIFIC_ITEMS` target type.

              <Expandable title="specificItemsInfo properties">

                <ResponseField name="scopes" type="Scope[]">

                  All associated scopes for `SPECIFIC_ITEMS` target type.

                  Constraints: `minItems 1`, `maxItems 50`.

                  _Nested structure omitted for brevity — see the `Scope[]` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="discountType" type="string">

              Discount type.

              **Supported values:**

              - `PERCENTAGE` — Price is reduced by percentage value.
              - `FIXED_AMOUNT` — Price is reduced by fixed amount.
              - `FIXED_PRICE` — Price will be set to fixed amount.
              - `QUANTITY_BASED` — Quantity based options for Buy X For Amount.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="status" type="string">

      Discount rule status.
      Default: `UNDEFINED`

      **Read-only.**

      **Supported values:**

      - `LIVE` — Rule status is live.
      - `EXPIRED` — Rule status is expired, it might have been live in the past.
      - `PENDING` — Rule status is pending, it might be live in the future.

    </ResponseField>

    <ResponseField name="usageCount" type="integer">

      Number of times the discount rule was used.

      **Read-only.**

    </ResponseField>

    <ResponseField name="tags" type="Tags">

      Tags for this discount rule.

      <Expandable title="tags properties">

        <ResponseField name="privateTags" type="TagList">

          Tags that require an additional permission in order to access them, normally not given to site members or visitors.

          <Expandable title="privateTags properties">

            <ResponseField name="tagIds" type="string[]">

              List of tag IDs.

              Constraints: `maxLength 5`, `maxItems 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="publicTags" type="TagList">

          Tags that are exposed to anyone who has access to the labeled entity itself, including site members and visitors.

          <Expandable title="publicTags properties">

            <ResponseField name="tagIds" type="string[]">

              List of tag IDs.

              Constraints: `maxLength 5`, `maxItems 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>
