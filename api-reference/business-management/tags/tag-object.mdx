<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Tag ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the tag is updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the tag was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the tag was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Tag name.

    </ResponseField>

    <ResponseField name="fqdn" type="string">

      FQDN of the entity that belongs to this tag.

    </ResponseField>
