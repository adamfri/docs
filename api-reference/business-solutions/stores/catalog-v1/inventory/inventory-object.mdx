Read & Write inventory from Wix Stores Inventory

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Inventory item ID.

    </ResponseField>

    <ResponseField name="externalId" type="string">

      Deprecated: use productId.

      **Read-only.**

    </ResponseField>

    <ResponseField name="productId" type="string">

      Product ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="trackQuantity" type="boolean">

      Whether quantity is being tracked.

    </ResponseField>

    <ResponseField name="variants" type="InventoryVariantV2[]">

      Variants associated with this inventory item.

      <Expandable title="variants properties">

        <ResponseField name="variantId" type="string">

          Variant ID.

        </ResponseField>

        <ResponseField name="inStock" type="boolean">

          Whether the product is listed as in stock.

        </ResponseField>

        <ResponseField name="quantity" type="integer">

          Quantity currently left in inventory.

        </ResponseField>

        <ResponseField name="availableForPreorder" type="boolean">

          Whether the variant is available for preorder. When `true`, the variant is out of stock and preorder is enabled on inventory level.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="lastUpdated" type="datetime">

      Last updated timestamp.

      **Read-only.**

    </ResponseField>

    <ResponseField name="numericId" type="number">

      Inventory’s unique numeric ID (assigned in ascending order).
      Primarily for sorting and filtering when crawling all inventories.

      **Read-only.**

    </ResponseField>

    <ResponseField name="preorderInfo" type="PreorderInfo">

      Preorder information.

      <Expandable title="preorderInfo properties">

        <ResponseField name="enabled" type="boolean">

          Whether the item is available for preorder.

        </ResponseField>

        <ResponseField name="message" type="string">

          A message the buyer will see when the item is out of stock and preorder is enabled.

        </ResponseField>

        <ResponseField name="limit" type="integer">

          Number of products that can be preordered after stock reaches zero.

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Tab>
