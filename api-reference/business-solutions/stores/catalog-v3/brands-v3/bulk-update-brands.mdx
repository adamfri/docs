    Updates multiple brands.

    Each time a brand is updated, `revision` increments by 1.
    The current `revision` must be passed when updating a brand.
    This ensures you're working with the latest brand and prevents unintended overwrites.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.STORES.BRAND_WRITE">Brand write in v3 catalog</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/stores/v3/bulk/brands/update
    ```

    ## Body Params

    <ParamField body="brands" type="MaskedBrand[]" required>

      List of brands to update.

      Constraints: `minItems 1`, `maxItems 100`.

      <Expandable title="brands properties">

        <ParamField body="brand" type="Brand">

          Brand to update.

          See the [Brand Object](/api-reference/business-solutions/stores/catalog-v3/brands-v3/brand-object) for this object's fields.

        </ParamField>

        <ParamField body="fieldMask" type="FieldMask">

          Explicit list of fields to update.

          <Expandable title="fieldMask properties">

            <ParamField body="paths" type="string[]" />

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="returnEntity" type="boolean">

      Whether to return the full updated brand entities in the response.

      Default: `false`

    </ParamField>

    <ParamField body="fields" type="string[]">

      Fields to include in the response.

      Supported values: `ASSIGNED_PRODUCTS_COUNT`

      Constraints: `maxItems 100`.

      **Supported values:**

      - `ASSIGNED_PRODUCTS_COUNT`

    </ParamField>

    import { brandsV3 } from "@wix/stores";

    const brands = [
      {
        brand: {
          id: "f436b313-8437-4b91-a650-dc3944273781",
          revision: "1",
          name: "Asics"
        }
      },
      {
        brand: {

          revision: "1",
          name: "NB"
        }
      }
    ];

    const options = {
      returnEntity: true
    };

    async function bulkUpdateBrands() {
      const response = await brandsV3.bulkUpdateBrands(brands, options);
