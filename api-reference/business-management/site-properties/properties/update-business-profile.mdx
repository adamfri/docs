    Updates a site's business profile.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-SITEPROP.MANAGE-BUSINESS-PROFILE">Manage Business Profile</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/site-properties/v4/properties/business-profile
    ```

    ## Body Params

    <ParamField body="businessProfile" type="BusinessProfileData">

      The site's business profile.

      <Expandable title="businessProfile properties">

        <ParamField body="siteDisplayName" type="string">

          Site display name.

        </ParamField>

        <ParamField body="businessName" type="string">

          Business name.

        </ParamField>

        <ParamField body="logo" type="string">

          Path to the site's logo in Wix Media (without Wix Media base URL).

        </ParamField>

        <ParamField body="description" type="string">

          Site description.

        </ParamField>

        <ParamField body="companyId" type="string">

          Company ID issued to the business by local authorities. Appears on invoices and price quotes.

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="fields" type="FieldMask">

      The properties of `businessProfile` to be updated. Properties not explicitly specified here are ignored. Properties included here but excluded from `businessProfile` are cleared.

      <Expandable title="fields properties">

        <ParamField body="paths" type="string[]" />

      </Expandable>

    </ParamField>

    ```javascript updateBusinessProfile() for only specific fields (with $w)
    /****************************************************
     * Backend code - update-business-profile.web.js/ts *
     ***************************************************/
    import { Permissions, webMethod } from '@wix/web-methods';
    import { siteProperties } from '@wix/business-tools';
    import { auth } from '@wix/essentials';

    export const profileDetailsUpdate = webMethod(Permissions.Anyone, async (businessProfile, fields) => {
      try {
        const elevatedUpdateBusinessProfile = auth.elevate(siteProperties.updateBusinessProfile);
        await elevatedUpdateBusinessProfile(businessProfile, fields);

        console.log('Success! Profile Details Updated');
        return true;
      } catch (error) {
        console.error(error);
        throw new Error(error);
      }
    });

    /*************
     * Page code *
     ************/

    import { profileDetailsUpdate } from 'backend/update-business-profile.web';

    $w.onReady(() => {
      $w('#submit').onClick(async () => {
        const paramDetails = getParamDetails();

        if (!paramDetails.checkedFields.length) {
          $w('#changeAFieldsMsg').show();
          setTimeout(() => {
            $w('#changeAFieldsMsg').hide();
          }, 10000);
        } else {
          const isUpdated = await profileDetailsUpdate(paramDetails.updateDetails, paramDetails.checkedFields);

          if (isUpdated) {
            console.log('Business Profile successfully updated');
            $w('#successfulUpdateMsg').show();
            setTimeout(() => {
              $w('#successfulUpdateMsg').hide();
            }, 10000);
          }
        }
      });
    });

    function getParamDetails() {
      const checkedFields = [];
      const updateDetails = {};
      if ($w('#description').value.length) {
          checkedFields.push('description');
          updateDetails.description = $w('#description').value;
      }
      if ($w('#logo').value.length) {
          checkedFields.push('logo');
          updateDetails.logo = $w('#logo').value;
      }

      return {updateDetails, checkedFields};
    }
    ```
