---
title: "Marketing Consent Object"
description: "The Marketing Consent object in the Wix Marketing Consent API."
icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Marketing consent ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number.

      **Read-only.**

    </ResponseField>

    <ResponseField name="details" type="MarketingConsentDetails">

      Marketing consent communication details.

      <Expandable title="details properties">

        <ResponseField name="type" type="string">

          The communication channel of the marketing consent.

          **Supported values:**

          - `UNKNOWN_TYPE`
          - `EMAIL`
          - `PHONE`

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="identifier" type="identifier">

          identifier

          <Expandable title="identifier properties">

            <ResponseField name="email" type="string">

              Email address used for the marketing consent.

              Constraints: `format EMAIL`.

            </ResponseField>

            <ResponseField name="phone" type="string">

              Phone number in [E.164](https://en.wikipedia.org/wiki/E.164) format used for the SMS marketing consent.

              Constraints: `format PHONE`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="state" type="string">

      Marketing consent state.

      Default: `UNKNOWN_STATE`.

      **Supported values:**

      - `UNKNOWN_STATE` — State of the marketing consent is unknown.
      - `NEVER_CONFIRMED` — The site visitor never confirmed to receive marketing consents.
      - `REVOKED` — The marketing consent has been removed, for example, when a site visitor unsubscribes from a newsletter.
      - `PENDING` — The marketing consent is pending confirmation. Relevant only for `{"optInLevel": "DOUBLE_CONFIRMATION"}`.
      - `CONFIRMED` — The site visitor has confirmed their marketing consent.

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the marketing consent was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the marketing consent was updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="lastConfirmationActivity" type="LastConfirmationActivity">

      Last confirmation activity of the marketing consent.

      <Expandable title="lastConfirmationActivity properties">

        <ResponseField name="source" type="string">

          Source of the given consent (how the site visitor signed up).

          **Supported values:**

          - `IN_PERSON`
          - `FORM`
          - `LINK_CONFIRMATION`
          - `EMAIL_SERVICE`
          - `WIX_USERS`
          - `OTHER`

        </ResponseField>

        <ResponseField name="description" type="string">

          Consent description.

        </ResponseField>

        <ResponseField name="updatedDate" type="datetime">

          Date and time the consent was updated.

        </ResponseField>

        <ResponseField name="optInLevel" type="string">

          Consent opt in level, either single or double confirmation.

          **Supported values:**

          - `UNKNOWN_OPT_IN_LEVEL`
          - `SINGLE_CONFIRMATION`
          - `DOUBLE_CONFIRMATION`

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="lastRevokeActivity" type="LastRevokeActivity">

      Last revoke activity of the marketing consent.

      <Expandable title="lastRevokeActivity properties">

        <ResponseField name="source" type="string">

          Source of the given revoke (how the visitor signed up).

          **Supported values:**

          - `IN_PERSON`
          - `FORM`
          - `REVOKE_LINK`
          - `WIX_USERS`
          - `OTHER`

        </ResponseField>

        <ResponseField name="description" type="string">

          Details about the revoke, if relevant.

        </ResponseField>

        <ResponseField name="updatedDate" type="datetime">

          Date and time the consent was updated.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Additional fields.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "id": "aa8a62f5-9a79-4b21-b1a4-5c835c08d937",
        "revision": "1",
        "details": {
          "type": "EMAIL",
          "email": "test@test.com"
        },
        "state": "CONFIRMED",
        "lastConfirmationActivity": {
          "source": "IN_PERSON",
          "description": "Person filed a printed document",
          "updatedDate": "2022-10-02T08:43:25.765Z",
          "optInLevel": "UNKNOWN_OPT_IN_LEVEL"
        },
        "extendedFields": {},
        "created_date": "2022-11-16T14:07:50.150Z",
        "updated_date": "2022-11-16T14:07:50.150Z"
      }
      ```

    </CodeGroup>
