    <div className="wix-ref">

    Retrieves a list of up to 200 files with the specified filtering, sorting, and cursor paging.

    If no parameters are specified, the method returns all files in the `MEDIA_ROOT` folder.

    ## Defaults

    Search Files has the following default setting, which you can override:
    + Sorted by `updatedDate` in descending order.

    ## Filters

    When using filters for dates, you must use
    [UTC time](https://en.wikipedia.org/wiki/Coordinated_Universal_Time).

    ## See also

    To learn about working with *Search* methods, see
    [API Query Language](https://dev.wix.com/docs/rest/articles/get-started/api-query-language)
    and   [Sorting and Paging](https://dev.wix.com/docs/rest/articles/get-started/sorting-and-paging).

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-MEDIA.READ-MEDIAMANAGER">Read Media Manager</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/site-media/v1/files/search
    ```

    ## Body Params

    <ParamField body="search" type="string">

      Free text to match in searchable fields.

      For example, search for specific values in a file's
      `displayName`, `mimeType`, or `label`.

    </ParamField>

    <ParamField body="rootFolder" type="string">

      The root folder in the media manager to search in.
      For more information, see Root Folders in [Terminology](https://dev.wix.com/docs/rest/assets/media/media-manager/introduction#terminology).

      Default: `MEDIA_ROOT`.

      **Supported values:**

      - `MEDIA_ROOT` — Contains all files and folders in the Media Manager's **Site Files** tab in the UI.
      - `TRASH_ROOT` — Contains all files and folders in the Media Manager's **Trash** tab in the UI.
      - `VISITOR_UPLOADS_ROOT` — Contains all files and folders created by site visitors or members.

    </ParamField>

    <ParamField body="mediaTypes" type="string[]">

      File media type.

      Constraints: `maxItems 30`.

      **Supported values:**

      - `IMAGE`
      - `VIDEO`
      - `AUDIO`
      - `DOCUMENT`
      - `VECTOR`
      - `ARCHIVE`
      - `MODEL3D`
      - `OTHER`

    </ParamField>

    <ParamField body="private" type="boolean">

      Whether to return only private or only public files.
      + `true`: Returns only private files.
      + `false`: Returns only public files.
      + `undefined`: Returns public and private files.

      Learn more [about private files](https://dev.wix.com/docs/api-reference/assets/media/media-manager/files/private-files).

    </ParamField>

    <ParamField body="sort" type="Sorting">

      Field name and order to sort by.

      One of:
      `displayName`
      `updatedDate`
      `sizeInBytes`

      Default: `updatedDate` in `desc` order.

      <Expandable title="sort properties">

        <ParamField body="fieldName" type="string">

          Name of the field to sort by.

          Constraints: `maxLength 512`.

        </ParamField>

        <ParamField body="order" type="string">

          Sort order.

          **Supported values:**

          - `ASC`
          - `DESC`

        </ParamField>

        <ParamField body="origin" type="AddressLocation">

          Origin point for geo-distance sorting on a GEO field
          results are ordered by distance from this point (ASC = nearest first, DESC = farthest first).

          <Expandable title="origin properties">

            <ParamField body="latitude" type="number">

              Address latitude.

            </ParamField>

            <ParamField body="longitude" type="number">

              Address longitude.

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="paging" type="CursorPaging">

      Cursor and paging information.

      <Expandable title="paging properties">

        <ParamField body="limit" type="integer">

          Maximum number of items to return in the results.

        </ParamField>

        <ParamField body="cursor" type="string">

          Pointer to the next or previous page in the list of results.

          Pass the relevant cursor token from the `pagingMetadata` object in the previous call's response.
          Not relevant for the first request.

        </ParamField>

      </Expandable>

    </ParamField>

    ## Response Object

    <ResponseField name="files" type="FileDescriptor[]">

      Retrieved files that match the search criteria specified in the request.

      Each file includes all standard file information.

      Constraints: `maxItems 200`.

      See the [File Descriptor Object](/api-reference/assets/media/media-manager/files/file-descriptor-object) for this object's fields.

    </ResponseField>

    <ResponseField name="nextCursor" type="PagingMetadataV2">

      The next cursor if it exists.

      <Expandable title="nextCursor properties">

        <ResponseField name="total" type="integer">

          Total number of items that match the query. Returned if offset paging is used and the `tooManyToCount` flag is not set.

        </ResponseField>

        <ResponseField name="cursors" type="Cursors">

          Cursors to navigate through the result pages using `next` and `prev`. Returned if cursor paging is used.

          <Expandable title="cursors properties">

            <ResponseField name="next" type="string">

              Cursor string pointing to the next page in the list of results.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    ## Errors

    | Status | Code | Application code | Description |
    | --- | --- | --- | --- |
    | `400` | `INVALID_ARGUMENT` | `UNSUPPORTED_MEDIA_TYPE` | Media type isn't supported for filtering. |
    | `400` | `INVALID_ARGUMENT` | `UNSUPPORTED_SORT_VALUE` | Sort value isn't supported. |

      This method may also return standard errors. [Learn more about standard Wix errors](https://dev.wix.com/docs/rest/articles/getting-started/errors).

    import { Permissions, webMethod } from '@wix/web-methods';
    import { files } from '@wix/media';
    import { auth } from '@wix/essentials';

    export const deleteFiles = webMethod(Permissions.Anyone, async (fileIds, options) => {
      try {
        const elevatedBulkDeleteFiles = auth.elevate(files.bulkDeleteFiles);
        await elevatedBulkDeleteFiles(fileIds, options);

        console.log(`Permanently deleted files with ids: ${fileIds.toString()}.`);
        return true;
      } catch (error) {
        console.error(error);
      }
    });

    export const searchFileIds = webMethod(Permissions.Anyone, async (parentFolderId) => {
      try {
        const options = { parentFolder: parentFolderId };
        const elevatedSearchFiles = auth.elevate(files.searchFiles);
        const returnedFiles = await elevatedSearchFiles(options);

        const fileIds = returnedFiles.files.map((file) => {
          return file._id;
        });

        return fileIds;
      } catch (error) {
        console.error(error);
      }
    });

        const parentFolder = $w('#parentFolder').value;
        const fileIds = await searchFileIds(parentFolder);

        await deleteFiles(fileIds, options);
        $w('#successMessage').show();
      });
