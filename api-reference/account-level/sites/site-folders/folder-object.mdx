<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Folder ID.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="name" type="string">

      Folder name.

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date the folder was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date the folder was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="siteCount" type="integer">

      Number of sites in the folder.

      **Read-only.**

    </ResponseField>

    <ResponseField name="parentId" type="string">

      Parent folder ID. When empty, the site is at root level.

    </ResponseField>

    <ResponseField name="path" type="FolderBasicInfo[]">

      Path including all parent folders from root to the current folder, in order. When empty, the site or folder is at root level.

      **Read-only.** Constraints: `maxItems 50`.

      <Expandable title="path properties">

        <ResponseField name="id" type="string">

          Folder ID.

          **Read-only.** Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="name" type="string">

          Folder name.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Tab>
