    ```javascript Full flow for creating a new product (with $w)
    /*************************************
     * Backend code - products.web.js/ts *
     *************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { products } from '@wix/stores';

    export const createProduct = webMethod(Permissions.Anyone, (product) => {
      return products.createProduct(product);
    });

    export const addProductsToCollection = webMethod(Permissions.Anyone, (_id, productIds) => {
      return products.addProductsToCollection(_id, productIds);
    });

    export const addProductMedia = webMethod(Permissions.Anyone, (_id, media) => {
      return products.addProductMedia(_id, media);
    });

     * Page code *
    ```
