---
title: "Post Object"

icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Post ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="categoryId" type="string">

      Category ID the post is listed under.

    </ResponseField>

    <ResponseField name="ownerId" type="string">

      Post owner's member ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="title" type="string">

      Post title.

    </ResponseField>

    <ResponseField name="contentText" type="string">

      The post content in plain text.

    </ResponseField>

    <ResponseField name="commentCount" type="integer">

      Total number of post comments.

      **Read-only.**

    </ResponseField>

    <ResponseField name="likeCount" type="integer">

      Total number of post likes.

      **Read-only.**

    </ResponseField>

    <ResponseField name="viewCount" type="integer">

      Total number of post views.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date post was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="lastActivityDate" type="datetime">

      Date of latest activity on this post (e.g., date the latest comment was added to post).

      **Read-only.**

    </ResponseField>

    <ResponseField name="url" type="PageUrl">

      Post URL.

      **Read-only.**

      <Expandable title="url properties">

        <ResponseField name="base" type="string">

          The base URL. For premium sites, this is the domain. For free sites, this is the site URL. For example, `mysite.wixsite.com/mysite`.

        </ResponseField>

        <ResponseField name="path" type="string">

          The relative path for the page within the site. For example, `/product-page/a-product`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="slug" type="string">

      Post slug.

      **Read-only.**

    </ResponseField>

    <ResponseField name="bestAnswerCommentId" type="string">

      ID of the best answer comment for this post (selected by forum admin or post owner).
      Deprecated on '2022-08-24' use markedComments instead.

      **Read-only.**

    </ResponseField>

    <ResponseField name="content" type="string">

      The post content as a stringified [Rich Content document](https://dev.wix.com/docs/ricos/api-reference/ricos-document)

      **Read-only.**

    </ResponseField>

    <ResponseField name="images" type="Image[]">

      Images from post content.

      **Read-only.**

      <Expandable title="images properties">

        <ResponseField name="id" type="string">

          WixMedia image ID.

        </ResponseField>

        <ResponseField name="url" type="string">

          Image URL.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image height.

          **Read-only.**

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image width.

          **Read-only.**

        </ResponseField>

        <ResponseField name="altText" type="string">

          Image alt text.

        </ResponseField>

        <ResponseField name="filename" type="string">

          Image filename.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="videos" type="VideoV2[]">

      Videos from post content.

      **Read-only.**

      <Expandable title="videos properties">

        <ResponseField name="id" type="string">

          WixMedia ID.

        </ResponseField>

        <ResponseField name="resolutions" type="VideoResolution[]">

          Available resolutions for the video, starting with the optimal resolution.

          **Read-only.** Constraints: `maxItems 100`.

          <Expandable title="resolutions properties">

            <ResponseField name="url" type="string">

              Video URL.

            </ResponseField>

            <ResponseField name="height" type="integer">

              Video height.

            </ResponseField>

            <ResponseField name="width" type="integer">

              Video width.

            </ResponseField>

            <ResponseField name="format" type="string">

              Video format for example, mp4, hls.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="filename" type="string">

          Video filename.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="pinned" type="boolean">

      Whether the post is pinned. Places the post at the top of the post list.

    </ResponseField>

    <ResponseField name="closed" type="boolean">

      Whether post is closed and commenting is disabled.

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date post was updated.

    </ResponseField>

    <ResponseField name="commentInteraction" type="string">

      Defines what interactions may be applied on the comment under the post, vote or like.

      **Supported values:**

      - `REACTION`
      - `VOTE`
      - `NONE`

    </ResponseField>

    <ResponseField name="markedComments" type="string[]">

      IDs of the marked comments for this post. Posts are marked by the forum admin or the post owner.

      **Read-only.** Constraints: `maxItems 10`.

    </ResponseField>

    <ResponseField name="recentActivity" type="RecentActivity">

      Recent activity of post. For example, a comment added to a post.

      **Read-only.**

      <Expandable title="recentActivity properties">

        <ResponseField name="entityId" type="string">

          ID of recent activity entity.

        </ResponseField>

        <ResponseField name="type" type="string">

          Type of entity.

          **Supported values:**

          - `COMMENT`
          - `REPLY`

        </ResponseField>

        <ResponseField name="activityDate" type="string">

          Date activity was created.

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="initiator" type="initiator">

          initiator

          <Expandable title="initiator properties">

            <ResponseField name="memberId" type="string">

              Site member ID.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="reactionIdentities" type="ReactionIdentity[]">

      List of reaction identities grouped by reaction code.

      **Read-only.** Constraints: `maxItems 6`.

      <Expandable title="reactionIdentities properties">

        <ResponseField name="reactionCode" type="string">

          Reaction code.

        </ResponseField>

        <ResponseField name="identities" type="Identity[]">

          Reacted identities.

          **Read-only.** Constraints: `maxItems 500`.

          <Expandable title="identities properties">

            <ResponseField name="identityId" type="string">

              Identity id.

            </ResponseField>

            <ResponseField name="identityType" type="string">

              Reacted identity type.

              **Supported values:**

              - `VISITOR`
              - `MEMBER`

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="count" type="integer">

          Total number of reacted identities count.

        </ResponseField>

      </Expandable>

    </ResponseField>
