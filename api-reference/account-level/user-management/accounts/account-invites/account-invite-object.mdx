This service manages all invites related actions (site and account level)

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Invite ID.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="accountId" type="string">

      Account ID.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="email" type="string">

      Email address where the invite was sent.

      Constraints: `format EMAIL`.

    </ResponseField>

    <ResponseField name="role" type="string" deprecated>

      Deprecated. Use `policyIds`.

      **Deprecated.**

    </ResponseField>

    <ResponseField name="inviterId" type="string" deprecated>

      Deprecated. Use `inviterAccountId`.

      **Read-only.** **Deprecated.**

    </ResponseField>

    <ResponseField name="status" type="string">

      Invite status.

      Supported values:
      - **Pending:** The invite has been sent and is valid, waiting for the user's response.
      - **Used:** The invite has been accepted.
      - **Deleted:** The invite has been deleted or revoked.
      - **Declined:** The user has declined the invite.
      - **Expired:** The invite has expired without being accepted.

      **Supported values:**

      - `Pending`
      - `Used`
      - `Deleted`
      - `Declined`
      - `Expired`

    </ResponseField>

    <ResponseField name="acceptLink" type="string">

      Link to accept the invite.

    </ResponseField>

    <ResponseField name="inviterAccountId" type="string">

      Inviting account ID.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="acceptedByAccountId" type="string">

      Account ID that accepted the invite. Populated only once the invite is accepted.

      **Read-only.**

    </ResponseField>

    <ResponseField name="dateCreated" type="datetime">

      Date the invite was created.

    </ResponseField>

    <ResponseField name="policyIds" type="string[]">

      Role IDs included in the invite.

    </ResponseField>

    <ResponseField name="dateUpdated" type="datetime">

      Date the invite was last updated.

    </ResponseField>

    <ResponseField name="assignments" type="InviteResourceAssignment[]">

      Assets the users are invited to join.

      <Expandable title="assignments properties">

        <ResponseField name="policyId" type="string">

          Role ID.

        </ResponseField>

        <ResponseField name="assignments" type="InviteAssignment[]">

          Resources the user will be able to access.

          <Expandable title="assignments properties">

            <ResponseField name="fullNameResource" type="FullNameResource">

              Full name of resource to be assigned.

              <Expandable title="One of the following">

                <ResponseField name="ResourceContext" type="ResourceContext">

                  ResourceContext

                  _Nested structure omitted for brevity — see the `ResourceContext` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="expirationDate" type="datetime">

      Invite expiration date.

    </ResponseField>
