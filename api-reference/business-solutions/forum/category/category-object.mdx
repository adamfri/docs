<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Category ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="parentId" type="string">

      Category parent ID.

    </ResponseField>

    <ResponseField name="name" type="string">

      Category name.

    </ResponseField>

    <ResponseField name="headerTitle" type="string">

      Category header title.

    </ResponseField>

    <ResponseField name="headerType" type="string">

      Category header type.

      **Supported values:**

      - `NONE`
      - `COLOR`
      - `IMAGE`

    </ResponseField>

    <ResponseField name="headerImage" type="Image">

      Category header image.

      <Expandable title="headerImage properties">

        <ResponseField name="id" type="string">

          WixMedia image ID.

        </ResponseField>

        <ResponseField name="url" type="string">

          Image URL.

        </ResponseField>

        <ResponseField name="height" type="integer">

          Original image height.

          **Read-only.**

        </ResponseField>

        <ResponseField name="width" type="integer">

          Original image width.

          **Read-only.**

        </ResponseField>

        <ResponseField name="altText" type="string">

          Image alt text.

        </ResponseField>

        <ResponseField name="filename" type="string">

          Image filename.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="rank" type="integer">

      Category rank (display order).

      **Read-only.**

    </ResponseField>

    <ResponseField name="postCount" type="integer">

      Total number of posts in category.

      **Read-only.**

    </ResponseField>

    <ResponseField name="postViewCount" type="integer">

      Total number of post views in category.

      **Read-only.**

    </ResponseField>

    <ResponseField name="slug" type="string">

      Category slug.

      **Read-only.**

    </ResponseField>

    <ResponseField name="url" type="PageUrl">

      Category URL.

      **Read-only.**

      <Expandable title="url properties">

        <ResponseField name="base" type="string">

          The base URL. For premium sites, this is the domain. For free sites, this is the site URL. For example, `mysite.wixsite.com/mysite`.

        </ResponseField>

        <ResponseField name="path" type="string">

          The relative path for the page within the site. For example, `/product-page/a-product`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="description" type="string">

      Category description.

      Constraints: `maxLength 140`.

    </ResponseField>

    <ResponseField name="postTypes" type="string[]" deprecated>

      List of post type that Category supports.
      Deprecated on '2021-08-03'.

      **Deprecated.**

      **Supported values:**

      - `DISCUSSION`
      - `QUESTION`

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date category was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="categoryAccess" type="string">

      Whether all site visitors and members (`PUBLIC`), all members (`MEMBERS_ONLY`), or only specific members and groups of members with roles, badges or paid subscriptions (`PRIVATE`) can access posts in category.

      **Read-only.**

      **Supported values:**

      - `PUBLIC`
      - `MEMBERS_ONLY`
      - `PRIVATE`

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date category was updated.

    </ResponseField>

    <ResponseField name="noMemberPosting" type="boolean">

      Whether only admins and moderators can write posts in category.

    </ResponseField>

    <ResponseField name="groups" type="string[]">

      IDs of groups which can access this category if `categoryAccess` is `PRIVATE`.

      **Read-only.**

    </ResponseField>

    <ResponseField name="commentInteraction" type="string">

      Defines what interactions may be applied on the comment under the posts created in this category, vote or like.

      **Supported values:**

      - `REACTION`
      - `VOTE`
      - `NONE`

    </ResponseField>

    <ResponseField name="mainCommentReaction" type="Reaction">

      Defines which icon will be used for the main reaction on the comment under the posts created in this category.

      Default `type`: `"LIKE"`.

      <Expandable title="mainCommentReaction properties">

        <ResponseField name="code" type="string">

          **Read-only.**

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="reaction" type="reaction">

          reaction

          <Expandable title="reaction properties">

            <ResponseField name="type" type="string">

              **Supported values:**

              - `LIKE`
              - `ANGRY`
              - `CLAP`
              - `LOVE`
              - `HAHA`
              - `SAD`
              - `SMILE`
              - `WOW`
              - `THINKING`
              - `THUMBS_UP`
              - `THUMBS_DOWN`

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="additionalCommentReactions" type="Reaction[]">

      Defines what additional reactions may be applied on the comment under the posts created in this category.

      Constraints: `maxItems 5`.

      <Expandable title="additionalCommentReactions properties">

        <ResponseField name="code" type="string">

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="postInteraction" type="string">

      Defines what interactions may be applied on the posts created in this category, like or multiple reactions.

      **Supported values:**

      - `REACTION`
      - `NONE`

    </ResponseField>

    <ResponseField name="mainPostReaction" type="Reaction">

      Defines which icon will be used for main reaction on the posts created in this category.

      Default `type`: `"LIKE"`.

      <Expandable title="mainPostReaction properties">

        <ResponseField name="code" type="string">

          **Read-only.**

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="reaction" type="reaction">

          reaction

          <Expandable title="reaction properties">

            <ResponseField name="type" type="string">

              **Supported values:**

              - `LIKE`
              - `ANGRY`
              - `CLAP`
              - `LOVE`
              - `HAHA`
              - `SAD`
              - `SMILE`
              - `WOW`
              - `THINKING`
              - `THUMBS_UP`
              - `THUMBS_DOWN`

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="additionalPostReactions" type="Reaction[]">

      Defines what additional reactions may be applied on the posts created in this category.

      Constraints: `maxItems 5`.

      <Expandable title="additionalPostReactions properties">

        <ResponseField name="code" type="string">

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>
