A Receipt preset is a set of customizable receipt configurations that control the information, design, and branding of receipts to align with a site's business needs.

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Receipt preset ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the receipt preset is updated.
      To prevent conflicting changes, the current revision must be passed when updating the receipt preset.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the receipt preset was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the receipt preset was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Receipt preset name.

      Constraints: `maxLength 100`.

    </ResponseField>

    <ResponseField name="displaySettings" type="DisplaySettings">

      Display settings.

      <Expandable title="displaySettings properties">

        <ResponseField name="businessDisplayOptions" type="BusinessDisplayOptions">

          Business display options.

          <Expandable title="businessDisplayOptions properties">

            <ResponseField name="showBusinessEmail" type="boolean">

              Whether to display the business email in the receipt.

            </ResponseField>

            <ResponseField name="showBusinessPhoneNumber" type="boolean">

              Whether to display the business phone number in the receipt.

            </ResponseField>

            <ResponseField name="showBusinessAddress" type="boolean">

              Whether to display the business address in the receipt.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="customerDisplayOptions" type="CustomerDisplayOptions">

          Customer display options.

          <Expandable title="customerDisplayOptions properties">

            <ResponseField name="showCustomerEmail" type="boolean">

              Whether to display the customer's email in the receipt.

            </ResponseField>

            <ResponseField name="showCustomerPhoneNumber" type="boolean">

              Whether to display the customer's phone number in the receipt.

            </ResponseField>

            <ResponseField name="showCustomerShippingAddress" type="boolean">

              Whether to display the customer's shipping address in the receipt.

            </ResponseField>

            <ResponseField name="showCustomerBillingAddress" type="boolean">

              Whether to display the customer's billing address in the receipt.

            </ResponseField>

            <ResponseField name="showCustomerVatId" type="boolean">

              Whether to display the customer's VAT ID in the receipt.

            </ResponseField>

            <ResponseField name="showCustomerCompanyName" type="boolean">

              Whether to display the customer's company's name.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="itemsDisplayOptions" type="ItemsDisplayOption">

          Item display options.

          <Expandable title="itemsDisplayOptions properties">

            <ResponseField name="showItemsDescription" type="boolean">

              Whether to display item descriptions in the receipt.

            </ResponseField>

            <ResponseField name="showTax" type="boolean">

              Whether to display item taxes in the receipt.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="totalsDisplayOptions" type="TotalsDisplayOption">

          Totals display options.

          <Expandable title="totalsDisplayOptions properties">

            <ResponseField name="showTaxBreakdown" type="boolean">

              Whether to display the tax breakdown in the receipt.

            </ResponseField>

            <ResponseField name="showItemSubtotalPerTaxBreakdown" type="boolean">

              Whether to display the item subtotal per tax breakdown in the receipt.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="customFields" type="CustomFields">

      Custom fields.
      Values appear on the receipt in the specified order.

      <Expandable title="customFields properties">

        <ResponseField name="headerCustomFields" type="CustomField[]">

          Header custom fields.

          Constraints: `maxItems 4`.

          <Expandable title="headerCustomFields properties">

            <ResponseField name="title" type="string">

              Title.

            </ResponseField>

            <ResponseField name="value" type="string">

              Value.

              Constraints: `maxLength 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="businessCustomFields" type="CustomField[]">

          Business custom fields.

          Constraints: `maxItems 4`.

          <Expandable title="businessCustomFields properties">

            <ResponseField name="title" type="string">

              Title.

            </ResponseField>

            <ResponseField name="value" type="string">

              Value.

              Constraints: `maxLength 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="customerCustomFields" type="CustomField[]">

          Customer custom fields.

          Constraints: `maxItems 4`.

          <Expandable title="customerCustomFields properties">

            <ResponseField name="title" type="string">

              Title.

            </ResponseField>

            <ResponseField name="value" type="string">

              Value.

              Constraints: `maxLength 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="footerCustomFields" type="FooterCustomField[]">

          Footer custom fields.

          Constraints: `maxItems 4`.

          <Expandable title="footerCustomFields properties">

            <ResponseField name="title" type="string">

              Title.

              Constraints: `minLength 1`, `maxLength 200`.

            </ResponseField>

            <ResponseField name="value" type="string">

              Value.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="default" type="boolean">

      Whether this is the default receipt preset.
      The default preset is used during receipt creation when a preset isn't specified or doesn't exist.

      **Read-only.**

    </ResponseField>

    <ResponseField name="displayValues" type="DisplayValues">

      Display values.

      <Expandable title="displayValues properties">

        <ResponseField name="titleOverride" type="string">

          Text override for receipt title.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Custom field data for the receipt preset.

      Extended fields must be configured in the app dashboard before they can be accessed with API calls.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>
