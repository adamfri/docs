title: "Function Type Object"
description: "A function type defines the integration point where your function's custom logic runs."

A function type defines the integration point where your function's custom logic runs.

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Function type ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="function" type="Function">

      Settings for functions created from this function type.

      <Expandable title="function properties">

        <ResponseField name="definitions" type="Definition[]">

          Function method definitions that specify when and how the function executes.

          Constraints: `maxItems 100`.

          <Expandable title="definitions properties">

            <ResponseField name="method" type="string">

              Method name.

              Constraints: `maxLength 255`.

            </ResponseField>

            <ResponseField name="input" type="Input">

              Function input schema and UI definition.

              <Expandable title="input properties">

                <ResponseField name="schema" type="object">

                  JSON schema that defines the function input.

                </ResponseField>

                <ResponseField name="ui" type="object">

                  Defines how the function input is displayed in a site dashboard's function builder.

                </ResponseField>

                <ResponseField name="displayDescription" type="string">

                  Input description displayed in a site dashboard's function builder.

                </ResponseField>

                <ResponseField name="builderServicePlugin" type="BuilderServicePlugin">

                  Service plugin configuration for dynamic input schema fetching.

                  _Nested structure omitted for brevity — see the `BuilderServicePlugin` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="output" type="Output">

              Function output schema and UI definition.

              <Expandable title="output properties">

                <ResponseField name="schema" type="object">

                  JSON schema that defines the function output.

                </ResponseField>

                <ResponseField name="ui" type="object">

                  Defines how the function output is displayed in a site dashboard's function builder.

                </ResponseField>

                <ResponseField name="defined" type="State">

                  Message for when the logic returns a result.

                  _Nested structure omitted for brevity — see the `State` object._

                </ResponseField>

                <ResponseField name="empty" type="State">

                  Message for when the logic returns no result.

                  _Nested structure omitted for brevity — see the `State` object._

                </ResponseField>

                <ResponseField name="staticContent" type="StaticContent">

                  Static content for the output

                  _Nested structure omitted for brevity — see the `StaticContent` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="triggerId" type="string">

              ID of the [trigger](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/about-triggers) that activates this function definition.

              **Read-only.**

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="configuration" type="Configuration">

          Function configuration including service plugin settings and behavior options.

          <Expandable title="configuration properties">

            <ResponseField name="multipleInstancesAllowed" type="boolean">

              Whether the function can be implemented multiple times on a single site.

            </ResponseField>

            <ResponseField name="customInitialPreset" type="CustomInitialPreset">

              Initial preset to apply when building a function from scratch.

              <Expandable title="customInitialPreset properties">

                <ResponseField name="functionTemplateExtensionId" type="string">

                  ID of the template to apply.

                </ResponseField>

                <ResponseField name="enabled" type="boolean">

                  Whether to apply the template to functions created from scratch.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="scope" type="string">

              Scope of the function when multiple_instances_allowed is false

              **Supported values:**

              - `SITE` — One function instance is allowed per site
              - `ENTITY` — One function instance is allowed per entity

            </ResponseField>

            <ResponseField name="entityScopeErrorMessage" type="string">

              Error message to display when entity scope validation fails.

            </ResponseField>

            <ResponseField name="servicePlugin" type="ServicePlugin">

              Service plugin configuration for function integration with business solutions.

              <Expandable title="servicePlugin properties">

                <ResponseField name="fqdn" type="string">

                  Service plugin fqdn.

                </ResponseField>

                <ResponseField name="schema" type="object">

                  Service plugin configuration schema.

                </ResponseField>

                <ResponseField name="ui" type="object">

                  How the service plugin is displayed in a site dashboard's function builder.

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="description" type="Description">

          Function descriptions, such as text to display in a site dashboard's function builder.

          <Expandable title="description properties">

            <ResponseField name="aboutDescription" type="string">

              Explains what the function does.

            </ResponseField>

            <ResponseField name="howItWorksDescription" type="string">

              Explains how the function works.

            </ResponseField>

            <ResponseField name="outcomeDescription" type="string">

              Expected outcome when the function runs.

            </ResponseField>

            <ResponseField name="representativeImage" type="Image">

              Function type's image displayed in the function catalog and builder.

              <Expandable title="representativeImage properties">

                <ResponseField name="id" type="string">

                  WixMedia image ID.

                </ResponseField>

                <ResponseField name="url" type="string">

                  Image URL.

                  Constraints: `maxLength 10000`.

                </ResponseField>

                <ResponseField name="height" type="integer">

                  Original image height.

                  **Read-only.**

                </ResponseField>

                <ResponseField name="width" type="integer">

                  Original image width.

                  **Read-only.**

                </ResponseField>

                <ResponseField name="altText" type="string">

                  Image alt text.

                </ResponseField>

                <ResponseField name="filename" type="string">

                  Image filename.

                  **Read-only.**

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="builderlessTemplateSelectionModalTitle" type="string">

              Title of the builderless function template selection modal.

            </ResponseField>

            <ResponseField name="builderlessTemplateSelectionModalSubtitle" type="string">

              Subtitle of the builderless function template selection modal.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="appDefId" type="string">

      ID of the Wix app that defines this function type.

    </ResponseField>

    <ResponseField name="appName" type="string">

      Display name of the Wix app that defines this function type.

      **Read-only.**

    </ResponseField>

    <ResponseField name="functionExtensionName" type="string">

      Display name of the function type.

    </ResponseField>
