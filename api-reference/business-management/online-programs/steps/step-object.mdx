title: "Step Object"

---

A step represents a piece of program content within a section.
A step can contain article, video, or quiz content.

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Step ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the step is updated. To prevent conflicting changes, the current `revision` must be passed when updating the step.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the step was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the step was updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="sectionId" type="string">

      Parent section ID. To move a step within a section or to another section, call `MoveStep`.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="description" type="Description">

      Step title, rich text content, and optional media.

      <Expandable title="description properties">

        <ResponseField name="title" type="string">

          Constraints: `maxLength 250`.

        </ResponseField>

        <ResponseField name="details" type="string" />

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="media" type="media">

          media

          <Expandable title="media properties">

            <ResponseField name="image" type="Image">

              Image displayed with the step content.

              <Expandable title="image properties">

                <ResponseField name="id" type="string">

                  WixMedia image ID.

                </ResponseField>

                <ResponseField name="url" type="string">

                  Image URL.

                </ResponseField>

                <ResponseField name="height" type="integer">

                  Original image height.

                  **Read-only.**

                </ResponseField>

                <ResponseField name="width" type="integer">

                  Original image width.

                  **Read-only.**

                </ResponseField>

                <ResponseField name="altText" type="string">

                  Image alt text.

                </ResponseField>

                <ResponseField name="filename" type="string">

                  Image filename.

                  **Read-only.**

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="video" type="VideoV2">

              Video displayed with the step content.

              <Expandable title="video properties">

                <ResponseField name="id" type="string">

                  WixMedia ID.

                </ResponseField>

                <ResponseField name="resolutions" type="VideoResolution[]">

                  Available resolutions for the video, starting with the optimal resolution.

                  **Read-only.** Constraints: `maxItems 100`.

                  _Nested structure omitted for brevity — see the `VideoResolution[]` object._

                </ResponseField>

                <ResponseField name="filename" type="string">

                  Video filename.

                  **Read-only.**

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="ordering" type="number">

      Step position within the section.

    </ResponseField>

    <ResponseField name="stepType" type="string">

      Type of content in the step.

      **Supported values:**

      - `ARTICLE` — Text-based step.
      - `VIDEO` — Step with video content.
      - `QUIZ` — Step linked to a quiz.

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Custom field data for the step object.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="programId" type="string">

      Program ID.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>
