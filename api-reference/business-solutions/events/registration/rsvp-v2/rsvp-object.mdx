---
title: "Rsvp Object"
description: "The Rsvp object in the Wix Rsvp API."
icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      ID of the RSVP.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the RSVP is updated. To prevent conflicting changes, the existing `revision` must be used when updating the RSVP.

      **Read-only.**

    </ResponseField>

    <ResponseField name="eventId" type="string">

      Event ID to which the RSVP belongs.

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the RSVP was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the RSVP was updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="memberId" type="string">

      Site member ID.

    </ResponseField>

    <ResponseField name="contactId" type="string">

      Contact ID of a guest who filled in the RSVP form. See the Contacts API for more details.

    </ResponseField>

    <ResponseField name="firstName" type="string">

      First name of a guest who filled in the RSVP form.

    </ResponseField>

    <ResponseField name="lastName" type="string">

      Last name of a guest who filled in the RSVP form.

    </ResponseField>

    <ResponseField name="email" type="string">

      Email of a guest who filled in the RSVP form.

    </ResponseField>

    <ResponseField name="form" type="FormResponse">

      Event registration form response. Each item in `inputValues` must use a defined input name and follow the applicable mandatory-field and predefined-option rules.
      Call [Get Event](https://dev.wix.com/docs/rest/business-solutions/events/events-v3/get-event) with `fields: ["FORM"]` to inspect the expected input names, mandatory fields, and predefined options.

      <Expandable title="form properties">

        <ResponseField name="inputValues" type="InputValue[]">

          Form field inputs.

          Constraints: `maxItems 200`.

          <Expandable title="inputValues properties">

            <ResponseField name="inputName" type="string">

              Form field input name.

              Constraints: `maxLength 100`.

            </ResponseField>

            <ResponseField name="value" type="string">

              Form field value.

              Constraints: `maxLength 5000`.

            </ResponseField>

            <ResponseField name="values" type="string[]">

              Form field values.

              Constraints: `maxLength 5000`, `maxItems 100`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="status" type="string">

      RSVP response status.

      **Supported values:**

      - `YES` — Guest is attending the event.
      - `NO` — Guest isn't attending the event.
      - `WAITLIST` — Guest is on a waitlist.

    </ResponseField>

    <ResponseField name="totalGuests" type="integer">

      Total number of guests per 1 RSVP.

      **Read-only.**

    </ResponseField>

    <ResponseField name="guestList" type="GuestList">

      Guest list.

      **Read-only.**

      <Expandable title="guestList properties">

        <ResponseField name="guests" type="Guest[]">

          List of guests.

          Constraints: `maxItems 11`.

          <Expandable title="guests properties">

            <ResponseField name="index" type="integer">

              Index of each guest in the array.

            </ResponseField>

            <ResponseField name="fullName" type="string">

              Full name.

            </ResponseField>

            <ResponseField name="checkInDetails" type="CheckInDetails">

              Check-in details.

              <Expandable title="checkInDetails properties">

                <ResponseField name="checkedIn" type="boolean">

                  Whether the guest has checked in.

                </ResponseField>

                <ResponseField name="checkInDate" type="datetime">

                  Date and time the guest checked in.

                </ResponseField>

                <ResponseField name="checkInCanceledDate" type="datetime">

                  Date and time the guest cancelled check-in.

                </ResponseField>

                <ResponseField name="checkedInBy" type="string">

                  Id of the user who last performed check-in

                </ResponseField>

                <ResponseField name="checkInCanceledBy" type="string">

                  Id of the user who last canceled check-in

                </ResponseField>

                <ResponseField name="checkedInByStaffMember" type="string">

                  Id of the Staff Member who last performed check-in

                </ResponseField>

                <ResponseField name="checkInCanceledByStaffMember" type="string">

                  Id of theStaff Member who last canceled check-in

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="guestId" type="integer">

              Guest ID.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="language" type="string">

      2-letter language code in the [ISO 639-1](https://en.wikipedia.org/wiki/List_of_ISO_639-1_codes) format.

      **Read-only.**

    </ResponseField>

    <ResponseField name="locale" type="string">

      Locale in the [IETF BCP 47](https://en.wikipedia.org/wiki/IETF_language_tag) format.

      **Read-only.**

    </ResponseField>

    <ResponseField name="fullyCheckedIn" type="boolean">

      Whether all guests are checked in.

      **Read-only.**

    </ResponseField>

    <ResponseField name="checkedInGuestCount" type="integer">

      Number of checked in guests.

      **Read-only.**

    </ResponseField>

    <ResponseField name="additionalGuestDetails" type="AdditionalGuestDetails">

      Additional guest details.

      Available only when the event's form includes a guest control (`GUEST_CONTROL` type). If the event doesn't support additional guests, omit this field.

      <Expandable title="additionalGuestDetails properties">

        <ResponseField name="guestCount" type="integer">

          Number of additional guests.

        </ResponseField>

        <ResponseField name="guestNames" type="string[]">

          Names of additional guests.

          Constraints: `maxLength 101`, `maxItems 10`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="disableNotifications" type="boolean">

      Whether notifications are disabled.

    </ResponseField>

    <ResponseField name="onlineConferencingLogin" type="OnlineConferencingLogin">

      Guest login details for accessing the online conference event.

      **Read-only.**

      <Expandable title="onlineConferencingLogin properties">

        <ResponseField name="link" type="string">

          Link URL to the online conference.

          **Read-only.** Constraints: `format WEB_URL`.

        </ResponseField>

        <ResponseField name="password" type="string">

          Password for the online conference.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="anonymized" type="boolean">

      Whether an RSVP is anonymized.

      **Read-only.**

    </ResponseField>

    <ResponseField name="marketingConsent" type="boolean">

      Whether a guest agreed to marketing.

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Custom field data for the RSVP object.

      **Note:** You must configure extended fields using schema plugin extensions in your app's dashboard before you can access the extended fields with API calls.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>
