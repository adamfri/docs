<Tabs>

  <Tab title="REST">

    <ResponseField name="ticketNumber" type="string">

      Unique ticket number which is assigned automatically when creating a ticket.

    </ResponseField>

    <ResponseField name="orderNumber" type="string">

      Associated order number.

    </ResponseField>

    <ResponseField name="ticketDefinitionId" type="string">

      Ticket definition ID. See the Ticket Definition API for additional information.

      Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="name" type="string">

      Ticket type.

    </ResponseField>

    <ResponseField name="price" type="Money">

      Ticket price.

      <Expandable title="price properties">

        <ResponseField name="amount" type="string" deprecated>

          *Deprecated:** Use `value` instead.

          **Deprecated.** Constraints: `format DECIMAL_VALUE`.

        </ResponseField>

        <ResponseField name="currency" type="string">

          3-letter currency code in [ISO-4217 alphabetic](https://en.wikipedia.org/wiki/ISO_4217#Active_codes) format. For example, `USD`.

          Constraints: `format CURRENCY`.

        </ResponseField>

        <ResponseField name="value" type="string">

          Monetary amount. Decimal string with a period as a decimal separator (e.g., 3.99). Optionally, starts with a single (-), to indicate that the amount is negative.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="free" type="boolean">

      Whether the ticket requires payment.

      **Read-only.**

    </ResponseField>

    <ResponseField name="policy" type="string">

      Ticket and event policies.

    </ResponseField>

    <ResponseField name="checkIn" type="CheckIn">

      Ticket check-in.

      <Expandable title="checkIn properties">

        <ResponseField name="created" type="datetime">

          Time of a ticket's check-in.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="orderStatus" type="string">

      Associated order status.

      **Supported values:**

      - `NA_ORDER_STATUS` — Order status isn't available for this request fieldset.
      - `FREE` — Order is confirmed, no payment is required.
      - `PENDING` — Order was paid, but the payment gateway suspended the payment.
      - `PAID` — Order is paid.
      - `OFFLINE_PENDING` — Order is confirmed but must be paid via offline payment. Status needs to be manually updated to `"PAID"`.
      - `INITIATED` — Order is waiting for payment at the cashier.
      - `CANCELED` — Order is canceled.
      - `DECLINED` — Order payment is declined.
      - `AUTHORIZED` — Order payment is authorized.
      - `VOIDED` — Order payment is voided.
      - `PARTIALLY_PAID` — Order is partially paid with less than the total amount.

    </ResponseField>

    <ResponseField name="orderArchived" type="boolean">

      Whether the order and ticket are archived.
      If set to `true`, they aren't visible in the order list.

    </ResponseField>

    <ResponseField name="orderFullName" type="string">

      Buyer full name.

    </ResponseField>

    <ResponseField name="guestFullName" type="string">

      Guest full name.

    </ResponseField>

    <ResponseField name="guestDetails" type="GuestDetails">

      Guest personal details.

      <Expandable title="guestDetails properties">

        <ResponseField name="guestAssigned" type="boolean">

          Whether ticket belongs to assigned guest.

        </ResponseField>

        <ResponseField name="firstName" type="string">

          Guest first name.

        </ResponseField>

        <ResponseField name="lastName" type="string">

          Guest last name.

        </ResponseField>

        <ResponseField name="email" type="string">

          Guest email.

        </ResponseField>

        <ResponseField name="form" type="FormResponse">

          Full form response.

          <Expandable title="form properties">

            <ResponseField name="inputValues" type="InputValue[]">

              Form field inputs.

              Constraints: `maxItems 200`.

              <Expandable title="inputValues properties">

                <ResponseField name="inputName" type="string">

                  Form field input name.

                  Constraints: `maxLength 100`.

                </ResponseField>

                <ResponseField name="value" type="string">

                  Form field value.

                  Constraints: `maxLength 5000`.

                </ResponseField>

                <ResponseField name="values" type="string[]">

                  Form field values.

                  Constraints: `maxLength 5000`, `maxItems 100`.

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="contactId" type="string">

          Contact ID associated with this guest.

        </ResponseField>

        <ResponseField name="phone" type="string">

          Guest phone number.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="archived" type="boolean">

      Whether the ticket is visible in an order.

    </ResponseField>

    <ResponseField name="memberId" type="string">

      Ticket owner member ID.

    </ResponseField>

    <ResponseField name="anonymized" type="boolean">

      Whether ticket was anonymized by GDPR delete.
      Anonymized tickets no longer contain personally identifiable information (PII).

    </ResponseField>

    <ResponseField name="checkInUrl" type="string">

      Ticket check-in URL.

      Shown as QR code image in PDF.  &lt;br/>
      Format: `"https://www.wixevents.com/check-in/{ticket number},{event id}"`  &lt;br/>
      Example: `"https://www.wixevents.com/check-in/AAAA-AAAA-BB021,00000000-0000-0000-0000-000000000000"`

    </ResponseField>

    <ResponseField name="ticketPdfUrl" type="string">

      URL for ticket PDF download.

    </ResponseField>

    <ResponseField name="channel" type="string">

      Associated order checkout channel type.

      **Supported values:**

      - `ONLINE` — Buyer created the order via an online channel, such as a website or mobile app.
      - `OFFLINE_POS` — Sales person created the order and collected the money.

    </ResponseField>

    <ResponseField name="walletPassUrl" type="string">

      URL to download a ticket in `.pkpass` format for Apple Wallet.

      **Read-only.** Constraints: `format WEB_URL`.

    </ResponseField>

    <ResponseField name="canceled" type="boolean">

      Whether ticket is canceled.

      **Read-only.**

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "ticket": {
          "ticketNumber": "FNVL-NIJT-WP021",
          "orderNumber": "FNVL-NIJT-WP",
          "ticketDefinitionId": "d86ffbbd-aa9b-488d-a14b-11752b78e272",
          "name": "Early Bird",
          "price": {
            "amount": "12.34",
            "currency": "USD"
          },
          "free": false,
          "policy": "This is your event ticket. Ticket holders must present their tickets on entry.",
          "orderStatus": "PAID",
          "orderArchived": false,
          "orderFullName": "John Doe",
          "guestDetails": {
            "guestAssigned": false,
            "firstName": "John",
            "lastName": "Doe",
            "email": "john.doe@example.com",
            "contactId": "3ef81e9f-106e-408d-a3ee-3df60838acc6"
          },
          "archived": false,
          "anonymized": false,
          "checkInUrl": "https://www.wixevents.com/check-in/FNVL-NIJT-WP021,ad18d12e-a6a9-4c17-abfa-6ff119479be2",
          "ticketPdfUrl": "https://apps.wix.com/events/doc/tickets/Ticket.pdf?request=<TOKEN>"
        }
      }
      ```

    </CodeGroup>
