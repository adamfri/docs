    Retrieves a list of items given the provided paging, filtering, and sorting. Up to 500 items can be returned per request.

    To learn how to query items, see [API Query Language](https://dev.wix.com/docs/rest/articles/getting-started/api-query-language).

    | Property | Filters | Sort |
    | --- | --- | --- |
    | `id` | `$eq`, `$ne`, `$in`, `$exists`, `$gt`, `$gte`, `$lt`, `$lte`, `$startsWith` | Sort: `ASC`, `DESC` |
    | `image.id` | `$eq`, `$ne`, `$in`, `$exists`, `$gt`, `$gte`, `$lt`, `$lte`, `$startsWith` | Sort: `ASC`, `DESC` |
    | `businessLocationIds` | `$hasSome`, `$hasAll`, `$exists`, `$isEmpty` | Sort: `ASC`, `DESC` |
    | `name` | `$eq`, `$ne`, `$in`, `$exists`, `$gt`, `$gte`, `$lt`, `$lte`, `$startsWith` | Sort: `ASC`, `DESC` |
    | `description` | `$eq`, `$ne`, `$in`, `$exists`, `$gt`, `$gte`, `$lt`, `$lte`, `$startsWith` | Sort: `ASC`, `DESC` |
    | `visible` | `$eq`, `$ne`, `$in`, `$exists` | Sort: `ASC`, `DESC` |
    | `featured` | `$eq`, `$ne`, `$in`, `$exists` | Sort: `ASC`, `DESC` |
    | `createdDate` | `$eq`, `$ne`, `$in`, `$exists`, `$gt`, `$gte`, `$lt`, `$lte` | Sort: `ASC`, `DESC` |
    | `updatedDate` | `$eq`, `$ne`, `$in`, `$exists`, `$gt`, `$gte`, `$lt`, `$lte` | Sort: `ASC`, `DESC` |
    | `priceVariants.variants.price` | `$in`, `$nin`, `$eq`, `$ne`, `$exists` | Sort: `ASC`, `DESC` |
    | `orderSettings.inStock` | `$in`, `$nin`, `$eq`, `$ne`, `$exists` | Sort: `ASC`, `DESC` |
    | `orderSettings.acceptSpecialRequests` | `$in`, `$nin`, `$eq`, `$ne`, `$exists` | Sort: `ASC`, `DESC` |
    | `featuredInfo.type` | `$in`, `$nin`, `$eq`, `$ne`, `$exists` | Sort: `ASC`, `DESC` |
    | `labels.id` | `$eq`, `$ne`, `$in`, `$exists`, `$gt`, `$gte`, `$lt`, `$lte`, `$startsWith` | None |

    ## Body Params

    <ParamField body="query" type="CursorQuery">

      Query options.

      <Expandable title="query properties">

        <ParamField body="filter" type="object">

          Filter object in the following format:
          `"filter" : {`
          "fieldName1": "value1",
          "fieldName2":&#123;"$operator":"value2"&#125;
          &#125;``
          Example of operators: `$eq`, `$ne`, `$lt`, `$lte`, `$gt`, `$gte`, `$in`, `$hasSome`, `$hasAll`, `$startsWith`, `$contains`

        </ParamField>

        <ParamField body="sort" type="Sorting[]">

          Sort object in the following format:
          `[{"fieldName":"sortField1","order":"ASC"},{"fieldName":"sortField2","order":"DESC"}]`

          <Expandable title="sort properties">

            <ParamField body="fieldName" type="string">

              Name of the field to sort by.

              Constraints: `maxLength 512`.

            </ParamField>

            <ParamField body="order" type="string">

              Sort order.

              **Supported values:**

              - `ASC`
              - `DESC`

            </ParamField>

            <ParamField body="origin" type="AddressLocation">

              Origin point for geo-distance sorting on a GEO field
              results are ordered by distance from this point (ASC = nearest first, DESC = farthest first).

              _Nested structure omitted for brevity — see the `AddressLocation` object._

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

      <Expandable title="One of the following">

        <ParamField body="pagingMethod" type="pagingMethod">

          pagingMethod

          <Expandable title="pagingMethod properties">

            <ParamField body="cursorPaging" type="CursorPaging">

              Cursor token pointing to a page of results. Not used in the first request. Following requests use the cursor token and not `filter` or `sort`.

              _Nested structure omitted for brevity — see the `CursorPaging` object._

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>
