Metadata for an app extension type, including catalog-managed settings and business schema data.

<Tabs>

  <Tab title="REST">

    <ResponseField name="type" type="string">

      Component type.

      Constraints: `maxLength 128`.

    </ResponseField>

    <ResponseField name="devCenterComponentTypeData" type="DevCenterComponentTypeData">

      Metadata defined in the components catalog.

      <Expandable title="devCenterComponentTypeData properties">

        <ResponseField name="displayName" type="string">

          Component type display name.

          Constraints: `maxLength 64`.

        </ResponseField>

        <ResponseField name="requiresApproval" type="boolean">

          Whether components of this type require approval before release.

        </ResponseField>

        <ResponseField name="labels" type="string[]">

          Labels shown for components of this type.

          Constraints: `maxLength 64`, `maxItems 32`.

        </ResponseField>

        <ResponseField name="attributes" type="Attributes">

          Rules enforced for components of this type, such as singular, non-updatable, and non-deletable behavior.

          <Expandable title="attributes properties">

            <ResponseField name="singular" type="boolean">

              Whether only one component of this type can exist per app.

            </ResponseField>

            <ResponseField name="nonDeletable" type="boolean">

              Whether components of this type can't be deleted.

            </ResponseField>

            <ResponseField name="nonUpdatable" type="boolean">

              Whether components of this type can't be updated.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="validationProvider" type="string">

          App ID of the validation provider called during create and update operations.

        </ResponseField>

        <ResponseField name="dynamicTranslationsProvider" type="string">

          App ID of the provider for additional translation keys.

        </ResponseField>

        <ResponseField name="ownershipTag" type="string">

          Ownership tag for components of this type.

        </ResponseField>

        <ResponseField name="componentReferenceDataProvider" type="string">

          App ID of the reference data provider for components of this type.

        </ResponseField>

        <ResponseField name="componentExample" type="ComponentExample">

          Pointer to an example component.

          <Expandable title="componentExample properties">

            <ResponseField name="appId" type="string">

              App ID of the example component.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="componentId" type="string">

              Component ID of the example component.

              Constraints: `format GUID`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="validationProviders" type="string[]">

          App IDs of validation providers called during create and update operations.

          Constraints: `format GUID`, `maxItems 100`.

        </ResponseField>

        <ResponseField name="enricherProvider" type="string">

          App ID of the enrichment provider called during create and update operations.

        </ResponseField>

        <ResponseField name="requiredPermissions" type="Permission[]">

          Permissions required to create a component of this type. Currently supports 1 permission.

          Constraints: `maxItems 1`.

          <Expandable title="requiredPermissions properties">

            <ResponseField name="permissionId" type="string">

              Permission ID.

              To find the permission ID, navigate to the [Permissions page](https://dev.wix.com/app-selector?title=Select+an+App&primaryButtonText=Select+Site&actionUrl=https%3A%2F%2Fdev.wix.com%2Fapps%2F%7BappId%7D%2Fdev-center-permissions/add) in the app's dashboard and search for the [permission you want to configure](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/configure-permissions-for-your-app#step-1--identify-required-permissions).

              Constraints: `minLength 1`, `maxLength 164`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="businessSchemaComponentTypeData" type="BusinessSchemaComponentTypeData">

      Metadata defined in business schema.

      <Expandable title="businessSchemaComponentTypeData properties">

        <ResponseField name="description" type="string">

          Component type description shown in the App Dashboard.

        </ResponseField>

        <ResponseField name="exposure" type="string">

          Exposure level of the component type.

          **Supported values:**

          - `PRIVATE` — Unexposed. to be used by dev team only
          - `INTERNAL` — Exposed internally
          - `PUBLIC` — Exposed to the world

        </ResponseField>

        <ResponseField name="learnMoreLink" type="string">

          Link to learn more about the component type.

        </ResponseField>

        <ResponseField name="schemaName" type="string">

          Business schema name.

          Constraints: `maxLength 128`.

        </ResponseField>

        <ResponseField name="spiComponent" type="boolean">

          Whether the component type configures a service plugin.

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Tab>
