---
title: "Trigger Object"
description: "The Trigger object in the Wix Trigger API."
icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Trigger ID. (per component version)

      **Read-only.**

    </ResponseField>

    <ResponseField name="appId" type="string">

      ID of the [app that created the trigger](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/add-a-trigger-to-your-app).

      Constraints: `format GUID`, `immutable`.

    </ResponseField>

    <ResponseField name="triggerKey" type="string">

      Trigger key.

      Learn about [setting up a trigger](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/add-a-trigger-to-your-app#step-1--set-up-the-trigger).

      Constraints: `minLength 1`, `maxLength 100`, `immutable`.

    </ResponseField>

    <ResponseField name="payloadDataSchema" type="object">

      Trigger [payload schema](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/the-trigger-payload-schema).

    </ResponseField>

    <ResponseField name="displayName" type="string">

      Trigger display name. Users see this name when they [choose a trigger for an automation](https://support.wix.com/en/article/wix-automations-creating-an-automation-with-the-new-builder#step-2-choose-a-trigger).

      **Read-only.** Constraints: `minLength 5`, `maxLength 80`.

    </ResponseField>

    <ResponseField name="filters" type="Filter[]">

      [Filter fields](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/about-triggers#filter-fields) applied to the trigger payload schema.

      Constraints: `maxItems 5`.

      <Expandable title="filters properties">

        <ResponseField name="id" type="string">

          Filter ID.

          Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="fieldKeyType" type="string">

          Field key type.

          **Supported values:**

          - `UNKNOWN_FIELD_KEY_TYPE` — The field key type is unknown.
          - `STATIC_FIELD_KEY` — The field key is a static value.
          - `DYNAMIC_FIELD_KEY` — The field key changes depending on the filter selection.

        </ResponseField>

        <ResponseField name="valueInput" type="ValueInput">

          Input field metadata.

          <Expandable title="valueInput properties">

            <ResponseField name="type" type="string">

              Input value type.

              **Supported values:**

              - `UNKNOWN_VALUE_INPUT` — Unknown filter field input.
              - `ENTITY_SELECTOR` — Filter field input is a selection of one from multiple entities, such as a form.
              - `USER_INPUT` — Filter field value is input from a Wix user.
              - `STATIC_ITEMS` — Filter field value is a static value defined in the payload schema.

            </ResponseField>

            <ResponseField name="label" type="string">

              Input field label.

            </ResponseField>

            <ResponseField name="description" type="string">

              Input field description. May be used for a tooltip.

            </ResponseField>

            <ResponseField name="required" type="boolean">

              Whether the Wix user is required to enter a value in the field.

              Default: `false`

            </ResponseField>

            <ResponseField name="reevaluateDynamicSchema" type="boolean">

              Whether Wix should reevaluate the dynamic schema when the field value changes.

              Default: `false`

            </ResponseField>

          </Expandable>

          <Expandable title="One of the following">

            <ResponseField name="inputOptions" type="inputOptions">

              inputOptions

              <Expandable title="inputOptions properties">

                <ResponseField name="entitySelector" type="EntitySelector">

                  Entity selector details.

                  _Nested structure omitted for brevity — see the `EntitySelector` object._

                </ResponseField>

                <ResponseField name="userInput" type="UserInput">

                  User input details.

                  _Nested structure omitted for brevity — see the `UserInput` object._

                </ResponseField>

                <ResponseField name="staticItems" type="StaticItems">

                  Static value details.

                  _Nested structure omitted for brevity — see the `StaticItems` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="followUpFilters" type="Filter[]">

          Filters to display after first selection is made.

          Constraints: `maxItems 5`.

          _Recursive reference to `Filter[]`._

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="implementedMethods" type="ImplementedMethods">

      Additional [service plugin methods](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/trigger-provider-service-plugin/introduction#implement-the-service-plugin-methods) implemented for this trigger.

      <Expandable title="implementedMethods properties">

        <ResponseField name="validateConfiguration" type="boolean">

          Whether [Validate Configuration](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/trigger-provider-service-plugin/validate-configuration) is implemented.

          Default: `false`.

        </ResponseField>

        <ResponseField name="getDynamicSchema" type="boolean">

          Whether [Get Dynamic Schema](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/trigger-provider-service-plugin/get-dynamic-schema) is implemented.

          Default: `false`.

        </ResponseField>

        <ResponseField name="refreshPayload" type="boolean">

          Whether [Refresh Payload](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/trigger-provider-service-plugin/refresh-payload) is implemented.

          Default: `false`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the trigger was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the trigger was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="baseUri" type="string">

      Trigger base URI. Wix calls your trigger at this base address.

      Learn more about the [Wix Automations Trigger Provider service plugin](https://dev.wix.com/docs/api-reference/business-management/automations/triggers/trigger-provider-service-plugin/introduction).

    </ResponseField>

  </Tab>
