---
title: "Slot Availability Object"
description: "Get the availability of a calendar's sessions, slots, and schedules."
icon: "cube"
---

Get the availability of a calendar's sessions, slots, and schedules.

<Tabs>

  <Tab title="REST">

    <ResponseField name="slot" type="Slot">

      The slot for the corresponding session, when the session is either a single session
      or a specific session generated from a recurring session.

      <Expandable title="slot properties">

        <ResponseField name="sessionId" type="string">

          ID for the slot's corresponding session, when the session is either a single session
          or a specific session generated from a recurring session.

          Deprecated. Please use `eventId` instead.

        </ResponseField>

        <ResponseField name="serviceId" type="string">

          Service ID.

        </ResponseField>

        <ResponseField name="scheduleId" type="string">

          Schedule ID.

        </ResponseField>

        <ResponseField name="startDate" type="string">

          The start time of this slot in [RFC 3339](https://www.rfc-editor.org/rfc/rfc3339)
          format.

          If `timezone` is specified,
          dates are based on the local date/time. This means that the timezone offset
          in the `start_date` is ignored.

        </ResponseField>

        <ResponseField name="endDate" type="string">

          The end time of this slot in
          [RFC 3339](https://www.rfc-editor.org/rfc/rfc3339) format.

          If `timezone` is specified,
          dates are based on the local date/time. This means that the timezone offset
          in the `end_date` is ignored.

        </ResponseField>

        <ResponseField name="timezone" type="string">

          The timezone for which slot availability is to be calculated.

          Learn more about [handling Daylight Savings Time (DST) for local time zones](https://dev.wix.com/api/rest/wix-bookings/availability-calendar/query-availability#wix-bookings_availability-calendar_query-availability_handling-daylight-savings-time-dst-for-local-time-zones)
          when calculating availability.

        </ResponseField>

        <ResponseField name="resource" type="SlotResource">

          The resource required for this slot. Currently, the only supported resource
          is the relevant staff member for the slot.

          <Expandable title="resource properties">

            <ResponseField name="id" type="string">

              Resource ID.

              **Read-only.**

            </ResponseField>

            <ResponseField name="name" type="string">

              Resource name. Read only.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="location" type="Location">

          Geographic location of the slot.

          <Expandable title="location properties">

            <ResponseField name="id" type="string">

              Business location ID. Available only for locations that are business locations,
              meaning the `location_type` is `"OWNER_BUSINESS"`.

            </ResponseField>

            <ResponseField name="name" type="string">

              Location name.

            </ResponseField>

            <ResponseField name="formattedAddress" type="string">

              The full address of this location.

            </ResponseField>

            <ResponseField name="formattedAddressTranslated" type="string">

              The full translated address of this location.

            </ResponseField>

            <ResponseField name="locationType" type="string">

              Location type.

              **Supported values:**

              - `UNDEFINED` — Undefined location type.
              - `OWNER_BUSINESS` — The business address, as set in the site’s general settings.
              - `OWNER_CUSTOM` — The address as set when creating the service.
              - `CUSTOM` — The address as set for the individual session.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="eventId" type="string">

          ID for the slot's corresponding event, when the event is either a single event
          or a specific event generated from a recurring event.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="bookable" type="boolean">

      Whether the slot is bookable. Bookability is determined by checking a
      session's open slots and booking policies. Locks are not taken into
      account.

    </ResponseField>

    <ResponseField name="totalSpots" type="integer">

      Total number of spots for this slot.
      For example, if a session has a total of 10 spots and 3 spots are booked,
      `spotsTotal` is 10 and `openSpots` is 7.

    </ResponseField>

    <ResponseField name="openSpots" type="integer">

      Number of open spots for this slot.

    </ResponseField>

    <ResponseField name="waitingList" type="WaitingList">

      An object describing the slot's waitlist and its occupancy.

      <Expandable title="waitingList properties">

        <ResponseField name="totalSpots" type="integer">

          Total number of spots and open spots for this waitlist.
          For example, a Yoga class with 10 waitlist spots and 3 registered
          on the waitlist has 10 `total_spots` and 7 `open_spots`.

        </ResponseField>

        <ResponseField name="openSpots" type="integer">

          Number of open spots for this waitlist.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="bookingPolicyViolations" type="BookingPolicyViolations">

      Booking policy violations for the slot.

      <Expandable title="bookingPolicyViolations properties">

        <ResponseField name="tooEarlyToBook" type="boolean">

          Bookings policy violation. Too early to book this slot.

        </ResponseField>

        <ResponseField name="tooLateToBook" type="boolean">

          Bookings policy violation. Too late to book this slot.

        </ResponseField>

        <ResponseField name="bookOnlineDisabled" type="boolean">

          Bookings policy violation. Online booking is disabled for this slot.

        </ResponseField>

        <ResponseField name="validSubmissionRequired" type="boolean">

          Bookings policy violation. Intake Form submission is required for this slot.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="locked" type="boolean">

      Indicates whether the slot is locked because a waitlist exists.
      When a slot frees up, the slot is offered to the next customer on the waitlist. Read-only.

    </ResponseField>

    <ResponseField name="isFromV2" type="boolean" deprecated>

      Deprecated. not in use since 2025-06-15.

      **Deprecated.** Deprecated. not in use since 2025-06-15. Removal: 2025-06-15.

    </ResponseField>

  </Tab>
