---
title: "Question Entry Object"
description: "The Question Entry object in the Wix Question Entry API."
icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Question entry ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time a question entry is updated.
      To prevent conflicting changes,
      the current revision must be specified when updating the question entry.

      Ignored when creating a question entry.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the question entry was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the question entry was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="question" type="string">

      Question text displayed to site visitors.

    </ResponseField>

    <ResponseField name="labels" type="Label[]">

      Labels assigned to this question entry for additional organization and integration purposes.
      Labels are not visible to site visitors.

      Constraints: `maxItems 100`.

      <Expandable title="labels properties">

        <ResponseField name="title" type="string">

          Label name.

          Constraints: `minLength 1`, `maxLength 36`.

        </ResponseField>

        <ResponseField name="sortOrder" type="number">

          Order of the question entry within the label.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="sortOrder" type="number">

      Order of the question entry within its category. Lower values appear first.

    </ResponseField>

    <ResponseField name="categoryId" type="string">

      ID of the category that this question entry belongs to.
      Each question entry must be assigned to exactly one category.

    </ResponseField>

    <ResponseField name="slug" type="string">

      URL slug generated from the question text for SEO-friendly URLs.

      **Read-only.**

    </ResponseField>

    <ResponseField name="shareLink" type="string">

      Direct link to this specific question entry for sharing.

      **Read-only.**

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Custom field data for the `questionEntry` object.

      [Extended fields](https://dev.wix.com/docs/build-apps/develop-your-app/extensions/backend-extensions/schema-plugins/about-schema-plugin-extensions) must be configured in the app dashboard before they can be accessed with API calls.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Tab>
