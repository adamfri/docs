title: "Program Object"
description: "A program is an Online Programs course, challenge, or structured journey."

A program is an Online Programs course, challenge, or structured journey.
Programs contain lifecycle state, public description, timeline, access rules,
pricing, SEO, rewards, and summary information for their content.

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Program ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the program is updated. To prevent conflicting changes, the current revision must be specified when updating a program.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the program was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the program was last updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="status" type="string">

      Current program lifecycle status.

      **Read-only.**

      **Supported values:**

      - `DRAFT` — The program is editable and visible only in management contexts.
      - `PUBLISHED` — The program is available according to its access restrictions.
      - `ENDED` — The program is no longer available for new participation.
      - `ARCHIVED` — The program is closed and removed from normal public discovery.

    </ResponseField>

    <ResponseField name="statusUpdatedDate" type="datetime">

      Date and time when `status` last changed.

      **Read-only.**

    </ResponseField>

    <ResponseField name="categoryIds" type="string[]">

      Category IDs assigned to the program. This field is read-only; use the Program Categories API to assign categories.

      **Read-only.** Constraints: `format GUID`, `maxItems 10`.

    </ResponseField>

    <ResponseField name="description" type="Description">

      Public-facing program title, details, and media.

      <Expandable title="description properties">

        <ResponseField name="title" type="string">

          Program title.

          Constraints: `maxLength 150`.

        </ResponseField>

        <ResponseField name="details" type="string">

          Detailed program description.

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

          <Expandable title="media properties">

              <Expandable title="image properties">

                <ResponseField name="id" type="string">

                  WixMedia image ID.

                </ResponseField>

                  Image URL.

                </ResponseField>

                <ResponseField name="height" type="integer">

                  Original image height.

                  **Read-only.**

                  Original image width.

                  **Read-only.**

                </ResponseField>

                <ResponseField name="altText" type="string">

                  Image alt text.

                </ResponseField>

                <ResponseField name="filename" type="string">

                  Image filename.

                  **Read-only.**

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="video" type="VideoV2">

              Program video.

              <Expandable title="video properties">

                <ResponseField name="id" type="string">

                  WixMedia ID.

                <ResponseField name="resolutions" type="VideoResolution[]">

                  Available resolutions for the video, starting with the optimal resolution.

                  **Read-only.** Constraints: `maxItems 100`.

                  _Nested structure omitted for brevity — see the `VideoResolution[]` object._

                <ResponseField name="filename" type="string">

                  Video filename.

                  **Read-only.**

              </Expandable>

    </ResponseField>

    <ResponseField name="timeline" type="Timeline">

      Program timeline settings. Use this to configure self-paced or scheduled programs.

      <Expandable title="timeline properties">

        <ResponseField name="selfPaced" type="boolean">

          Whether participants can progress at their own pace.

        </ResponseField>

        <ResponseField name="startDate" type="string">

          Local date when the scheduled program starts.

        </ResponseField>

        <ResponseField name="durationInDays" type="integer">

          Scheduled program duration in days.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="restrictions" type="Restrictions">

      Program access, participation, and progression rules.

      <Expandable title="restrictions properties">

        <ResponseField name="maxParticipants" type="integer">

          Maximum number of active participants allowed in the program.

        </ResponseField>

        <ResponseField name="hideFutureSteps" type="boolean">

          Whether future steps are hidden from participants until they become available.

        </ResponseField>

        <ResponseField name="resolveStepsInOrder" type="boolean">

          Whether participants must complete steps in order.

        </ResponseField>

        <ResponseField name="shareProgress" type="boolean">

          Whether participants can share progress in the connected group.

        </ResponseField>

        <ResponseField name="accessType" type="string">

          Program discovery and join access.

          **Supported values:**

          - `PUBLIC` — Anyone who can access the site can discover the program and start the join flow.
          - `PRIVATE` — The program can be discovered, but joining requires owner approval.
          - `SECRET` — The program is hidden from public discovery and visible only to participants, owners, and trusted apps.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="price" type="Money">

      Optional one-time payment price for joining the program.

      <Expandable title="price properties">

        <ResponseField name="value" type="string">

          Monetary amount. Decimal string with a period as a decimal separator (e.g., 3.99).

        </ResponseField>

        <ResponseField name="currency" type="string">

          Currency code. Must be valid ISO 4217 currency code (e.g., USD).

          **Read-only.** Constraints: `format CURRENCY`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="seo" type="Seo">

      SEO settings used for the program page.

      <Expandable title="seo properties">

        <ResponseField name="slug" type="string">

          Unique URL slug for the program page.

          Constraints: `format URL_SLUG`.

        </ResponseField>

        <ResponseField name="seoData" type="SeoSchema">

          Advanced SEO schema data.

          <Expandable title="seoData properties">

            <ResponseField name="tags" type="Tag[]">

              SEO tag information.

              <Expandable title="tags properties">

                <ResponseField name="type" type="string">

                  SEO tag type.

                  Supported values: `title`, `meta`, `script`, `link`.

                </ResponseField>

                <ResponseField name="props" type="object">

                  A `{"key": "value"}` pair object where each SEO tag property (`"name"`, `"content"`, `"rel"`, `"href"`) contains a value.
                  For example: `{"name": "description", "content": "the description itself"}`.

                </ResponseField>

                <ResponseField name="meta" type="object">

                  SEO tag metadata. For example, `{"height": 300, "width": 240}`.

                </ResponseField>

                <ResponseField name="children" type="string">

                  SEO tag inner content. For example, `<title> inner content </title>`.

                </ResponseField>

                <ResponseField name="custom" type="boolean">

                  Whether the tag is a [custom tag](https://support.wix.com/en/article/adding-additional-meta-tags-to-your-sites-pages) —
                  i.e. an entry from the site's Advanced/Custom Tags list.

                  This flag is scoped to that feature only. It is NOT an indicator of whether a standard tag
                  (`title`, `description`, Open Graph, etc.) reflects a manually configured value versus a computed
                  default/pattern — standard tags always resolve with `custom: false`, even when their content comes
                  from a page's manually saved SEO title/description override. Don't use this field to verify whether
                  a live page reflects a manual per-page SEO edit.

                  Limitation: when `ResolveStaticPageSeoTags` is called without `seoData`, the per-page SEO data is read
                  only from the Vibe/Wix-managed-headless override store. That store holds nothing for classic Wix Editor
                  or Wix Studio pages, so even genuine Advanced/Custom Tags saved on those pages are missing from the
                  response entirely — there, `false` means this API could not tell, not that no custom tag exists.
                  Pass `seoData` explicitly to resolve against a known set of tags.

                </ResponseField>

                <ResponseField name="disabled" type="boolean">

                  Whether the tag is disabled. If the tag is disabled, people can't find your page when searching for this phrase in search engines.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="settings" type="Settings">

              SEO general settings.

              <Expandable title="settings properties">

                <ResponseField name="preventAutoRedirect" type="boolean">

                  Whether the [automatical redirect visits](https://support.wix.com/en/article/customizing-your-pages-seo-settings-in-the-seo-panel) from the old URL to the new one is enabled.

                  Default: `false` (automatical redirect is enabled).

                </ResponseField>

                <ResponseField name="keywords" type="Keyword[]">

                  User-selected keyword terms for a specific page.

                  Constraints: `maxItems 5`.

                  _Nested structure omitted for brevity — see the `Keyword[]` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="url" type="PageUrl">

          Program page URL.

          **Read-only.**

          <Expandable title="url properties">

            <ResponseField name="base" type="string">

              The base URL. For premium sites, this is the domain. For free sites, this is the site URL. For example, `mysite.wixsite.com/mysite`.

            </ResponseField>

            <ResponseField name="path" type="string">

              The relative path for the page within the site. For example, `/product-page/a-product`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="imageUrl" type="string">

          Image URL derived from the program description media.

        </ResponseField>

        <ResponseField name="imageAlt" type="string">

          Alternative text for the SEO image.

          Constraints: `maxLength 1000`.

        </ResponseField>

        <ResponseField name="courseWorkload" type="string">

          Approximate SEO program duration in ISO 8601 duration format.

          **Read-only.** Constraints: `maxLength 100`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="rewards" type="Reward[]">

      Rewards assigned to participants when they reach configured milestones.

      Constraints: `maxItems 10`.

      <Expandable title="rewards properties">

        <ResponseField name="trigger" type="string">

          Program milestone that grants the reward. If omitted, the trigger defaults to `JOINED_TO_PROGRAM` for compatibility with existing rewards.

          **Supported values:**

          - `JOINED_TO_PROGRAM` — Default trigger for existing rewards. The reward is granted when a member joins the program.
          - `STEP_COMPLETED` — The reward is granted when at least one step is completed.
          - `ALL_STEPS_COMPLETED` — The reward is granted when all required steps are completed.

        </ResponseField>

        <ResponseField name="badgeIds" type="string[]">

          Badge IDs to assign when the reward is granted.

          Constraints: `format GUID`, `maxItems 50`.

        </ResponseField>

        <ResponseField name="certificate" type="Certificate">

          Certificate assigned when the reward is granted.

          <Expandable title="certificate properties">

            <ResponseField name="id" type="string">

              Certificate ID. Currently this is the program GUID.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="connectedDate" type="datetime">

              Date and time the certificate was connected to the program.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="socialGroupId" type="string">

      ID of the social group connected to the program, if one exists.

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Custom fields for apps that extend the Program entity.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="shouldSendInvoice" type="boolean">

      Whether to send an invoice after a single-payment purchase. When false, the buyer receives a payment confirmation email instead.

    </ResponseField>

    <ResponseField name="videoStepsSettings" type="VideoStepsSettings">

      Settings that control how video steps behave in this program.

      <Expandable title="videoStepsSettings properties">

        <ResponseField name="autoNavigate" type="boolean">

          Whether the participant is automatically moved to the next video step after finishing the current one.

        </ResponseField>

        <ResponseField name="requiredCompletionPercentage" type="number">

          Percentage of a video that must be watched to complete a video step.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="contentSummary" type="ContentSummary">

      Read-only counts of content entities in the program.

      **Read-only.**

      <Expandable title="contentSummary properties">

        <ResponseField name="stepCount" type="integer">

          Number of steps in the program.

          Constraints: `max 3000`.

        </ResponseField>

        <ResponseField name="sectionCount" type="integer">

          Number of sections in the program.

          Constraints: `max 1000`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="ownerUserId" type="string">

      ID of the site user who created the program.

      **Read-only.**

    </ResponseField>
