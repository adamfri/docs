    Retrieves a list of up to 100 pricing plan orders for currently logged-in member.

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.PRICING-PLANS.OWN-EXTERNAL">Perform member actions</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`, `MEMBER`.

    ## Endpoint

    ```http
    GET https://www.wixapis.com/pricing-plans/v2/member/orders
    ```

    ## Query Params

    <ParamField query="planIds" type="string[]">

      Filter by plan IDs.

      Constraints: `format GUID`.

    </ParamField>

    <ParamField query="autoRenewCanceled" type="boolean">

      Filter for orders where auto renewal was canceled.

    </ParamField>

    <ParamField query="orderStatuses" type="string[]">

      Filter by order status.

      **Supported values:**

      - `UNDEFINED` — Undefined order status.
      - `DRAFT` — Order has been initiated but payment hasn't been processed yet. The plan isn't yet available for use to the buyer.
      - `PENDING` — Order has been purchased and its start date is set in the future.
      - `ACTIVE` — Order has been processed. The plan is available for use.
      - `PAUSED` — Order, and use of the plan, is paused. The order, and use of the plan, can be resumed.
      - `ENDED` — Order has completed its duration and is no longer available for use.
      - `CANCELED` — Order has been canceled.

    </ParamField>

    <ParamField query="paymentStatuses" type="string[]">

      Filter by payment status.

      **Supported values:**

      - `UNDEFINED` — Undefined payment status.
      - `PAID` — Payment has been paid.
      - `REFUNDED` — Payment has been refunded.
      - `FAILED` — Payment transaction didn't complete.
      - `UNPAID` — Payment has not been paid.
      - `PENDING` — Billing has been initialized, but actual charge is yet to be made. This can happen for free trials and payments made with PayPal.
      - `NOT_APPLICABLE` — No payment was necessary. For example, for free plans or free trials.

    </ParamField>

    <ParamField query="limit" type="integer">

      Limit the number of pricing plans returned. Default limit is 50.

    </ParamField>

    <ParamField query="offset" type="integer">

      Number of entries to offset.

    </ParamField>

    <ParamField query="sorting" type="Sorting">

      Sorting direction (defaults to ASC) and field to sort by.

      <Expandable title="sorting properties">

        <ParamField query="fieldName" type="string">

          Name of the field to sort by.

          Constraints: `maxLength 512`.

        </ParamField>

        <ParamField query="order" type="string">

          Sort order.

          **Supported values:**

          - `ASC`
          - `DESC`

        </ParamField>

        <ParamField query="origin" type="AddressLocation">

          Origin point for geo-distance sorting on a GEO field
          results are ordered by distance from this point (ASC = nearest first, DESC = farthest first).

          <Expandable title="origin properties">

            <ParamField query="latitude" type="number">

              Address latitude.

            </ParamField>

            <ParamField query="longitude" type="number">

              Address longitude.

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField query="fieldSet" type="string">

      Predefined set of fields to return.

      Default: If `fieldSet` is omitted, no order form submission data is returned.

      **Supported values:**

      - `UNKNOWN_SET` — Same behavior as `BASIC`.``
      - `BASIC` — Doesn't return any order form submission data.
      - `FULL` — Returns all order form submission data.

    </ParamField>

    export const listCurrentMemberOrders = webMethod(

        try {

          return result;
        } catch (error) {
          console.error(error);

        }

    export const requestCancellation = webMethod(
      Permissions.Anyone, 
      async (orderId) => {
        const effectiveAt = 'NEXT_PAYMENT_DATE';

          return;
        } catch (error) {
          console.error(error);
          // Handle the error
        }
      });

    export const pauseOrder = webMethod(
      Permissions.Anyone, 
      async (orderId) => {
        try {

        }

    /**********************************
     * Backend code - order.web.js/ts *
     *********************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { orders } from '@wix/pricing-plans';
    import { auth } from '@wix/essentials';

    const elevatedResumeOrder = auth.elevate(orders.resumeOrder);

    export const getPausedMemberOrders = webMethod(
      Permissions.Anyone,
      async () => {
        try {
          const options = {
            orderStatuses: 'PAUSED'
          };
          const result = await orders.memberListOrders(options);
          const getPausedMemberOrders = result.orders;

          return result;
        } catch (error) {
          console.error(error);
          // Handle the error
        }
      });

    export const resumeOrder = webMethod(
      Permissions.Anyone,
      async (orderId) => {
        try {
          await elevatedResumeOrder(orderId);

          return;
        } catch (error) {
          console.error(error);
          // Handle the error
        }
      });

    /*************
     * Page code *
     *************/

    import { getPausedMemberOrders, resumeOrder } from 'backend/order.web';

    $w.onReady(function () {
      let selectedOrderId;
      populateOrderButtonList();

      $w('#orderSelection').onChange(() => {
        selectedOrderId = $w('#orderSelection').value;
        $w('#resumeOrderBtn').enable();
      });

      $w('#resumeOrderBtn').onClick(async () => {
        await resumeOrder(selectedOrderId);
      });

    });

    async function populateOrderButtonList() {
      const pausedOrders = await getPausedMemberOrders();
      $w('#pausedOrders').options = pausedOrders.orders.map(item => {
        return {
          label: item.planName,
          value: item._id
        }
      });

      $w('#pausedOrders').show();
    }
