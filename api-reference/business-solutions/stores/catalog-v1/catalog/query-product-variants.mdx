    Retrieves product variants, based on either choices (option-choice key-value pairs) or variant IDs.
    See [Stores Pagination](https://dev.wix.com/api/rest/wix-stores/pagination) for more information.

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-STORES.READ-PRODUCTS">Read Products</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`, `MEMBER`, `VISITOR`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/stores/v1/products/{id}/variants/query
    ```

    ## Path Params

    <ParamField path="id" type="string" required>

      Requested product ID.

    </ParamField>

    ## Body Params

    <ParamField body="choices" type="object">

      The specific choices available or chosen from within a selection (e.g., choosing the red Selection triggers the red Choice).
      You may specify all the relevant choices for a specific variant, or only some of the options, which will return all corresponding variants (not relevant when passing variant IDs).

      Constraints: `format map`.

      <Expandable title="choices properties">

        <ParamField body="key" type="string" />

        <ParamField body="value" type="string" />

      </Expandable>

    </ParamField>

    <ParamField body="variantIds" type="string[]">

      List of variant IDs (not relevant when passing choices).

    </ParamField>

    <ParamField body="paging" type="Paging">

      paging

      <Expandable title="paging properties">

        <ParamField body="limit" type="integer">

          Amount of items to load per page

        </ParamField>

        <ParamField body="offset" type="integer">

          Number of items to skip in the display (relevant for all pages after the first)

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="includeMerchantSpecificData" type="boolean">

      Whether merchant specific data should be included in the response. Requires permissions to manage products.

    </ParamField>

    ## Response Object

    import { Permissions, webMethod } from '@wix/web-methods';
    import { products } from '@wix/stores';

    export const getProductVariants = webMethod(Permissions.Anyone, (_id, options) => {
      return products.queryProductVariants(_id, options);
    });

    /*************
     * Page code *
     *************/

    import { getProductVariants } from 'backend/products.web';

    let _id = "3fb6a3c8-988b-7895-04bd-5c59ae0b18ea"; // get product ID
    let options = {
      choices: {
        "Size": "Large",
        "Color": "Red"
