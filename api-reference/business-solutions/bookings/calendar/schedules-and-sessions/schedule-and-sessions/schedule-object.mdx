<ResponseField name="id" type="string">

  Schedule ID.

</ResponseField>

<ResponseField name="scheduleOwnerId" type="string">

  ID of the schedule's owner entity. This may be a resource ID or a service ID.

</ResponseField>

<ResponseField name="timeZone" type="string">

  Schedule's time zone in [Area/Location](https://en.wikipedia.org/wiki/Tz_database) format. Read-only.
  Derived from the Wix Business time zone.

  **Read-only.**

</ResponseField>

<ResponseField name="intervals" type="RecurringInterval[]" deprecated>

  Deprecated. Please use the [Sessions API](https://dev.wix.com/api/rest/wix-bookings/schedules-and-sessions/session) instead.

  **Deprecated.**

  <Expandable title="intervals properties">

    <ResponseField name="id" type="string">

      The recurring interval identifier.

      **Read-only.**

    </ResponseField>

    <ResponseField name="start" type="datetime">

      The start time of the recurring interval. Required.

    </ResponseField>

    <ResponseField name="end" type="datetime">

      The end time of the recurring interval. Optional. Empty value indicates that there is no end time.

    </ResponseField>

    <ResponseField name="interval" type="Interval">

      The interval rules. The day, hour and minutes the interval is recurring.

      <Expandable title="interval properties">

        <ResponseField name="daysOfWeek" type="string">

          The day the interval occurs. Optional. The default is the day of the recurring interval's start time.

          **Supported values:**

          - `UNDEFINED` — Undefined.
          - `MON` — Monday.
          - `TUE` — Tuesday.
          - `WED` — Wednesday.
          - `THU` — Thursday.
          - `FRI` — Friday.
          - `SAT` — Saturday.
          - `SUN` — Sunday.

        </ResponseField>

        <ResponseField name="hourOfDay" type="integer">

          The hour of the day the interval occurs. Must be consistent with the interval start time. Optional. The default is 0. Minimum: 0, maximum: 23.

        </ResponseField>

        <ResponseField name="minuteOfHour" type="integer">

          The minutes of the hour the interval accrues. Must be consistent with the interval end time. Optional. The default is 0. Minimum: 0, maximum: 59.

        </ResponseField>

        <ResponseField name="duration" type="integer">

          The duration of the interval in minutes. Required. Part of the session end time calculation.

          Constraints: `min 0`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="frequency" type="Frequency">

      The frequency of the interval. Optional. The default is frequency with the default repetition.

      <Expandable title="frequency properties">

        <ResponseField name="repetition" type="integer">

          The frequency of the recurrence in weeks. i.e. when this value is 4, the interval occurs every 4 weeks. Optional. The default is 1. minimum: 1, maximum: 52.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="affectedSchedules" type="LinkedSchedule[]">

      Specifies the list of linked schedules and the way this link affects the corresponding schedules' availability. Can be calculated from the schedule or overridden on the recurring interval.

      <Expandable title="affectedSchedules properties">

        <ResponseField name="scheduleId" type="string">

          Schedule ID.

          Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="transparency" type="string">

          Sets this schedule's availability for the duration of the linked schedule's sessions. Default is `"BUSY"`.

          **Supported values:**

          - `UNDEFINED`
          - `FREE` — The schedule can have available slots during the linked schedule's sessions.
          - `BUSY` — The schedule can't have available slots during the linked schedule's sessions.

        </ResponseField>

        <ResponseField name="scheduleOwnerId" type="string">

          Owner ID, of the linked schedule.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="intervalType" type="string">

      The type of recurring interval.

      **Supported values:**

      - `UNDEFINED` — The default value. Sessions for this interval will be of type EVENT.
      - `EVENT` — A recurring interval of events.
      - `TIME_AVAILABILITY` — Deprecated.
      - `AVAILABILITY` — A recurring interval for availability.

    </ResponseField>

  </Expandable>

</ResponseField>

<ResponseField name="title" type="string">

  Default title for the schedule's sessions. Maximum length: 6000 characters.

</ResponseField>

<ResponseField name="tags" type="string[]" deprecated>

  __Deprecated.__
  Tags for grouping schedules. These tags are the default tags for the schedule's sessions.
  The Wix Bookings app uses the following predefined tags to set schedule type: `"INDIVIDUAL"`, `"GROUP"`, and `"COURSE"`. Once the schedule type is set using these tags, you cannot update it. In addition to the app's tags, you can create and update your own tags.

  **Deprecated.**

</ResponseField>

<ResponseField name="location" type="Location">

  Default location for the schedule's sessions.

  <Expandable title="location properties">

    <ResponseField name="locationType" type="string">

      Location type.

      **Supported values:**

      - `UNDEFINED` — Undefined location type.
      - `OWNER_BUSINESS` — The business address as set in the site’s general settings.
      - `OWNER_CUSTOM` — The address as set when creating the service.
      - `CUSTOM` — The address set for the individual session.

    </ResponseField>

    <ResponseField name="address" type="string">

      Free text address used when locationType is `OWNER_CUSTOM`.

    </ResponseField>

    <ResponseField name="customAddress" type="Address">

      Custom address, used when locationType is `"OWNER_CUSTOM"`. Might be used when locationType is `"CUSTOM"` in case the owner sets a custom address for the session which is different from the default.

      <Expandable title="customAddress properties">

        <ResponseField name="country" type="string">

          Country code.

        </ResponseField>

        <ResponseField name="subdivision" type="string">

          Subdivision. Usually state, region, prefecture or province code, according to [ISO 3166-2](https://en.wikipedia.org/wiki/ISO_3166-2).

        </ResponseField>

        <ResponseField name="city" type="string">

          City name.

        </ResponseField>

        <ResponseField name="postalCode" type="string">

          Zip/postal code.

        </ResponseField>

        <ResponseField name="addressLine2" type="string">

          Free text providing more detailed address info. Usually contains Apt, Suite, and Floor.

        </ResponseField>

        <ResponseField name="formattedAddress" type="string">

          A string containing the full address of this location.

        </ResponseField>

        <ResponseField name="hint" type="string">

          Free text to help find the address.

        </ResponseField>

        <ResponseField name="geocode" type="AddressLocation">

          Coordinates of the physical address.

          <Expandable title="geocode properties">

            <ResponseField name="latitude" type="number">

              Address latitude.

            </ResponseField>

            <ResponseField name="longitude" type="number">

              Address longitude.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="countryFullname" type="string">

          Country full name.

        </ResponseField>

        <ResponseField name="subdivisions" type="Subdivision[]">

          Multi-level subdivisions from top to bottom.

          <Expandable title="subdivisions properties">

            <ResponseField name="code" type="string">

              Subdivision code. Usually state, region, prefecture or province code, according to [ISO 3166-2](https://en.wikipedia.org/wiki/ISO_3166-2).

            </ResponseField>

            <ResponseField name="name" type="string">

              Subdivision full name.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="street" type="street">

          street

          <Expandable title="street properties">

            <ResponseField name="streetAddress" type="StreetAddress">

              Street name, number and apartment number.

              _Nested structure omitted for brevity — see the `StreetAddress` object._

            </ResponseField>

            <ResponseField name="addressLine" type="string">

              Main address line, usually street and number, as free text.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Expandable>

</ResponseField>

<ResponseField name="capacity" type="integer">

  Maximum number of participants that can be added to the schedule's sessions.
  Must be at most `1` for schedule whose availability is affected by another schedule. E.g, appointment schedules of the Wix Bookings app.

</ResponseField>

<ResponseField name="availability" type="Availability" deprecated>

  __Deprecated.__

  **Deprecated.**

  <Expandable title="availability properties">

    <ResponseField name="start" type="datetime">

      Date and time the schedule starts to be available for booking.

    </ResponseField>

    <ResponseField name="end" type="datetime">

      Date and time the schedule stops being available for booking. No value indicates no end time.

    </ResponseField>

    <ResponseField name="linkedSchedules" type="LinkedSchedule[]">

      Other schedules that impact the availability calculation. Relevant only when there are availability constraints.

      <Expandable title="linkedSchedules properties">

        <ResponseField name="scheduleId" type="string">

          Schedule ID.

          Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="transparency" type="string">

          Sets this schedule's availability for the duration of the linked schedule's sessions. Default is `"BUSY"`.

          **Supported values:**

          - `UNDEFINED`
          - `FREE` — The schedule can have available slots during the linked schedule's sessions.
          - `BUSY` — The schedule can't have available slots during the linked schedule's sessions.

        </ResponseField>

        <ResponseField name="scheduleOwnerId" type="string">

          Owner ID, of the linked schedule.

          **Read-only.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="constraints" type="AvailabilityConstraints">

      Constraints for calculating the schedule's availability.

      <Expandable title="constraints properties">

        <ResponseField name="slotDurations" type="integer[]">

          A list of duration options for slots, in minutes. Minimum value for a duration is 1.
          The availability calculation generates slots with these durations, where there is no conflict with existing sessions or other availability constraints.

          Constraints: `min 1`.

        </ResponseField>

        <ResponseField name="timeBetweenSlots" type="integer">

          The number of minutes between the `end` of one slot, and the `start` of the next.
          Minimum value is 0, maximum value is 120.

          Constraints: `min 0`, `max 720`.

        </ResponseField>

        <ResponseField name="splitInterval" type="integer">

          Specify how to split the slots in intervals of minutes.
          This value indicates the time between available slots' start time. e.g., from 5 minute slots (`3:00`, `3:05`, `3:15`) and 1 hour slots (`3:00`, `4:00`, `5:00`).
          Optional. The default is the first duration in slot_durations field.
          Deprecated. Use the `split_slots_interval.value_in_minutes`.

        </ResponseField>

        <ResponseField name="slotsSplitInterval" type="SplitInterval">

          An object defining the time between available slots' start times.  For example, a slot with slots_split_interval=5 can start every 5 minutes. The default is the slot duration.

          **Read-only.**

          <Expandable title="slotsSplitInterval properties">

            <ResponseField name="sameAsDuration" type="boolean">

              Whether the slot duration is used as the split interval value.
              If `same_as_duration` is `true`, the `value_in_minutes` is the sum of the first duration in
              `schedule.availabilityConstraints.SlotDurations` field, and `schedule.availabilityConstraints.TimeBetweenSlots` field.

            </ResponseField>

            <ResponseField name="valueInMinutes" type="integer">

              Number of minutes between available slots' start times when `same_as_duration` is `false`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Expandable>

</ResponseField>

<ResponseField name="totalNumberOfParticipants" type="integer">

  Number of participants registered to sessions in this schedule, calculated as the sum of the party sizes.

  **Read-only.**

</ResponseField>

<ResponseField name="participants" type="Participant[]">

  *Partial list** of participants which are registered to sessions in this schedule.
  Participants who are registered in the schedule are automatically registered to any session that is created for the schedule.
  To retrieve the full list of schedule participants please use the [Query Extended Bookings API](https://dev.wix.com/api/rest/wix-bookings/bookings-reader-v2/query-extended-bookings).

  **Read-only.**

  <Expandable title="participants properties">

    <ResponseField name="id" type="string">

      Participant ID. Currently represents the booking.id.

      Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="contactId" type="string">

      Contact ID.

    </ResponseField>

    <ResponseField name="name" type="string">

      Participant's name.

    </ResponseField>

    <ResponseField name="phone" type="string">

      Participant's phone number.

    </ResponseField>

    <ResponseField name="email" type="string">

      Participant's email address.

    </ResponseField>

    <ResponseField name="partySize" type="integer">

      Group or party size. The number of people attending. Defaults to 0. Maximum is 250.

      Constraints: `min 0`.

    </ResponseField>

    <ResponseField name="approvalStatus" type="string">

      Approval status for the participant.
      &lt;!-- Commented out untill updateParticipant is exposed Generally the same status as the booking, unless updated using the `updateParticipant()` API. Defaults to `"UNDEFINED"`.-->

      **Supported values:**

      - `UNDEFINED` — Default.
      - `PENDING` — Pending business approval.
      - `APPROVED` — Approved by the business.
      - `DECLINED` — Declined by the business.

    </ResponseField>

    <ResponseField name="inherited" type="boolean">

      Whether the participant was inherited from the schedule, as opposed to being booked directly to the session.

      **Read-only.**

    </ResponseField>
