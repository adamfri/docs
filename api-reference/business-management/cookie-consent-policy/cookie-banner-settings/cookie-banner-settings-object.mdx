    <ResponseField name="appEnabled" type="boolean" deprecated>

      **Deprecated.** Use `enabled` instead. Removal: 2024-04-08.

    </ResponseField>

    <ResponseField name="audience" type="string" deprecated>

      **Deprecated.** Removal: 2024-04-08.

      **Supported values:**

      - `unknown_audience`
      - `all_visitors`
      - `eu_visitors`

    </ResponseField>

    <ResponseField name="cookiePolicyRequirementApproved" type="boolean">

      Whether the site contributor has acknowledged the consent policy requirements.

    </ResponseField>

    <ResponseField name="declineAllConfig" type="DeclineAllConfig">

      Configuration settings for the "decline all" button.

      <Expandable title="declineAllConfig properties">

        <ResponseField name="enabled" type="boolean">

          Whether to show the "decline all" button on the cookie banner.

        </ResponseField>

        <ResponseField name="geo" type="string[]" deprecated>

          **Deprecated.**

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="expiryDate" type="datetime">

      For internal use.

    </ResponseField>

    <ResponseField name="privacyPolicyPage" type="string" deprecated>

      **Deprecated.** Use `privacy_policy_link` instead. Removal: 2024-04-08.

    </ResponseField>

    <ResponseField name="privacyPolicyType" type="string" deprecated>

      **Deprecated.** Use `privacy_policy_link` instead. Removal: 2024-04-08.

      **Supported values:**

      - `unknown_privacy_policy_type`
      - `page_on_site`
      - `external_url`

    </ResponseField>

    <ResponseField name="texts" type="object">

      Texts to be displayed on the cookie banner. See [Banner texts]() for details.

      Constraints: `format map`.

      <Expandable title="texts properties">

        <ResponseField name="key" type="string" />

        <ResponseField name="value" type="string" />

      </Expandable>

    </ResponseField>

    <ResponseField name="theme" type="string">

      Visual theme.

      **Supported values:**

      - `unknown_theme`
      - `light`
      - `dark`
      - `custom`

    </ResponseField>

    <ResponseField name="revisitSettingsConfig" type="RevisitSettingsConfig">

      Configuration settings for the "revisit settings" button.

      <Expandable title="revisitSettingsConfig properties">

        <ResponseField name="enabled" type="boolean">

          Whether to show a minimized cookie banner on all pages at all times.

        </ResponseField>

        <ResponseField name="buttonPosition" type="string">

          Position of the minimized cookie banner.

          **Supported values:**

          - `unknown_button_position`
          - `bottom_right`
          - `bottom_left`
          - `center_right`
          - `center_left`

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="customThemeConfig" type="CustomThemeConfig">

      Custom theme configuration.

      <Expandable title="customThemeConfig properties">

        <ResponseField name="primaryColor" type="SiteColor">

          Primary color displayed on the cookie banner in [hex](https://www.color-hex.com/color-palette/1018144) format.

          <Expandable title="primaryColor properties">

            <ResponseField name="name" type="string">

              Color name.

            </ResponseField>

            <ResponseField name="value" type="string">

              Color value in [hex](https://www.color-hex.com/color-palette/1018144) format.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="secondaryColor" type="SiteColor">

          Secondary color displayed on the cookie banner in [hex](https://www.color-hex.com/color-palette/1018144) format.

          <Expandable title="secondaryColor properties">

            <ResponseField name="name" type="string">

              Color name.

            </ResponseField>

            <ResponseField name="value" type="string">

              Color value in [hex](https://www.color-hex.com/color-palette/1018144) format.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="fontFamily" type="string">

          Font displayed on the cookie banner.

        </ResponseField>

        <ResponseField name="fontSize" type="string">

          Font size displayed on the cookie banner.

        </ResponseField>

        <ResponseField name="cornerRadius" type="string">

          Corner style for buttons on the cookie banner.

          **Supported values:**

          - `unknown_corner_radius`
          - `square`
          - `rounded_corners`
          - `round`

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="enabled" type="boolean">

      Whether the banner is activated.

    </ResponseField>

    <ResponseField name="privacyPolicyPageInfo" type="PrivacyPolicyPageInfo">

      Location of the site's privacy policy.

      <Expandable title="privacyPolicyPageInfo properties">

        <ResponseField name="privacyPolicyDetailsReferenceType" type="string">

          **Supported values:**

          - `UNKNOWN_PRIVACY_POLICY_TYPE`
          - `SLUG`
          - `EXTERNAL_PAGE_URL`

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="details" type="details">

          details

          <Expandable title="details properties">

            <ResponseField name="slugDetails" type="SlugDetails">

              Internal page slug containing the privacy policy.

              <Expandable title="slugDetails properties">

                <ResponseField name="slug" type="string">

                  Slug of the site page containing the privacy policy.

                  Constraints: `format URL_SLUG`.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="externalPageUrlDetails" type="ExternalPageUrlDetails">

              External URL containing the privacy policy.

              <Expandable title="externalPageUrlDetails properties">

                <ResponseField name="externalPageUrl" type="string">

                  URL of the external page containing the privacy policy.

                  Constraints: `format WEB_URL`.

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>
