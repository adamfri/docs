  **Developer Preview.** This method is in Developer Preview and may change without notice.

</Info>

This endpoint is used to send a merchant's payment requests to a Payment Service Provider (PSP).

Wix calls this endpoint when a buyer chooses make a payment using the PSP on a merchant's Wix site.
Learn more about [implementing this endpoint](https://dev.wix.com/docs/rest/business-management/payments/service-plugins/payment-service-provider-service-plugin/processing-payments).

## Endpoint

```http
POST /sale
```

## Body Params

<ParamField body="wixTransactionId" type="string" required>

  Wix transaction ID.

</ParamField>

<ParamField body="paymentMethod" type="string">

  Requested [payment method](https://dev.wix.com/docs/rest/business-management/payments/service-plugins/payment-service-provider-service-plugin/payment-methods).
  If the value of this field is `creditCard`, the `paymentMethodData`
  field's value includes credit card data in the value of the `card` field.
  A PSP can specify the payment methods they accept in the [Wix Dev Ceter](https://dev.wix.com).

</ParamField>

<ParamField body="merchantCredentials" type="object" required>

  Credentials that the PSP provided when the merchant connected their account.

  Constraints: `format map`.

  <Expandable title="merchantCredentials properties">

    <ParamField body="key" type="string" />

    <ParamField body="value" type="string" />

  </Expandable>

</ParamField>

<ParamField body="order" type="Order" required>

  Order information.

  <Expandable title="order properties">

    <ParamField body="id" type="string">

      Unique order ID assigned by Wix.

    </ParamField>

    <ParamField body="description" type="OrderDescription" required>

      Order description.

      <Expandable title="description properties">

        <ParamField body="totalAmount" type="string">

          Total amount to pay in a currency’s smallest unit.
          Includes all extra charges such as shipping and taxes.

          Learn more about [currency units](https://dev.wix.com/docs/rest/api-reference/payment-provider-spi/currencies).

        </ParamField>

        <ParamField body="currency" type="string">

          Code for the merchant's preferred [currency](https://en.wikipedia.org/wiki/ISO_4217).

          Constraints: `format CURRENCY`.

        </ParamField>

        <ParamField body="items" type="OrderItem[]">

          Order items.

          _Nested structure omitted for brevity — see the `OrderItem[]` object._

        </ParamField>

        <ParamField body="charges" type="AdditionalCharges">

          Additional charges applied to the order.

          _Nested structure omitted for brevity — see the `AdditionalCharges` object._

        </ParamField>

        <ParamField body="billingAddress" type="Address">

          Order billing address.

          _Nested structure omitted for brevity — see the `Address` object._

        </ParamField>

        <ParamField body="shippingAddress" type="Address">

          Order shipping address.

          _Nested structure omitted for brevity — see the `Address` object._

        </ParamField>

        <ParamField body="buyerInfo" type="BuyerInfo">

          Buyer information.

          _Nested structure omitted for brevity — see the `BuyerInfo` object._

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="returnUrls" type="OrderReturnUrls">

      Redirect URLs for different payment outcomes. PSP-hosted payment pages use these URLs to redirect
      the buyer back to the merchant’s site.

      <Expandable title="returnUrls properties">

        <ParamField body="successUrl" type="string">

          Redirect URL for successful payments.

        </ParamField>

        <ParamField body="errorUrl" type="string">

          Redict URL for failed payments.

        </ParamField>

        <ParamField body="cancelUrl" type="string">

          Redirect URL for cancelled payments.

        </ParamField>

        <ParamField body="pendingUrl" type="string">

          Redirect URL for pending payments.

        </ParamField>

      </Expandable>

    </ParamField>

  </Expandable>

</ParamField>

<ParamField body="mode" type="string">

  Connection mode. Identifies whether a request is live or for testing purposes.

  **Supported values:**

  - `undefined` — Undefined mode.
  - `live` — Request is live.
  - `sandbox` — Request is for testing purposes.

</ParamField>

<ParamField body="installments" type="integer">

  Number of payment installments requested.

</ParamField>

<ParamField body="wixMerchantId" type="string" required>

  Wix merchant ID.

</ParamField>

<ParamField body="paymentMethodData" type="PaymentMethodData">

  Payment method data such as credit card information.

  <Expandable title="One of the following">

    <ParamField body="payment" type="payment">

      payment

      <Expandable title="payment properties">

        <ParamField body="card" type="Card">

          Credit card information.

          _Nested structure omitted for brevity — see the `Card` object._

        </ParamField>

        <ParamField body="ideal" type="IDeal">

          IDeal payment information.

          _Nested structure omitted for brevity — see the `IDeal` object._

        </ParamField>

        <ParamField body="reference" type="PaymentMethodReference">

          Recurring payment token data.

          _Nested structure omitted for brevity — see the `PaymentMethodReference` object._

        </ParamField>

        <ParamField body="applePay" type="ApplePay">

          ApplePay payment information.

          _Nested structure omitted for brevity — see the `ApplePay` object._

        </ParamField>

        <ParamField body="googlePay" type="GooglePay">

          GooglePay payment information.

          _Nested structure omitted for brevity — see the `GooglePay` object._

        </ParamField>

        <ParamField body="cardReader" type="CardReader">

          POS card reader payment information.

          _Nested structure omitted for brevity — see the `CardReader` object._

        </ParamField>

      </Expandable>

    </ParamField>

  </Expandable>

</ParamField>

<ParamField body="fraudInformation" type="FraudInformation">

  Fraud prevention data.

  <Expandable title="fraudInformation properties">

    <ParamField body="remoteIp" type="string">

      Buyer's IP address.

    </ParamField>

  </Expandable>

</ParamField>

<ParamField body="setupCredentialsOnFile" type="SetupCredentialsOnFile">

  Data about storing credentials on file for future payments.

  <Expandable title="setupCredentialsOnFile properties">

    <ParamField body="offSession" type="boolean">

      Whether the request is for storing credentials for future off-session payments.
      Learn more about [recurring payments](https://dev.wix.com/docs/rest/api-reference/payment-provider-spi/sample-payment-flows#recurring-payments).

    </ParamField>

  </Expandable>

</ParamField>

<ParamField body="offSession" type="boolean">

  Whether the request is for a recurring payment or merchant initiated transaction. If the value of this field is `true`, the PSP should use a payment method token to process the payment.
  Learn more about [recurring payments](https://dev.wix.com/docs/rest/api-reference/payment-provider-spi/sample-payment-flows#recurring-payments).

</ParamField>

<ParamField body="moto" type="boolean">

  Whether the request is for a Mail Order/Telephone Order payment.

</ParamField>

<ParamField body="scaRequested" type="boolean">

  Request transaction uses strong customer authentication, such as 3-D Secure.

</ParamField>

<ParamField body="authorizeOnly" type="boolean">

  Whether to only authorize or to also immediately capture the transaction. If `true`, Wix sends requests to the PSP's Capture Transaction or Void Transaction endpoints to complete the transaction.
  &lt;br/>
  Default: `false`

</ParamField>

<ParamField body="recurring" type="boolean">

  Whether an off-session payment is scheduled.

  Only consult this field when `offSession` is `true`.
  If `true`, the payment is scheduled, such as a subscription payment.
  If `false` or absent, the payment is not scheduled.

</ParamField>

<ParamField body="siteId" type="string">

  Wix site ID.

</ParamField>

<ParamField body="partialAuthAllowed" type="boolean">

  Whether the partial authorization is allowed. If the payment is actually partially authorized,
  it should never be captured automatically regardless of the `authorizeOnly` field value.

</ParamField>

## Response Object

<ResponseField name="pluginTransactionId" type="string">

  PSP transaction ID.

</ResponseField>

<ResponseField name="reasonCode" type="integer">

  Wix [reason code](https://dev.wix.com/docs/rest/api-reference/payment-provider-spi/reason-codes) indicating a failed or pending request.

</ResponseField>

<ResponseField name="errorCode" type="string">

  PSP-specific error code.

</ResponseField>

<ResponseField name="errorMessage" type="string">

  PSP-specific error message.

</ResponseField>

<ResponseField name="redirectUrl" type="string">

  URL where Wix can redirect the buyer to complete the payment.

</ResponseField>

<ResponseField name="status" type="string" deprecated>

  This field is deprecated.

  **Deprecated.** This field is deprecated.

  **Supported values:**

  - `UNDEFINED`
  - `APPROVED`
  - `PENDING`
  - `CANCELED`
  - `DECLINED`

</ResponseField>

<ResponseField name="credentialsOnFile" type="CredentialsOnFile">

  Reference to a saved payment method.

  <Expandable title="credentialsOnFile properties">

    <ResponseField name="paymentMethod" type="string">

      Payment method (for example `creditCard`) to use for future payments, when it differs from the current one.

    </ResponseField>

  </Expandable>

  <Expandable title="One of the following">

    <ResponseField name="info" type="info">

      info

      <Expandable title="info properties">

        <ResponseField name="cardReference" type="CardReference">

          Network token data.

          _Nested structure omitted for brevity — see the `CardReference` object._

        </ResponseField>

        <ResponseField name="paymentMethodReference" type="PaymentMethodReference">

          Provider generated token data.

          _Nested structure omitted for brevity — see the `PaymentMethodReference` object._

        </ResponseField>

      </Expandable>

    </ResponseField>

  </Expandable>

</ResponseField>

<ResponseField name="authorizedAmount" type="integer">

  Total amount authorized in a currency's minor units. Required when reasonCode=5012.

</ResponseField>

## Errors

<Note>
