<ResponseField name="cardNumberToken" type="string">

  Credit/debit card number token.

</ResponseField>

<ResponseField name="securityCodeToken" type="string">

  Credit/debit card security code (CVV) token.

</ResponseField>

<ResponseField name="tokenExpirationDate" type="datetime">

  Token expiration date.
  Value encoded as a string in the
  [RFC 3339](https://www.ietf.org/rfc/rfc3339.txt) format. This
  format is "&#123;year&#125;-&#123;month&#125;-&#123;day&#125;T&#123;hour&#125;:&#123;min&#125;:&#123;sec&#125;[.&#123;frac_sec&#125;]Z".
  For example "2017-01-15T01:30:15.01Z"

</ResponseField>
