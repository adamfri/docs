<ResponseField name="id" type="string">

  Wix Payments transaction ID.

  Constraints: `format GUID`.

</ResponseField>

<ResponseField name="currency" type="string">

  Payment currency according to ISO 4217. For example: USD, EUR, GBP.

  Constraints: `format CURRENCY`.

</ResponseField>

<ResponseField name="capturableAmount" type="number">

  Amount that can be captured or voided in minor units.
  Usually matches authorization amount when the transaction has been authorized but not captured.

</ResponseField>

<ResponseField name="refundableAmount" type="number">

  Amount that can be refunded in minor units.
  Usually matches capture amount when the transaction has been captured but not refunded or charged back.

</ResponseField>

<ResponseField name="externalInvoiceId" type="string">

  Invoice ID provided by the external entity that initiated the transaction.

</ResponseField>

<ResponseField name="externalOrderId" type="string">

  Order ID provided by the external entity that initiated the transaction.

</ResponseField>

<ResponseField name="externalTransactionId" type="string">

  Transaction ID provided by external entity.

</ResponseField>

<ResponseField name="providerTransactionId" type="string">

  Transaction ID provided by the payment provider.

</ResponseField>

<ResponseField name="metadata" type="object">

  Metadata stored when the transaction is created.

  Constraints: `format map`, `maxItems 16`.

  <Expandable title="metadata properties">

    <ResponseField name="key" type="string" />

    <ResponseField name="value" type="string" />

  </Expandable>

</ResponseField>

          Constraints: `minLength 1`, `maxLength 1000`.

      </Expandable>

    </ResponseField>

      <Expandable title="networkReference properties">
