<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Location ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Location name.

      Constraints: `maxLength 150`.

    </ResponseField>

    <ResponseField name="description" type="string">

      Location description.

    </ResponseField>

    <ResponseField name="default" type="boolean">

      Whether this is the default location. There can only be one default location per site. The default location can't be archived.

      **Read-only.**

    </ResponseField>

    <ResponseField name="status" type="string">

      Location status. Defaults to `ACTIVE`.
      __Notes:__
      - [Archiving a location](https://dev.wix.com/api/rest/business-info/locations/archive-location)
      doesn't affect the location's status.
      - `INACTIVE` status is currently not supported.

      **Supported values:**

      - `ACTIVE`
      - `INACTIVE`

    </ResponseField>

    <ResponseField name="locationType" type="string" deprecated>

      Location type.

      **Note:** Currently not supported.

      **Deprecated.**

      **Supported values:**

      - `BRANCH`
      - `OFFICES`
      - `RECEPTION`
      - `HEADQUARTERS`
      - `INVENTORY`

    </ResponseField>

    <ResponseField name="fax" type="string">

      Fax number.

    </ResponseField>

    <ResponseField name="timeZone" type="string">

      Timezone in `America/New_York` format.

    </ResponseField>

    <ResponseField name="email" type="string">

      Email address.

    </ResponseField>

    <ResponseField name="phone" type="string">

      Phone number.

    </ResponseField>

    <ResponseField name="address" type="Address">

      Address.

      <Expandable title="address properties">

        <ResponseField name="country" type="string">

          2-letter country code in an [ISO-3166 alpha-2](https://en.wikipedia.org/wiki/ISO_3166-1_alpha-2) format.

        </ResponseField>

        <ResponseField name="subdivision" type="string">

          Code for a subdivision (such as state, prefecture, or province) in [ISO 3166-2](https://en.wikipedia.org/wiki/ISO_3166-2) format.

        </ResponseField>

        <ResponseField name="city" type="string">

          City name.

        </ResponseField>

        <ResponseField name="postalCode" type="string">

          Postal or zip code.

        </ResponseField>

        <ResponseField name="streetAddress" type="StreetAddress">

          Street address. Includes street name, number, and apartment number in separate fields.

          <Expandable title="streetAddress properties">

            <ResponseField name="number" type="string">

              Street number.

            </ResponseField>

            <ResponseField name="name" type="string">

              Street name.

            </ResponseField>

            <ResponseField name="apt" type="string">

              Apartment number.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="formattedAddress" type="string">

          Full address of the location.

        </ResponseField>

        <ResponseField name="hint" type="string">

          Extra information that helps finding the location.

        </ResponseField>

        <ResponseField name="geocode" type="AddressLocation">

          Geographic coordinates of location.

          <Expandable title="geocode properties">

            <ResponseField name="latitude" type="number">

              Latitude of the location. Must be between -90 and 90.

            </ResponseField>

            <ResponseField name="longitude" type="number">

              Longitude of the location. Must be between -180 and 180.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="businessSchedule" type="BusinessSchedule">

      Business schedule. Array of weekly recurring time periods when the location is open for business. Limited to 100 time periods.

      __Note:__ Not supported by Wix Bookings.

      <Expandable title="businessSchedule properties">

        <ResponseField name="periods" type="TimePeriod[]">

          Weekly recurring time periods when the business is regularly open or the service is available. Limited to 100 time periods.

          Constraints: `maxItems 100`.

          <Expandable title="periods properties">

            <ResponseField name="openDay" type="string">

              Day of the week the period starts on.

              **Supported values:**

              - `MONDAY`
              - `TUESDAY`
              - `WEDNESDAY`
              - `THURSDAY`
              - `FRIDAY`
              - `SATURDAY`
              - `SUNDAY`

            </ResponseField>

            <ResponseField name="openTime" type="string">

              Time the period starts in 24-hour [ISO 8601](https://en.wikipedia.org/wiki/ISO_8601) extended format. Valid values are `00:00` to `24:00`, where `24:00` represents
              midnight at the end of the specified day.

            </ResponseField>

            <ResponseField name="closeDay" type="string">

              Day of the week the period ends on.

              **Supported values:**

              - `MONDAY`
              - `TUESDAY`
              - `WEDNESDAY`
              - `THURSDAY`
              - `FRIDAY`
              - `SATURDAY`
              - `SUNDAY`

            </ResponseField>

            <ResponseField name="closeTime" type="string">

              Time the period ends in 24-hour [ISO 8601](https://en.wikipedia.org/wiki/ISO_8601) extended format. Valid values are `00:00` to `24:00`, where `24:00` represents
              midnight at the end of the specified day.

              __Note:__ If `openDay` and `closeDay` specify the same day of the week `closeTime` must be later than `openTime`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="specialHourPeriod" type="SpecialHourPeriod[]">

          Exceptions to the business's regular hours. The business can be open or closed during the exception.

          Constraints: `maxItems 100`.

          <Expandable title="specialHourPeriod properties">

            <ResponseField name="startDate" type="string">

              Start date and time of the exception in [ISO 8601](https://en.wikipedia.org/wiki/ISO_8601) format and [Coordinated Universal Time (UTC)](https://en.wikipedia.org/wiki/Coordinated_Universal_Time).

            </ResponseField>

            <ResponseField name="endDate" type="string">

              End date and time of the exception in [ISO 8601](https://en.wikipedia.org/wiki/ISO_8601) format and [Coordinated Universal Time (UTC)](https://en.wikipedia.org/wiki/Coordinated_Universal_Time).

            </ResponseField>

            <ResponseField name="isClosed" type="boolean">

              Whether the business is closed (or the service is not available) during the exception.

              Default: `true`.

            </ResponseField>

            <ResponseField name="comment" type="string">

              Additional info about the exception. For example, "We close earlier on New Year's Eve."

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the location is updated.
      To prevent conflicting changes, the existing revision must be used when updating a location.

    </ResponseField>

    <ResponseField name="archived" type="boolean">

      Whether the location is archived. Archived locations can't be updated.
      __Note:__ [Archiving a location](https://dev.wix.com/api/rest/business-info/locations/archive-location)
      doesn't affect its `status`.

      **Read-only.**

    </ResponseField>

    <ResponseField name="locationTypes" type="string[]">

      Location types.

      Constraints: `maxItems 10`.

      **Supported values:**

      - `BRANCH`
      - `OFFICES`
      - `RECEPTION`
      - `HEADQUARTERS`
      - `INVENTORY`

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Extended fields for data extensions.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>
