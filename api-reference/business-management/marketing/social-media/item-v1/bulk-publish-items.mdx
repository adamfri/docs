    Publishes multiple items in a single call.

    Each item is published independently. To publish a single item, call Publish Item.

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.PROMOTE.MANAGE-SOCIAL-POSTS">Manage Social Posts</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    POST https://www.wixapis.com/social-publisher/v1/bulk/publish
    ```

    ## Body Params

    <ParamField body="items" type="Item[]" required>

      Items to publish.

      Constraints: `minItems 1`, `maxItems 20`.

      See the [Item Object](/api-reference/business-management/marketing/social-media/item-v1/item-object) for this object's fields.

    </ParamField>

    <ParamField body="policy" type="string">

      Policy that determines how items are published when one of them fails.

      **Supported values:**

      - `CONCURRENT` — Publish all items, regardless of whether individual items fail.
      - `SEQUENTIAL_STOP_ON_FAIL` — Stop publishing remaining items as soon as one item fails.

    </ParamField>

    <ParamField body="schedulingInfo" type="SchedulingInfo">

      Scheduling info applied to all items. If set, the items are scheduled instead of published immediately.

      <Expandable title="schedulingInfo properties">

        <ParamField body="scheduledDate" type="datetime">

          Date and time the item is scheduled to be published.

        </ParamField>

        <ParamField body="schedulingTimestamp" type="datetime">

          Date and time the item was scheduled.

          **Read-only.**

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="itemExtras" type="ItemExtras">

      itemExtras

      <Expandable title="itemExtras properties">

        <ParamField body="userOrigin" type="string">

          Where the user began the flow to open the publisher. For example, `stores.product-list`.

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField body="returnEntity" type="boolean">

      Whether to return the published items in the response.

      Default: `false`

    </ParamField>

    ## Response Object

    <ResponseField name="results" type="BulkItemsResult[]">

      Results of the bulk publish, one per item, in the order they were sent.

      Constraints: `minItems 1`, `maxItems 20`.

      <Expandable title="results properties">

        <ResponseField name="itemMetadata" type="ItemMetadata">

          Metadata for the individual item's publish result.

          <Expandable title="itemMetadata properties">

            <ResponseField name="id" type="string">

              Item ID. Should always be available, unless it's impossible (for example, when failing to create an item).

            </ResponseField>

            <ResponseField name="originalIndex" type="integer">

              Index of the item within the request array. Allows for correlation between request and response items.

            </ResponseField>

            <ResponseField name="success" type="boolean">

              Whether the requested action was successful for this item. When `false`, the `error` field is populated.

            </ResponseField>

            <ResponseField name="error" type="ApplicationError">

              Details about the error in case of failure.

              _Nested structure omitted for brevity — see the `ApplicationError` object._

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="item" type="Item">

          Published item. Returned only when `returnEntity` is `true` in the request.

          See the [Item Object](/api-reference/business-management/marketing/social-media/item-v1/item-object) for this object's fields.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="bulkActionMetadata" type="BulkActionMetadata">

      Bulk action metadata.

      <Expandable title="bulkActionMetadata properties">

        <ResponseField name="totalSuccesses" type="integer">

          Number of items that were successfully processed.

        </ResponseField>

        <ResponseField name="totalFailures" type="integer">

          Number of items that couldn't be processed.

        </ResponseField>

        <ResponseField name="undetailedFailures" type="integer">

          Number of failures without details because detailed failure threshold was exceeded.

        </ResponseField>

      </Expandable>

    </ResponseField>

    ## Errors

    | Status | Code | Application code | Description |
    | --- | --- | --- | --- |
    | `428` | `FAILED_PRECONDITION` | `INELIGIBLE_FOR_FEATURE` | The site's social media marketing plan doesn't permit this action, or its quota for the current period is exhausted. |

    <Note>

      This method may also return standard errors. [Learn more about standard Wix errors](https://dev.wix.com/docs/rest/articles/getting-started/errors).

    </Note>

    import { items } from "@wix/promote-growth-tools-publisher";

    const itemsToPublish = [
      {
        channel: {

          accountId: "UCa8fkLm3QxR7vTq2NpZ9wEd",
        },
        type: items.ItemType.VIDEO,
        youtubeVideo: {
          channelId: "UCa8fkLm3QxR7vTq2NpZ9wEd",

          title: "Glazing techniques for beginners",
          description: "Our three favorite glazing techniques, explained.",
        },
      },
    ];

    const options = {
      policy: items.BulkPublishItemsPolicy.CONCURRENT,
      schedulingInfo: {
        scheduledDate: new Date("2026-07-20T12:00:00.000Z"),
      },
      returnEntity: true,
    };

    async function bulkPublishItems() {
      const response = await items.bulkPublishItems(itemsToPublish, options);
    }
