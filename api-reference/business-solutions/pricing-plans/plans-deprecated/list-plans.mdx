    <Warning>

      **Deprecated.** This method is deprecated.

    </Warning>

    Retrieves a list of up to 100 pricing plans (including public, hidden, and archived plans).

    <Note>

      **Authentication**

      You can only call this method when authenticated as a Wix app or Wix user [identity](https://dev.wix.com/docs/api-reference/articles/authentication/about-identities).

    </Note>

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-PAIDPLANS.READ-PLANS">Read Pricing Plans</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`.

    ## Endpoint

    ```http
    GET https://www.wixapis.com/pricing-plans/v2/plans
    ```

    ## Query Params

    <ParamField query="archived" type="string">

      Archived filter. Defaults to `ACTIVE` (not archived) only.

      **Supported values:**

      - `ACTIVE` — Returns all plans that are active.
      - `ARCHIVED` — Returns all plans that are archived.
      - `ARCHIVED_AND_ACTIVE` — Returns all plans that are active and archived.

    </ParamField>

    <ParamField query="public" type="string">

      Visibility filter. Defaults to `PUBLIC_AND_HIDDEN` (meaning, both are listed).

      **Supported values:**

      - `PUBLIC_AND_HIDDEN` — Returns all public and hidden plans.
      - `PUBLIC` — Returns only public plans.
      - `HIDDEN` — Returns only hidden plans.

    </ParamField>

    <ParamField query="limit" type="integer">

      Number of pricing plans to list. Defaults to `75`.

    </ParamField>

    <ParamField query="offset" type="integer">

      Number of pricing plans to skip. Defaults to `0`.

    </ParamField>

    <ParamField query="planIds" type="string[]">

      IDs of plans to list. If non-existent IDs are specified, they are ignored and don't cause errors. If no IDs are specified, all public and hidden plans are listed according to the order displayed in a site's dashboard (which you can set with Arrange Plans). You can specify a maximum of 100 IDs.

      Constraints: `format GUID`, `maxItems 100`.

    </ParamField>

    ## Response Object

    <ResponseField name="plans" type="Plan[]">

      List of all public and hidden pricing plans.

      See the [Plan Object](/api-reference/business-solutions/pricing-plans/plans-deprecated/plan-object) for this object's fields.

    </ResponseField>

    <ResponseField name="pagingMetadata" type="PagingMetadataV2">

      Object containing paging-related data (such as the number of plans returned, the offset, and so on).

      <Expandable title="pagingMetadata properties">

        <ResponseField name="count" type="integer">

          Number of items returned in the response.

        </ResponseField>

        <ResponseField name="offset" type="integer">

          Offset that was requested.

        </ResponseField>

        <ResponseField name="total" type="integer">

          Total number of items that match the query. Returned if offset paging is used and the `tooManyToCount` flag is not set.

        </ResponseField>

        <ResponseField name="tooManyToCount" type="boolean">

          Flag that indicates the server failed to calculate the `total` field.

        </ResponseField>

        <ResponseField name="cursors" type="Cursors">

          Cursors to navigate through the result pages using `next` and `prev`. Returned if cursor paging is used.

          <Expandable title="cursors properties">

            <ResponseField name="next" type="string">

              Cursor string pointing to the next page in the list of results.

            </ResponseField>

            <ResponseField name="prev" type="string">

              Cursor pointing to the previous page in the list of results.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    /*******************************************
     * Backend code - plan-functions.web.js/ts *

    import { Permissions, webMethod } from '@wix/web-methods';
    import { plans } from '@wix/pricing-plans';
    import { auth } from '@wix/essentials';

    const elevatedListPlans = auth.elevate(plans.listPlans);
    const elevatedGetPlan = auth.elevate(plans.getPlan);
    const elevatedCreatePlan = auth.elevate(plans.createPlan);

    export const listPlans = webMethod(Permissions.Anyone, async () => {
      try {
        const response = await elevatedListPlans();
        const allPlans = response.plans;

        return allPlans;
      } catch (error) {
        console.error(error);
        // Handle the error
      }
    });

    export const getPlan = webMethod(Permissions.Anyone, async (planId) => {
      try {
        const selectedPlan = await elevatedGetPlan(planId);

        return selectedPlan;
      } catch (error) {
        console.error(error);
        // Handle the error
      }
    });

    export const createPlan = webMethod(Permissions.Anyone, async (plan) => {
      try {
        const newPlan = await elevatedCreatePlan(plan);

        return newPlan;
      } catch (error) {
        console.error(error);
        // Handle the error
      }
    });

    /*************
     * Page code *
     *************/

    import { listPlans, getPlan, createPlan } from 'backend/plan-functions.web';
    $w.onReady(async function () {
      $w('#plansDropdown').disable();
      $w('#clonePlanBtn').disable();
      populatePlansDropdown();

      let selectedPlan;

      $w('#plansDropdown').onChange(async () => {
        const planId = $w('#plansDropdown').value;
        selectedPlan = await getPlan(planId);

        // Set default value for the new plan name
        $w('#planName').value = `${selectedPlan.name}-duplicate`
        $w('#clonePlanBtn').enable();
      });

      $w('#clonePlanBtn').onClick(async () => {
        const newPlan = selectedPlan;
        // Use inputs to set the new plan name and price
        newPlan.name = $w('#planName').value;
        newPlan.pricing.price.value = $w('#price').value;

        await createPlan(newPlan);
      });
    });

    async function populatePlansDropdown() {
      const plans = await listPlans();
      $w('#plansDropdown').options = plans.map((plan) => {
        return {
          label: plan.name,
          value: plan._id
        }
      });
      $w('#plansDropdown').enable();
    }
