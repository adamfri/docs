<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Job ID.

      **Read-only.** Constraints: `format GUID`.

    </ResponseField>

    <ResponseField name="metadata" type="object">

      Optional job metadata, used to store any parameters used during job execution, user-defined statuses, etc.

    </ResponseField>

    <ResponseField name="status" type="string">

      Current job execution status.

      **Supported values:**

      - `INITIALIZED` — Job is created, but hasn't started yet.
      - `PROCESSING` — Job has started and is in progress.
      - `FINISHED` — Job is finished.
      - `FAILED` — Job has failed.

    </ResponseField>

    <ResponseField name="counters" type="Counters" deprecated>

      Current counters.

      **Deprecated.** Current counters. Use `counts` instead. Removal: 2024-12-12.

      <Expandable title="counters properties">

        <ResponseField name="itemsToProcess" type="integer">

          Optional count of the dataset size, as specified during job creation. Can be used for progress bars, etc.

        </ResponseField>

        <ResponseField name="itemsSucceeded" type="integer">

          Amount of items that were successfully processed.

        </ResponseField>

        <ResponseField name="itemsFailed" type="integer">

          Amount of items whose processing failed.

        </ResponseField>

        <ResponseField name="errorCountByCode" type="object">

          Mapping between an error code and the number of failures associated with it.

          Constraints: `format map`.

          <Expandable title="errorCountByCode properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="integer" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the job was created in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#combined_date_and_time_representations) format.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the job was last updated in [ISO-8601](https://en.wikipedia.org/wiki/ISO_8601#combined_date_and_time_representations) format.

      **Read-only.**

    </ResponseField>

    <ResponseField name="owner" type="Owner" deprecated>

      Job owner.

      **Read-only.** **Deprecated.** Job owner. Use `created_by` instead. Removal: 2024-12-12.

      <Expandable title="One of the following">

        <ResponseField name="id" type="id">

          id

          <Expandable title="id properties">

            <ResponseField name="anonymousVisitorId" type="string">

              Visitor ID - when the job is owned by a site visitor that was **not** logged in.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="memberId" type="string">

              Member ID - When the job is owned by a by a **logged in** site visitor.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="wixUserId" type="string">

              User ID - When the job is owned by a Wix user.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="appId" type="string">

              App ID - When the job is owned by an external app or app by Wix.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="serviceId" type="string">

              Service ID.

              Constraints: `format GUID`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="createdBy" type="CreatedBy">

      Job creator.

      **Read-only.**

      <Expandable title="createdBy properties">

        <ResponseField name="type" type="string">

          The type of the creator.

          **Supported values:**

          - `UNKNOWN_TYPE`
          - `ANONYMOUS_VISITOR_ID` — Job was created by a site visitor that was **not** logged in.
          - `MEMBER_ID` — Job was created by a **logged in** site visitor.
          - `WIX_USER_ID` — Job was created by a  Wix user.
          - `APP_ID` — Job was created by an external app or app by Wix.

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="options" type="options">

          <Expandable title="options properties">

            <ResponseField name="memberIdOptions" type="MemberIdOptions">

              <Expandable title="memberIdOptions properties">

                <ResponseField name="memberId" type="string">

                  Member ID - When the job is owned by a by a **logged in** site visitor.

                  Constraints: `format GUID`.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="wixUserIdOptions" type="WixUserIdOptions">

              User ID - When the job is created by a Wix user.

              <Expandable title="wixUserIdOptions properties">

                <ResponseField name="wixUserId" type="string">

                  User ID - When the job is owned by a site owner or contributor.

                  Constraints: `format GUID`.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="appIdOptions" type="AppIdOptions">

              App ID - When the job is created by an external app or app by Wix.

              <Expandable title="appIdOptions properties">

            </ResponseField>

        </ResponseField>

      </Expandable>

    </ResponseField>

      Current execution counts.

      <Expandable title="counts properties">

        <ResponseField name="total" type="integer">

          Optional count of the dataset size, specified during job creation. Can be used for progress bars, etc.

        </ResponseField>

        <ResponseField name="successCount" type="integer">

          Amount of items that were successfully processed.

        </ResponseField>

        <ResponseField name="failCount" type="integer">

          Amount of items whose processing failed.

        </ResponseField>

        <ResponseField name="errorByCodeCount" type="object">

          Constraints: `format map`.

          <Expandable title="errorByCodeCount properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="integer" />

        </ResponseField>

      </Expandable>

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "job": {
          "id": "076e0505-6944-4645-a248-80b197ee9cfd",
          "metadata": {
            "meta1": "value1"
          },
          "status": "INITIALIZED",
          "counters": {
            "items_to_process": null,
            "items_succeeded": 0,
            "items_failed": 0,
            "error_count_by_code": {}
          },
          "created_date": "2024-04-22T10:23:25.798Z",
          "updated_date": "2024-04-22T10:23:25.798Z",
          "owner": {
            "service_id": "338acb04-5b9b-4cdd-beb3-ea6739546d3a"
          }
        }
      }
      ```
