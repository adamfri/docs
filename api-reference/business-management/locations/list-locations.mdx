    Retrieves locations, given the specified filters, sorting, and paging.

    ## Permissions

    This method requires one of the following permission scopes:

    <div className="wix-scopes">

    <span className="wix-scope" title="SCOPE.DC-MULTILOCATION.READ-LOCATIONS">Read Locations</span>

    </div>

    Learn more about [app permissions](https://dev.wix.com/docs/build-apps/develop-your-app/access/authorization/about-permissions).

    Callable by: `APP`, `MEMBER`, `VISITOR`.

    ## Endpoint

    ```http
    GET https://www.wixapis.com/locations/v1/locations
    ```

    ## Query Params

    <ParamField query="sort" type="Sorting">

      Sort order.

      <Expandable title="sort properties">

        <ParamField query="fieldName" type="string">

          Name of the field to sort by.

          Constraints: `maxLength 512`.

        </ParamField>

        <ParamField query="order" type="string">

          Sort order.

          **Supported values:**

          - `ASC`
          - `DESC`

        </ParamField>

        <ParamField query="origin" type="AddressLocation">

          Origin point for geo-distance sorting on a GEO field
          results are ordered by distance from this point (ASC = nearest first, DESC = farthest first).

          <Expandable title="origin properties">

            <ParamField query="latitude" type="number">

              Address latitude.

            </ParamField>

            <ParamField query="longitude" type="number">

              Address longitude.

            </ParamField>

          </Expandable>

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField query="paging" type="Paging">

      Pagination.

      Default values:
      `offset`: 0
      `limit`: 50 (Max: 1000)

      <Expandable title="paging properties">

        <ParamField query="limit" type="integer">

          Number of items to load.

        </ParamField>

        <ParamField query="offset" type="integer">

          Number of items to skip in the current sort order.

        </ParamField>

      </Expandable>

    </ParamField>

    <ParamField query="includeArchived" type="boolean">

      Whether to include `archived` locations in the response.

      Default: `false`

    </ParamField>

    <ParamField query="filterAuthorizedLocationEntities" type="boolean">

      Whether to filter only authorized locations

    </ParamField>

    ## Response Object

    <ResponseField name="locations" type="Location[]">

      Retrieved locations.

      See the [Location Object](/api-reference/business-management/locations/location-object) for this object's fields.

    </ResponseField>

    <ResponseField name="pagingMetadata" type="PagingMetadata">

      Paging info.

      <Expandable title="pagingMetadata properties">

        <ResponseField name="count" type="integer">

          Number of items returned in the response.

        </ResponseField>

        <ResponseField name="offset" type="integer">

          Offset that was requested.

        </ResponseField>

        <ResponseField name="hasNext" type="boolean">

          Indicates if there are more results after the current page.
          If `true`, another page of results can be retrieved.
          If `false`, this is the last page.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="authorizedLocationEntities" type="string[]">

      Ids of the locations that the requesting entity can manage

      Constraints: `format GUID`, `minItems 0`, `maxItems 100`.

    </ResponseField>

    <Note>

    /*********************************************
     * Backend code - archive-location.web.js/ts *
     ********************************************/

    import { Permissions, webMethod } from '@wix/web-methods';
    import { locations } from '@wix/business-tools';
    import { auth } from '@wix/essentials';

    export const archiveLocationById = webMethod(Permissions.Anyone, async (locationId) => {
      try {
        const elevatedArchiveLocation = auth.elevate(locations.archiveLocation);
        const archivedLocation = await elevatedArchiveLocation(locationId);

        return archivedLocation;
      } catch (error) {
        console.error(error);
        throw new Error(error);
      }
    });

    export const listLocations = webMethod(Permissions.Anyone, async () => {
      try {
        const elevatedListLocations = auth.elevate(locations.listLocations);
        const results = await elevatedListLocations();

        return results.locations;
      } catch (error) {
        console.error(error);
        throw new Error(error);
      }
    });

    /*************
     * Page code *
     ************/

    import { archiveLocationById, listLocations } from 'backend/archive-location.web';

    $w.onReady(async () => {
      await populateStoresDropdown();

      $w('#archiveLocationBtn').onClick(async () => {
        const locationId = $w('#locationsDropdown').value;
        const archivedLocation = await archiveLocationById(locationId);

        console.log('The following location has been archived', archivedLocation);
        $w('#archivedMessage').show();
      });
    });

    async function populateStoresDropdown() {
      const locations = await listLocations();
      const dropdownOptions = locations.map((location) => {
        return {
          label: location.name,
          value: location._id
        }
      });

      $w('#locationsDropdown').options = dropdownOptions;
    }
