<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Message ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="content" type="MessageContent">

      Structured message content.

      <Expandable title="content properties">

        <ResponseField name="previewText" type="string">

          Summary of message contents.
          Displayed in Inbox in the
          [Message List](https://support.wix.com/en/article/wix-inbox-an-overview).

        </ResponseField>

        <ResponseField name="title" type="string">

          Message title.

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="payload" type="payload">

          payload

          <Expandable title="payload properties">

            <ResponseField name="basic" type="BasicMessagePayload">

              Plain text, file, or image message.

              <Expandable title="basic properties">

                <ResponseField name="items" type="BasicMessageData[]">

                  List of plain text messages, images, and/or files.
                  List items are displayed as separate messages
                  in Inbox and the site's chat widget.

                  Constraints: `minItems 1`.

                  _Nested structure omitted for brevity — see the `BasicMessageData[]` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="template" type="TemplateMessagePayload">

              Template containing an image, title, text,
              and/or up to 10 call-to-action buttons.

              <Expandable title="template properties">

                <ResponseField name="title" type="string">

                  Title displayed in the message.

                </ResponseField>

                <ResponseField name="buttons" type="Button[]">

                  List of buttons to display with the message.
                  Buttons can either open a URL or pass interaction details to the
                  Button Interacted event.

                  Maximum: 10 buttons.

                  Constraints: `maxItems 10`.

                  _Nested structure omitted for brevity — see the `Button[]` object._

                </ResponseField>

                <ResponseField name="orientation" type="string">

                  Controls whether the message is displayed in portrait or landscape layout.

                  Default: `PORTRAIT`.

                  **Supported values:**

                  - `UNKNOWN_ORIENTATION` — Unknown orientation.
                  - `PORTRAIT` — Portrait layout.
                  - `LANDSCAPE` — Landscape layout.

                </ResponseField>

                <ResponseField name="textLines" type="string[]">

                  List of lines of text. Each item in the array is displayed on a new line.

                  Max: 10 lines (up to 512 characters per line).

                  Constraints: `maxItems 10`.

                </ResponseField>

                <ResponseField name="imageUrl" type="string">

                  URL where the icon is hosted.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="minimal" type="MinimalMessagePayload">

              Minimal message containing a single line of text
              and an optional icon.
              Often reports an activity that took place.

              <Expandable title="minimal properties">

                <ResponseField name="text" type="string">

                  Message text.

                  Constraints: `minLength 1`, `maxLength 16384`.

                </ResponseField>

                <ResponseField name="iconUrl" type="string">

                  URL where the icon is hosted.

                </ResponseField>

                <ResponseField name="url" type="string">

                  If included, URL the user is redirected to when clicking the message.

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="form" type="FormMessagePayload">

              Message containing submitted form data.
              Typically sent with a `direction` of `PARTICIPANT_TO_BUSINESS`
              and message `visibility` of `BUSINESS`.

              <Expandable title="form properties">

                <ResponseField name="title" type="string">

                  Form title displayed in the message.

                </ResponseField>

                <ResponseField name="description" type="string">

                  Form description displayed below the title.

                </ResponseField>

                <ResponseField name="fields" type="FormField[]">

                  List of form fields and values.

                  Constraints: `minItems 1`, `maxItems 30`.

                  _Nested structure omitted for brevity — see the `FormField[]` object._

                </ResponseField>

                <ResponseField name="media" type="MediaItem[]">

                  List of files and/or images attached to the form.

                  List items are displayed as separate messages
                  in Inbox and the site's chat widget.

                  Constraints: `maxItems 15`.

                  _Nested structure omitted for brevity — see the `MediaItem[]` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="system" type="SystemMessagePayload">

              System message.
              For internal use.

              <Expandable title="system properties">

                <ResponseField name="text" type="string">

                  Text message. `\n` renders as a line break.

                  Constraints: `maxLength 16384`.

                </ResponseField>

                <ResponseField name="buttons" type="Button[]">

                  List of buttons to display with the message.
                  Buttons can either open a URL or pass interaction details to the
                  Button Interacted event.

                  Maximum: 10 buttons.

                  Constraints: `maxItems 10`.

                  _Nested structure omitted for brevity — see the `Button[]` object._

                </ResponseField>

                <ResponseField name="imageUrl" type="string">

                  URL where the icon is hosted.

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="sender" type="IdentificationData">

      ID of the message sender.

      Defaults to the caller's ID
      using the property that matches their identity type.
      For 3rd-party apps, defaults to using the app's `appId`.

      You can override the default behavior when calling
      the Send Message method
      by using the `sendAs` parameter.

      **Read-only.**

      <Expandable title="sender properties">

        <ResponseField name="contactId" type="string">

          Contact ID. See the Contacts API for more information.

          **Read-only.**

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="id" type="id">

          id

          <Expandable title="id properties">

            <ResponseField name="anonymousVisitorId" type="string">

              Anonymous site visitor ID.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="memberId" type="string">

              Member ID. See the Members API for more information.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="wixUserId" type="string">

              ID of the Wix user.

              Constraints: `format GUID`.

            </ResponseField>

            <ResponseField name="appId" type="string">

              App ID.

              Constraints: `format GUID`.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="badges" type="Badge[]">

      Optional icon and short text providing additional details about the message,
      such as the app from which the message was sent, or whether the message was an automated response.

      Currently only 1 badge is supported.

      Constraints: `maxItems 1`.

      <Expandable title="badges properties">

        <ResponseField name="text" type="string">

          Display text.

        </ResponseField>

        <ResponseField name="iconUrl" type="string">

          URL where the icon is hosted.

        </ResponseField>

        <ResponseField name="badgeVisibility" type="string">

          Controls whether the badge is visible to the participant.
          Currently not supported in the Chat widget.

          **Supported values:**

          - `UNKNOWN_BADGE_VISIBILITY` — Unkown visibility.
          - `BUSINESS_AND_PARTICIPANT` — Visible to the participant and site collaborators.
          - `BUSINESS` — Visible to site collaborators only.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="targetChannels" type="string[]" deprecated>

      Communication channel to send the message to.

      Currently messages can be sent to 1 channel only.

      **Deprecated.** Communication channel to send the message to.

      Currently messages can be sent to 1 channel only. Use `target_channel_ids` instead. Removal: 2026-06-30.

      **Supported values:**

      - `UNKNOWN_CHANNEL_TYPE`
      - `CHAT`
      - `EMAIL`
      - `SMS`
      - `FACEBOOK`
      - `INSTAGRAM`
      - `WHATSAPP`

    </ResponseField>

    <ResponseField name="sourceChannel" type="string" deprecated>

      Communication channel the message is sent from.

      **Deprecated.** Communication channel the message is sent from. Use `source_channel_id` instead. Removal: 2026-06-30.

      **Supported values:**

      - `UNKNOWN_CHANNEL_TYPE`
      - `CHAT`
      - `EMAIL`
      - `SMS`
      - `FACEBOOK`
      - `INSTAGRAM`
      - `WHATSAPP`

    </ResponseField>

    <ResponseField name="appId" type="string">

      ID of the app that sent the message.

      Omitted if the message was sent with the Chat widget.

      **Read-only.**

    </ResponseField>

    <ResponseField name="visibility" type="string">

      __Required.__
      Controls who can see the message.

      **Supported values:**

      - `UNKNOWN_VISIBILITY` — Unknown message visibility.
      - `BUSINESS_AND_PARTICIPANT` — Visible to the participant and site collaborators.
      - `BUSINESS` — Visible to site collaborators only.

    </ResponseField>

    <ResponseField name="sequence" type="integer">

      Sequential ordering of the message.

      Ensures more accurate sorting than `createdDate`
      if two messages are sent at the same time.

      **Read-only.**

    </ResponseField>

    <ResponseField name="direction" type="string">

      __Required.__
      Message direction.

      **Supported values:**

      - `UNKNOWN_DIRECTION` — Unknown message direction.
      - `BUSINESS_TO_PARTICIPANT` — The message was sent from the business to the participant.
      - `PARTICIPANT_TO_BUSINESS` — The message was sent from the participant to the business.

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the message was sent.

    </ResponseField>

    <ResponseField name="targetChannelIds" type="string[]">

      Target channel IDs to send the message to.

      Constraints: `maxLength 100`, `maxItems 100`.

    </ResponseField>

    <ResponseField name="sourceChannelId" type="string">

      The channel ID from which the message was sent.

    </ResponseField>

    <ResponseField name="silent" type="boolean">

      Controls whether the unread count and conversation summary
      are updated in the
      [Message List](https://support.wix.com/en/article/wix-inbox-an-overview).

      If `true`, unread count and conversation summary are not updated.

      Default: `false`

    </ResponseField>

  </Tab>
