title: "Coupon Object"
description: "The Coupon object represents all information available for a coupon including its basic definition and runtime information."
icon: "cube"
---

The `Coupon` object represents all information available for a coupon
including its basic definition and runtime information.

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Coupon ID.

    </ResponseField>

    <ResponseField name="specification" type="Specification">

      Basic coupon info.

      <Expandable title="specification properties">

        <ResponseField name="name" type="string">

          Name of coupon.

        </ResponseField>

        <ResponseField name="code" type="string">

          Coupon code. Must be unique for all coupons on your site.

          Max: 20 characters

        </ResponseField>

        <ResponseField name="startTime" type="integer">

          Coupon valid from this date and time, in milliseconds.
          For example, `"1554066000000"`.

        </ResponseField>

        <ResponseField name="expirationTime" type="integer">

          Coupon expires at this date and time, in milliseconds.
          For example, `"1554325199999"`.

        </ResponseField>

        <ResponseField name="usageLimit" type="integer">

          Maximum number of times the coupon can be used.

          >**Note:** Multiple purchases by the same customer or purchases by different customers are both counted toward `usageLimit`.

        </ResponseField>

        <ResponseField name="limitPerCustomer" type="integer">

          Maximum number of times the coupon can be used per customer.

        </ResponseField>

        <ResponseField name="limitedToOneItem" type="boolean">

          Whether the coupon is limited to one item.

          If `true` and a customer pays for multiple items, the discount applies to only the lowest priced item.
          Coupons with a `bookings` `scope.namespace` are always limited to one item.

        </ResponseField>

        <ResponseField name="appliesToSubscriptions" type="boolean">

          Whether the coupon applies to subscription products.

        </ResponseField>

        <ResponseField name="discountedCycleCount" type="integer">

          Specifies the amount of discounted cycles for a subscription item.

          + Can only be set when `scope.namespace = pricingPlans`.
          + If `discountedCycleCount` is empty, the coupon applies to all available cycles.
          + `discountedCycleCount` is ignored if `appliesToSubscriptions = true`.

          Max: `999`

        </ResponseField>

        <ResponseField name="active" type="boolean">

          Whether the coupon is currently [active](https://support.wix.com/en/article/wix-stores-activating-and-deactivating-coupons).

          Default: `true`

        </ResponseField>

        <ResponseField name="type" type="string">

          Coupon type. Read-only.

        </ResponseField>

        <ResponseField name="tags" type="string[]">

          Tags for categorizing the coupon.

          Constraints: `maxLength 255`, `maxItems 100`.

        </ResponseField>

      </Expandable>

      <Expandable title="One of the following">

        <ResponseField name="ScopeOrMinSubtotal" type="ScopeOrMinSubtotal">

          ScopeOrMinSubtotal

          <Expandable title="ScopeOrMinSubtotal properties">

            <ResponseField name="scope" type="Scope">

              Specifies the type of line items this coupon will apply to. See [valid scope values](https://dev.wix.com/api/rest/coupons/coupons/valid-scope-values).

              <Expandable title="scope properties">

                <ResponseField name="namespace" type="string">

                  Scope namespace (Wix Stores, Wix Bookings, Wix Events, Wix Pricing Plans)

                </ResponseField>

                <ResponseField name="group" type="Group">

                  Coupon scope's applied group (e.g., event or ticket in Wix Events)

                  _Nested structure omitted for brevity — see the `Group` object._

                </ResponseField>

              </Expandable>

            </ResponseField>

            <ResponseField name="minimumSubtotal" type="number">

              Limit the coupon to carts with a subtotal above this number.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="Behavior" type="Behavior">

          Behavior

          <Expandable title="Behavior properties">

            <ResponseField name="moneyOffAmount" type="number">

              Discount as a fixed amount.

            </ResponseField>

            <ResponseField name="percentOffRate" type="number">

              Discount as a percentage.

            </ResponseField>

            <ResponseField name="freeShipping" type="boolean">

              Free shipping.

              If `true`, the coupon applies to all items in all `namespaces`.

            </ResponseField>

            <ResponseField name="fixedPriceAmount" type="number">

              Fixed sale price.

            </ResponseField>

            <ResponseField name="buyXGetY" type="BuyXGetY">

              Receive free products when making a purchase.

              For example, purchase `x` number of products and receive `y` number of products for free.

              <Expandable title="buyXGetY properties">

                <ResponseField name="x" type="integer">

                  Number of purchased items required to receive free items.

                </ResponseField>

                <ResponseField name="y" type="integer">

                  Number of items received for free if required number of items were purchased.

                </ResponseField>

              </Expandable>

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="dateCreated" type="string">

      Time the coupon was created (UNIX Epoch time in milliseconds).

    </ResponseField>

    <ResponseField name="expired" type="boolean">

      Whether the coupon is expired.

    </ResponseField>

    <ResponseField name="numberOfUsages" type="integer">

      How many times this coupon has been used.

    </ResponseField>

    <ResponseField name="displayData" type="DisplayData">

      Coupon display information.

      <Expandable title="displayData properties">

        <ResponseField name="name" type="string">

          Coupon name to be displayed.

        </ResponseField>

        <ResponseField name="mediaItem" type="MediaItem">

          Displayed media item information.

          <Expandable title="mediaItem properties">

            <ResponseField name="url" type="string">

              Media item URL.

              Constraints: `format WEB_URL`.

            </ResponseField>

            <ResponseField name="width" type="integer">

              Media item width.

              Constraints: `min 0`, `max 5000`.

            </ResponseField>

            <ResponseField name="height" type="integer">

              Media item height.

              Constraints: `min 0`, `max 5000`.

            </ResponseField>

          </Expandable>

        </ResponseField>

        <ResponseField name="formattedPrice" type="string">

          Formatted price for display.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="appId" type="string">

      ID of the app that created the coupon. Empty if created by the site owner.

      **Read-only.**

    </ResponseField>

  </Tab>
