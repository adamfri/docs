---
title: "Bulk Job Object"
description: "The Bulk Job object in the Wix Bulk Job API."
icon: "cube"
---

<ResponseField name="id" type="string">

  Bulk job ID.

  **Read-only.** Constraints: `format GUID`.

</ResponseField>

<ResponseField name="type" type="string">

  Bulk job type.

  **Supported values:**

  - `UNKNOWN_TYPE`
  - `DELETE`
  - `UPDATE`
  - `ADD_LABELS`
  - `REMOVE_LABELS`
  - `ADD_REMOVE_LABELS`

</ResponseField>

<ResponseField name="status" type="string">

  Status of the bulk job.

  **Supported values:**

  - `UNKNOWN_STATUS`
  - `NEW`
  - `LOADING`
  - `RUNNING`
  - `COMPLETED`
  - `INDEXING`

</ResponseField>

<ResponseField name="startedDate" type="datetime">

  Date the bulk job started.

</ResponseField>

<ResponseField name="finishedDate" type="datetime">

  Date the bulk job finished.

</ResponseField>

<ResponseField name="totalItems" type="integer">

  Number of contacts that met the bulk job's filter criteria.

</ResponseField>

<ResponseField name="successTotal" type="integer">

  Number of contacts that were successfully processed.

</ResponseField>

<ResponseField name="failedTotal" type="integer">

  Number of contacts that couldn't be processed.

</ResponseField>

<ResponseField name="errors" type="Error[]">

  List of errors encountered.

  <Expandable title="errors properties">

    <ResponseField name="count" type="integer">

      Number of contacts with this error.

    </ResponseField>

    <ResponseField name="code" type="string">

      Error code.

    </ResponseField>

    <ResponseField name="message" type="string">

      Error details.

    </ResponseField>

  </Expandable>

</ResponseField>
