---
title: "Item Modifier Group Object"
description: "The Item Modifier Group object in the Wix Item Modifier Group API."
icon: "cube"
---

<Tabs>

  <Tab title="REST">

    <ResponseField name="id" type="string">

      Modifier group ID.

      **Read-only.**

    </ResponseField>

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the modifier group is updated. To prevent conflicting changes, the current revision must be passed when updating the modifier group. Ignored when creating a modifier group.

      **Read-only.**

    </ResponseField>

    <ResponseField name="createdDate" type="datetime">

      Date and time the modifier group was created.

      **Read-only.**

    </ResponseField>

    <ResponseField name="updatedDate" type="datetime">

      Date and time the modifier group was updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="name" type="string">

      Modifier group name.

    </ResponseField>

    <ResponseField name="modifiers" type="Modifier[]">

      Group of item modifiers.

      Constraints: `minItems 0`, `maxItems 500`.

      <Expandable title="modifiers properties">

        <ResponseField name="id" type="string">

          Item modifier ID.

          Constraints: `format GUID`.

        </ResponseField>

        <ResponseField name="preSelected" type="boolean">

          Whether the item modifier is pre-selected.
          Default: `false`.

        </ResponseField>

        <ResponseField name="additionalChargeInfo" type="AdditionalChargeInfo">

          Item modifier price details.

          <Expandable title="additionalChargeInfo properties">

            <ResponseField name="additionalCharge" type="string">

              Additional charge for the item modifier. A value of `0` means the item modifier is free.

            </ResponseField>

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="rule" type="Rule">

      Modifier group details.

      <Expandable title="rule properties">

        <ResponseField name="required" type="boolean">

          Whether the items from the modifier group must be selected.

        </ResponseField>

        <ResponseField name="minSelections" type="integer">

          Minimum number of item modifiers a site visitor must select. The value must be lower or equal to the available item modifiers in the group.
          Default: `0`.

        </ResponseField>

        <ResponseField name="maxSelections" type="integer">

          Maximum number of item modifiers a site visitor may select. Must be greater than or equal to the value of `minSelections`.

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="extendedFields" type="ExtendedFields">

      Extended fields.

      <Expandable title="extendedFields properties">

        <ResponseField name="namespaces" type="object">

          Extended field data. Each key corresponds to the namespace of the app that created the extended fields.
          The value of each key is structured according to the schema defined when the extended fields were configured.

          You can only access fields for which you have the appropriate permissions.

          Learn more about [extended fields](https://dev.wix.com/docs/rest/articles/getting-started/extended-fields).

          Constraints: `format map`.

          <Expandable title="namespaces properties">

            <ResponseField name="key" type="string" />

            <ResponseField name="value" type="object" />

          </Expandable>

        </ResponseField>

      </Expandable>

    </ResponseField>

    <ResponseField name="businessLocationIds" type="string[]">

      IDs of the business locations ([SDK](https://dev.wix.com/docs/sdk/backend-modules/restaurants/wix-restaurants-new/about-business-locations) | [REST](https://dev.wix.com/docs/rest/business-solutions/restaurants/wix-restaurants-new/about-business-locations)) where these item modifier groups are available.

      **Read-only.** Constraints: `format GUID`, `maxItems 100`.

    </ResponseField>

    ## Example

    <CodeGroup>

      ```json Object
      {
        "id": "116a2e4d-1ddd-4947-8050-4aea8d0af2b4",
        "revision": "1",
        "createdDate": "2023-12-04T08:20:38.784Z",
        "updatedDate": "2023-12-04T08:20:38.784Z",
        "name": "Topping",
        "modifiers": [
            {
              "id": "481135f7-1392-4a2e-ac6b-12fb0ef398b7",
              "additionalChargeInfo": {"additionalCharge": "2.50"}
            }
        ],
        "rule": {
            "mandatory": false,
            "minSelections": 0,
            "maxSelections": null
        },
        "extendedFields": null
      }
      ```

    </CodeGroup>
