<Tabs>

  <Tab title="REST">

    <ResponseField name="revision" type="integer">

      Revision number, which increments by 1 each time the payment link settings are updated.

      **Read-only.**

    </ResponseField>

    <ResponseField name="policyAgreementCheckboxDisabled" type="boolean">

      Whether the store policy agreement checkbox in the checkout page is disabled.
      Default: `false`.

    </ResponseField>

    <ResponseField name="continueShoppingButtonDisabled" type="boolean">

      Whether the continue shopping button in the checkout page is disabled.
      Default: `false`.

    </ResponseField>

  </Tab>
